import re
from collections import Counter

ERROR_RE = re.compile(r"\bERROR\b")

# Order matters: more specific patterns first.
MASKS = [
    (re.compile(r"\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}\b"), "<UUID>"),
    (re.compile(r"\b\d{1,3}(?:\.\d{1,3}){3}(?::\d{1,5})?\b"), "<IP>"),
    (re.compile(r"\b(?:[0-9a-fA-F]{1,4}:){2,7}[0-9a-fA-F]{1,4}\b"), "<IPV6>"),
    (re.compile(r"\b0x[0-9a-fA-F]+\b"), "<HEX>"),
    (re.compile(r"\b\d+(?:\.\d+)?\b"), "<NUM>"),
]


def normalize_message(msg):
    for pattern, token in MASKS:
        msg = pattern.sub(token, msg)
    return re.sub(r"\s+", " ", msg).strip()


def analyze_error_log(path, top_n=10):
    """
    Read a server log, extract ERROR lines, normalize them (mask IPs/numbers),
    and return:
      {
        "total_errors": int,
        "unique_signatures": int,
        "counts": Counter({signature: count}),
        "top": [(signature, count), ...]  # top_n most common
      }
    """
    counts = Counter()
    total = 0

    with open(path, "r", encoding="utf-8", errors="replace") as f:
        for line in f:
            m = ERROR_RE.search(line)
            if not m:
                continue
            total += 1
            # Drop timestamp/prefix; keep text after the ERROR level token.
            message = line[m.end():].lstrip(" :-]\t")
            counts[normalize_message(message)] += 1

    return {
        "total_errors": total,
        "unique_signatures": len(counts),
        "counts": counts,
        "top": counts.most_common(top_n),
    }


if __name__ == "__main__":
    import sys

    result = analyze_error_log(sys.argv[1], int(sys.argv[2]) if len(sys.argv) > 2 else 10)
    print(f"Total ERROR lines: {result['total_errors']}")
    print(f"Unique signatures: {result['unique_signatures']}")
    for sig, n in result["top"]:
        print(f"{n:>6}  {sig}")
