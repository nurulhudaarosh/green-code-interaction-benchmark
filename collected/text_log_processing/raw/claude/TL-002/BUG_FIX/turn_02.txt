```python
import csv
import re

_LOG_RE = re.compile(
    r'^(\S+)\s+\S+\s+\S+\s+\[([^\]]+)\]\s+'
    r'"(\S+)\s+(\S+)(?:\s+[^"]*)?"\s+'
    r'(\d{3})\s+(\d+|-)(?:\s.*)?$'
)

def _parse_line(line):
    """Return (ip, ts, method, url, status, status_class, size) or None if malformed."""
    m = _LOG_RE.match(line.strip())
    if not m:
        return None
    ip, ts, method, url, status, size = m.groups()
    try:
        status_int = int(status)
        size_int = 0 if size.strip() == "-" else int(size)
    except (ValueError, TypeError):
        return None
    return ip, ts, method, url, status_int, f"{status[0]}xx", size_int

def convert_access_log(in_path, out_path, methods, status_classes):
    allowed_methods = {str(m).upper() for m in methods} if methods else None
    allowed_classes = {str(c).lower() for c in status_classes} if status_classes else None
    written = skipped = 0

    with open(in_path, "r", encoding="utf-8-sig", errors="replace") as fin, \
         open(out_path, "w", newline="", encoding="utf-8") as fout:
        writer = csv.writer(fout)
        writer.writerow(["ip", "timestamp", "method", "url", "status", "statusClass", "size"])

        for line in fin:
            try:
                row = _parse_line(line)
            except Exception:
                row = None
            if row is None:
                skipped += 1
                continue

            ip, ts, method, url, status, status_class, size = row
            if allowed_methods is not None and method.upper() not in allowed_methods:
                continue
            if allowed_classes is not None and status_class not in allowed_classes:
                continue

            writer.writerow([ip, ts, method, url, status, status_class, size])
            written += 1

    return {"written": written, "skipped": skipped}
```

Parsing now lives in `_parse_line`, which returns `None` for anything it can't fully convert. Any per-line failure, including a bad int conversion, counts as skipped and the loop continues. `'-'` maps to `0` explicitly. I also made the filter normalization tolerant of non-string entries and added BOM handling.
