import re

def analyze_error_log(path, top_n=10):
    """
    Analyze a server log file for ERROR lines, normalize messages,
    and return counts by signature.
    
    Args:
        path: Path to the UTF-8 log file.
        top_n: Number of top signatures to return.
    
    Returns:
        dict with keys:
            - 'counts': dict of {signature: count}
            - 'malformed': number of blank/malformed lines
            - 'top': list of [signature, count] sorted by count desc, then sig asc,
                     limited to top_n entries.
    """
    # Regex to match a well-formed line: [timestamp] LEVEL message
    line_re = re.compile(r'^\[(?P<ts>[^\]]+)\]\s+(?P<level>[A-Z]+)\s+(?P<msg>.*)$')
    # Regex to replace IPv4 addresses
    ip_re = re.compile(r'\b(?:\d{1,3}\.){3}\d{1,3}\b')
    # Regex to replace numeric runs (including decimals)
    num_re = re.compile(r'\d+(?:\.\d+)?')
    # Regex to collapse whitespace
    ws_re = re.compile(r'\s+')

    counts = {}
    malformed = 0

    with open(path, 'r', encoding='utf-8') as f:
        for raw_line in f:
            line = raw_line.rstrip('\n').rstrip('\r')
            if not line.strip():
                malformed += 1
                continue

            m = line_re.match(line)
            if not m:
                malformed += 1
                continue

            level = m.group('level')
            if level != 'ERROR':
                continue

            msg = m.group('msg')
            # Normalize: replace IPs first, then numbers, then collapse whitespace
            msg = ip_re.sub('<IP>', msg)
            msg = num_re.sub('<N>', msg)
            msg = ws_re.sub(' ', msg).strip()

            if not msg:
                malformed += 1
                continue

            counts[msg] = counts.get(msg, 0) + 1

    # Sort by count desc, then signature asc
    sorted_items = sorted(counts.items(), key=lambda kv: (-kv[1], kv[0]))
    top = [[sig, cnt] for sig, cnt in sorted_items[:top_n]]

    return {'counts': counts, 'malformed': malformed, 'top': top}
