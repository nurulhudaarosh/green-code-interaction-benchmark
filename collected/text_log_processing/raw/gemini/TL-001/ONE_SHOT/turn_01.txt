import re
from collections import Counter
from typing import Any, Dict


def analyze_error_log(path: str, top_n: int = 10) -> Dict[str, Any]:
    # Pre-compiled regular expressions for performance
    line_pattern = re.compile(r"^\[(?P<timestamp>[^\]]+)\]\s+(?P<level>\S+)\s+(?P<message>.*)$")
    # Matches standard IPv4 addresses (4 octets of 1-3 digits separated by dots)
    ip_pattern = re.compile(r"\b(?:\d{1,3}\.){3}\d{1,3}\b")
    # Matches remaining numbers/integers
    number_pattern = re.compile(r"\b\d+\b")
    whitespace_pattern = re.compile(r"\s+")

    counts = Counter()
    malformed_count = 0

    with open(path, "r", encoding="utf-8", errors="replace") as file:
        for line in file:
            line_str = line.strip()
            if not line_str:
                malformed_count += 1
                continue

            match = line_pattern.match(line_str)
            if not match:
                malformed_count += 1
                continue

            level = match.group("level")
            message = match.group("message").strip()

            # Ensure the level is ERROR and message is present
            if level == "ERROR":
                # 1. Replace IP addresses first so numbers within IPs aren't replaced individually
                norm_msg = ip_pattern.sub("<IP>", message)
                # 2. Replace remaining standalone numbers
                norm_msg = number_pattern.sub("<N>", norm_msg)
                # 3. Collapse multiple whitespace characters into a single space
                norm_msg = whitespace_pattern.sub(" ", norm_msg).strip()

                counts[norm_msg] += 1

    # Sort by count desc (-x[1]), then signature asc (x[0])
    sorted_signatures = sorted(counts.items(), key=lambda item: (-item[1], item[0]))
    top_entries = [[sig, count] for sig, count in sorted_signatures[:top_n]]

    return {
        "counts": dict(counts),
        "malformed": malformed_count,
        "top": top_entries,
    }
