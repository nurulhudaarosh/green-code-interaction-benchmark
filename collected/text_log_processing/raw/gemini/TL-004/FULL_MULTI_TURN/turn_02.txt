def _clean_and_tokenize(text: Optional[str]) -> List[str]:
    """Handles empty inputs, joins hyphenated line breaks, and tokenizes."""
    if not text or not text.strip():
        return []

    # Rejoin words hyphenated across line breaks (e.g. "ma-\nchine" -> "machine")
    normalized_text = re.sub(r"(\b[a-zA-Z]+)-\s*\n\s*([a-zA-Z]+\b)", r"\1\2", text)

    # Convert all whitespace (newlines, tabs) to single spaces so sentences flow smoothly
    normalized_text = normalized_text.lower()

    # Match tokens including internal single quotes (e.g., "don't", "it's")
    return re.findall(r"\b[a-z]+(?:'[a-z]+)?\b", normalized_text)


def analyze_text(
    text: Optional[str],
    top_k: int = 10,
    stopwords: Optional[Iterable[str]] = None,
    min_length: int = 2,
    include_bigrams: bool = True
) -> dict:
    """
    Extracts top-k unigrams and cross-line bigrams with stopword filtering.
    Safe against empty strings, empty files, and NoneType inputs.
    """
    # 1. Guard against empty/invalid top_k
    if top_k <= 0:
        return {"unigrams": [], "bigrams": []}

    # 2. Case-insensitive stopword normalization
    if stopwords is None:
        stopword_set = DEFAULT_STOPWORDS
    else:
        stopword_set = {word.strip().lower() for word in stopwords if word}

    # 3. Tokenize with crash protection for empty text
    tokens = _clean_and_tokenize(text)
    if not tokens:
        return {"unigrams": [], "bigrams": []}

    # 4. Filter unigrams by stopword set and minimum character length
    filtered_unigrams = [
        token for token in tokens
        if token not in stopword_set and len(token) >= min_length
    ]

    unigram_counts = Counter(filtered_unigrams).most_common(top_k)

    # 5. Extract cross-line bigrams
    bigram_counts: List[Tuple[str, int]] = []
    if include_bigrams and len(filtered_unigrams) >= 2:
        bigram_list = [
            f"{filtered_unigrams[i]} {filtered_unigrams[i + 1]}"
            for i in range(len(filtered_unigrams) - 1)
        ]
        bigram_counts = Counter(bigram_list).most_common(top_k)

    return {
        "unigrams": unigram_counts,
        "bigrams": bigram_counts
    }


def analyze_file(filepath: str, top_k: int = 10, **kwargs) -> dict:
    """Safely reads and analyzes a file, handling empty files and missing files."""
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except (FileNotFoundError, IsADirectoryError):
        return {"unigrams": [], "bigrams": []}

    return analyze_text(content, top_k=top_k, **kwargs)

### Fixes Applied

1. **Case-Sensitivity:** User-provided stopwords are normalized with `{w.strip().lower() for w in stopwords}` so uppercase terms like `"THE"` or `"Into"` reliably match lowercase tokens.
2. **Crash Guards (Empty Inputs & Files):** `_clean_and_tokenize` and `analyze_file` validate `None`, zero-length strings, whitespace-only buffers, and non-existent files upfront, returning empty `[]` lists instead of raising exceptions or failing slicing operations.
3. **Cross-Line Bigrams & Hyphenation:**
* `re.sub(r"(\b[a-zA-Z]+)-\s*\n\s*([a-zA-Z]+\b)", r"\1\2", ...)` glues broken hyphenated words across line boundaries (e.g., `lan-\nguage` $\to$ `language`).
* Newlines are treated as regular whitespace during regex scanning, allowing bigrams like `("processing", "transforms")` to bridge across lines seamlessly.
