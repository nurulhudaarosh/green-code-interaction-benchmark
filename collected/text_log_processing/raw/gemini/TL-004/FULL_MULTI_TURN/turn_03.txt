Here is the refactored implementation featuring a dedicated `mode` selector (`"unigram"`, `"bigram"`, or `"all"`), configurable bigram stopword filtering strategies, and boundary handling across lines and punctuation.

```python

def _tokenize(text: Optional[str]) -> List[str]:
    """Cleans hyphenated line breaks and tokenizes into lowercase words."""
    if not text or not text.strip():
        return []

    # Rejoin words hyphenated across line breaks (e.g. "lan-\n   guage" -> "language")
    clean = re.sub(r"(\b[a-zA-Z]+)-\s*\n\s*([a-zA-Z]+\b)", r"\1\2", text)
    clean = clean.lower()

    # Extract alphabetic words including contractions
    return re.findall(r"\b[a-z]+(?:'[a-z]+)?\b", clean)


def analyze_frequencies(
    text: Optional[str],
    mode: Literal["unigram", "bigram", "all"] = "all",
    top_k: int = 10,
    stopwords: Optional[Iterable[str]] = None,
    min_length: int = 2,
    bigram_filter: Literal["any", "all", "none"] = "any"
) -> Union[List[Tuple[str, int]], Dict[str, List[Tuple[str, int]]]]:
    """
    Extracts top-k unigrams and/or bigrams with flexible filtering.

    :param text: Raw text input or multiline string.
    :param mode: "unigram", "bigram", or "all".
    :param top_k: Max frequency entries to return per mode.
    :param stopwords: Words to exclude (case-insensitive).
    :param min_length: Minimum character length for tokens.
    :param bigram_filter: 
        - "any": Exclude bigram if either word is a stopword (standard collocations).
        - "all": Exclude bigram only if both words are stopwords.
        - "none": Keep all valid token pairs regardless of stopwords.
    :return: List of (token, count) for single modes, or Dict with both lists for "all".
    """
    empty_result: Dict[str, List[Tuple[str, int]]] = {"unigrams": [], "bigrams": []}
    if top_k <= 0 or not text:
        return empty_result if mode == "all" else []

    # Normalize stopwords (handles case and whitespace)
    stopword_set = (
        DEFAULT_STOPWORDS
        if stopwords is None
        else {w.strip().lower() for w in stopwords if w and w.strip()}
    )

    tokens = _tokenize(text)
    if not tokens:
        return empty_result if mode == "all" else []

    # Filtered tokens for minimum word length
    length_valid_tokens = [t for t in tokens if len(t) >= min_length]

    # --- Unigram Computation ---
    unigram_results: List[Tuple[str, int]] = []
    if mode in ("unigram", "all"):
        filtered_unigrams = [t for t in length_valid_tokens if t not in stopword_set]
        unigram_results = Counter(filtered_unigrams).most_common(top_k)
        if mode == "unigram":
            return unigram_results

    # --- Bigram Computation ---
    bigram_results: List[Tuple[str, int]] = []
    if mode in ("bigram", "all"):
        bigrams: List[str] = []
        for i in range(len(length_valid_tokens) - 1):
            w1, w2 = length_valid_tokens[i], length_valid_tokens[i + 1]
            w1_stop, w2_stop = w1 in stopword_set, w2 in stopword_set

            if bigram_filter == "any" and (w1_stop or w2_stop):
                continue
            elif bigram_filter == "all" and (w1_stop and w2_stop):
                continue

            bigrams.append(f"{w1} {w2}")

        bigram_results = Counter(bigrams).most_common(top_k)
        if mode == "bigram":
            return bigram_results

    return {
        "unigrams": unigram_results,
        "bigrams": bigram_results
    }

```

if __name__ == "__main__":

    top_bigrams = analyze_frequencies(
        text_data,
        mode="bigram",

    )
    print("--- Top 3 Bigrams ---")

    # 2. Combined mode with empty input safety test
    combined = analyze_frequencies(text_data, mode="all", top_k=2)
    print("\n--- Combined (mode='all') ---")
    print("Unigrams:", combined["unigrams"])
    print("Bigrams: ", combined["bigrams"])

    # 3. Safe fallback on empty string
    print("\n--- Empty String Test ---")
    print(analyze_frequencies("", mode="bigram", top_k=5))

### Highlights

* **Explicit Mode Switching:** `mode="bigram"` returns a direct `List[Tuple[str, int]]` of bigram phrases instead of forcing dictionary unwrapping when you only need bigrams.
* **Stopword Filtering Control (`bigram_filter`):**
* `"any"`: Drops pairs where either word is a stopword (e.g., skips `"models and"`, preserves `"language models"`).
* `"all"`: Keeps phrases containing transitional prepositions (e.g., preserves `"state of the"`-style constructions while filtering double-stops like `"and the"`).


* **Sequential Flow Across Lines:** Line breaks inside bigrams (e.g., `machine\nlearning`) and hyphen breaks (`learn-\ning`) resolve back to canonical single-space tokens.
