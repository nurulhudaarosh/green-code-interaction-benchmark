```python
import re
from collections import Counter

def unigram_freq(path, top_k=20):
    counts = Counter()
    total_tokens = 0

    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            words = re.findall(r"[a-z0-9]+", line.lower())
            counts.update(words)
            total_tokens += len(words)

    top_words = sorted(counts.items(), key=lambda x: (-x[1], x[0]))[:top_k]

    return {
        "top_k": top_words,
        "total_tokens": total_tokens
    }
```
