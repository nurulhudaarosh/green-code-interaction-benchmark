```python
import re
from collections import Counter


def keyword_freq(paths, stopwords=(), top_k=20, include_bigrams=False):
    stopwords = {word.lower() for word in stopwords}
    token_pattern = re.compile(r"[a-z0-9]+")

    unigrams = Counter()
    bigrams = Counter()
    total_tokens = 0

    for path in paths:
        prev_token = None

        with open(path, "r", encoding="utf-8") as f:
            for line in f:
                tokens = [
                    token.lower()
                    for token in token_pattern.findall(line)
                    if token.lower() not in stopwords
                ]

                total_tokens += len(tokens)
                unigrams.update(tokens)

                if include_bigrams:
                    for token in tokens:
                        if prev_token is not None:
                            bigrams[(prev_token, token)] += 1
                        prev_token = token

    def top_k_items(counter):
        # Safely handles empty corpus, all-stopword files,
        # and top_k larger than the vocabulary.
        return sorted(
            counter.items(),
            key=lambda item: (-item[1], item[0])
        )[:max(0, top_k)]

    result = {
        "unigrams": top_k_items(unigrams),
        "total_tokens": total_tokens,
    }

    if include_bigrams:
        result["bigrams"] = top_k_items(bigrams)

    return result
```
