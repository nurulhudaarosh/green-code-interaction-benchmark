import pandas as pd
import numpy as np
import hashlib


def process_sensor_timeseries(
    readings: list[dict],
    sensor_key: str = "sensor_id",
    timestamp_key: str = "timestamp",
    value_key: str = "value",
    freq: str = "1h",
) -> pd.DataFrame:
    """
    Sort readings per sensor, fill missing interior hourly timestamps by linear
    interpolation, mark inserted rows, and return deterministic output.

    - Does NOT extrapolate before the first or after the last observation
      per sensor (only interior gaps are filled).
    - Handles consecutive missing hours via linear interpolation across the
      whole contiguous interior gap.
    - Deterministic output: sorted by (sensor_id, timestamp), stable columns,
      stable dtypes.
    """
    columns = [sensor_key, timestamp_key, value_key, "is_interpolated"]

    if not readings:
        return pd.DataFrame(columns=columns)

    df = pd.DataFrame(readings)
    df[timestamp_key] = pd.to_datetime(df[timestamp_key], utc=True, errors="coerce")
    df[value_key] = pd.to_numeric(df[value_key], errors="coerce")
    df = df.dropna(subset=[sensor_key, timestamp_key, value_key])

    if df.empty:
        return pd.DataFrame(columns=columns)

    # Deterministic pre-sort
    df = df.sort_values([sensor_key, timestamp_key], kind="mergesort").reset_index(
        drop=True
    )

    output_frames = []

    # groupby(sort=True) for deterministic sensor ordering
    for sensor, group in df.groupby(sensor_key, sort=True):
        group = group.sort_values(timestamp_key, kind="mergesort").reset_index(
            drop=True
        )

        # Deduplicate identical timestamps deterministically (first wins)
        group = group.drop_duplicates(subset=[timestamp_key], keep="first")

        series = group.set_index(timestamp_key)[value_key]

        # Full hourly grid spanning only [first_obs, last_obs] → no extrapolation
        full_index = pd.date_range(
            start=series.index.min(),
            end=series.index.max(),
            freq=freq,
            tz=series.index.tz,
        )

        reindexed = series.reindex(full_index)

        # Rows that were inserted (missing in the original data)
        is_interpolated = reindexed.isna().to_numpy()

        # Linear interpolation handles consecutive missing hours:
        # each NaN is filled by linear blend between the nearest known
        # neighbors. limit_area="inside" guarantees we never extrapolate
        # beyond the first/last observation (they are always known here).
        interpolated = reindexed.interpolate(method="linear", limit_area="inside")

        # Safety net: any residual NaN (shouldn't happen) is dropped
        valid_mask = interpolated.notna().to_numpy()
        interpolated = interpolated[valid_mask]
        is_interpolated = is_interpolated[valid_mask]

        result = pd.DataFrame(
            {
                sensor_key: sensor,
                timestamp_key: interpolated.index,
                value_key: interpolated.to_numpy(),
                "is_interpolated": is_interpolated,
            }
        )
        output_frames.append(result)

    final = pd.concat(output_frames, ignore_index=True)

    # Deterministic final ordering + column order + dtypes
    final = final.sort_values(
        [sensor_key, timestamp_key], kind="mergesort"
    ).reset_index(drop=True)
    final = final[columns]
    final["is_interpolated"] = final["is_interpolated"].astype(bool)
    final[value_key] = final[value_key].astype(float)

    return final


def write_deterministic_csv(df: pd.DataFrame, path: str) -> str:
    """
    Write DataFrame to CSV deterministically:
      - fixed timestamp format
      - fixed float precision
      - LF line endings
    Returns SHA-256 of the written file.
    """
    out = df.copy()
    if "value" in out.columns:
        out["value"] = out["value"].map(lambda x: f"{float(x):.6f}")
    if "timestamp" in out.columns:
        out["timestamp"] = out["timestamp"].dt.strftime("%Y-%m-%dT%H:%M:%S%z")

    out.to_csv(path, index=False, lineterminator="\n")

    with open(path, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()

if __name__ == "__main__":
    sample = [
        # Sensor A: gap of 2 consecutive missing hours (01:00, 02:00)
        {"sensor_id": "A", "timestamp": "2024-01-01T00:00:00Z", "value": 10.0},
        {"sensor_id": "A", "timestamp": "2024-01-01T03:00:00Z", "value": 16.0},
        # Sensor B: leading gap (before first obs) is NOT filled,
        # trailing gap (after last obs) is NOT filled,
        # and an interior 2-hour gap (02:00, 03:00) IS filled
        {"sensor_id": "B", "timestamp": "2024-01-01T01:00:00Z", "value": 7.0},
        {"sensor_id": "B", "timestamp": "2024-01-01T04:00:00Z", "value": 13.0},
        {"sensor_id": "B", "timestamp": "2024-01-01T05:00:00Z", "value": 15.0},
        # Sensor C: single interior gap
        {"sensor_id": "C", "timestamp": "2024-01-01T00:00:00Z", "value": 2.0},
        {"sensor_id": "C", "timestamp": "2024-01-01T02:00:00Z", "value": 6.0},
    ]

    df = process_sensor_timeseries(sample)
    print(df.to_string(index=False))

    digest = write_deterministic_csv(df, "sensor_output.csv")
    print("\nSHA-256:", digest)
