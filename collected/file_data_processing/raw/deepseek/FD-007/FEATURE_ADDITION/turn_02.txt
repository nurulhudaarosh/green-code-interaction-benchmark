from datetime import datetime, timedelta
from collections import defaultdict
import csv
import sys


def parse_timestamp(ts):
    """Parse a timestamp string into a datetime object.
    Supports ISO format and common formats.
    """
    ts = ts.strip()
    formats = [
        "%Y-%m-%dT%H:%M:%S",
        "%Y-%m-%d %H:%M:%S",
        "%Y-%m-%dT%H:%M:%S.%f",
        "%Y-%m-%d %H:%M:%S.%f",
        "%Y/%m/%d %H:%M:%S",
        "%d-%m-%Y %H:%M:%S",
    ]
    for fmt in formats:
        try:
            return datetime.strptime(ts, fmt)
        except ValueError:
            continue
    try:
        return datetime.fromisoformat(ts)
    except ValueError:
        raise ValueError(f"Unrecognized timestamp format: {ts}")


def load_events(path):
    """Load events from a CSV file with columns: user,event,timestamp."""
    events = []
    with open(path, "r", newline="") as f:
        reader = csv.DictReader(f)
        for row in reader:
            user = row["user"].strip()
            event = row["event"].strip().lower()
            ts = parse_timestamp(row["timestamp"])
            events.append((user, event, ts))
    return events


def build_sessions(events):
    """For each user, sort events chronologically, pair logins with next valid
    logout, and return raw sessions (start, end)."""
    by_user = defaultdict(list)
    for user, event, ts in events:
        by_user[user].append((ts, event))

    raw_sessions = defaultdict(list)

    for user, evs in by_user.items():
        evs.sort(key=lambda x: x[0])
        pending_login = None
        for ts, event in evs:
            if event == "login":
                if pending_login is None:
                    pending_login = ts
            elif event == "logout":
                if pending_login is not None:
                    raw_sessions[user].append((pending_login, ts))
                    pending_login = None

    return raw_sessions


def merge_sessions(sessions):
    """Merge overlapping or touching sessions. Sessions are (start, end) tuples."""
    if not sessions:
        return []
    sessions = sorted(sessions, key=lambda s: s[0])
    merged = [sessions[0]]
    for start, end in sessions[1:]:
        last_start, last_end = merged[-1]
        if start <= last_end:  # overlapping or touching
            merged[-1] = (last_start, max(last_end, end))
        else:
            merged.append((start, end))
    return merged


def active_seconds(sessions):
    return sum(int((end - start).total_seconds()) for start, end in sessions)


def analyze(events):
    raw_sessions = build_sessions(events)
    summaries = []
    for user, sessions in raw_sessions.items():
        merged = merge_sessions(sessions)
        summaries.append({
            "user": user,
            "sessions": merged,
            "merged_interval_count": len(merged),
            "active_seconds": active_seconds(merged),
        })
    summaries.sort(key=lambda x: x["user"])
    return summaries


def write_summaries(summaries, out_path):
    with open(out_path, "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow([
            "user",
            "merged_interval_count",
            "active_seconds",
            "sessions",
        ])
        for s in summaries:
            sessions_str = "; ".join(
                f"{start.isoformat()} -> {end.isoformat()}"
                for start, end in s["sessions"]
            )
            writer.writerow([
                s["user"],
                s["merged_interval_count"],
                s["active_seconds"],
                sessions_str,
            ])


def main():
    if len(sys.argv) != 3:
        print("Usage: python analyzer.py <input.csv> <output.csv>")
        sys.exit(1)

    input_path, output_path = sys.argv[1], sys.argv[2]
    events = load_events(input_path)
    summaries = analyze(events)
    write_summaries(summaries, output_path)

    for s in summaries:
        print(f"{s['user']}: {s['merged_interval_count']} merged interval(s), "
              f"{s['active_seconds']} active seconds")


if __name__ == "__main__":
    main()
