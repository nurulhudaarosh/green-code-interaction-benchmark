```python
import sys
import re
from pathlib import Path
from collections import Counter


def analyze_text_files(root):
    total_frequency = Counter()
    document_frequency = Counter()

    files = sorted(
        path for path in Path(root).rglob("*.txt")
        if path.is_file() and not path.is_symlink()
    )

    word_pattern = re.compile(r"[^\W\d_]+", re.UNICODE)

    for file_path in files:
        try:
            text = file_path.read_text(encoding="utf-8")
        except (OSError, UnicodeError):
            continue

        words = [
            match.group(0).casefold()
            for match in word_pattern.finditer(text)
        ]

        if not words:
            continue

        total_frequency.update(words)

        for word in set(words):
            document_frequency[word] += 1

    ranked_words = sorted(
        total_frequency,
        key=lambda word: (
            -document_frequency[word],
            -total_frequency[word],
            word
        )
    )

    for word in ranked_words:
        print(
            f"{word}\t"
            f"{total_frequency[word]}\t"
            f"{document_frequency[word]}"
        )


def main():
    if len(sys.argv) != 2:
        print("Usage: python analyzer.py <directory>")
        return

    analyze_text_files(sys.argv[1])


if __name__ == "__main__":
    main()
```
