```python
import csv
import sys
from collections import defaultdict

def main():
    customer_file = sys.argv[1]
    transaction_file = sys.argv[2]

    customers = {}

    with open(customer_file, "r", newline="", encoding="utf-8") as file:
        reader = csv.DictReader(file)

        for row in reader:
            customer_id = row["customer_id"]
            customers[customer_id] = row

    totals = defaultdict(float)

    with open(transaction_file, "r", newline="", encoding="utf-8") as file:
        reader = csv.DictReader(file)

        for row in reader:
            customer_id = row["customer_id"]

            if customer_id not in customers:
                continue

            try:
                amount = float(row["amount"])
            except (ValueError, TypeError):
                continue

            totals[customer_id] += amount

    fieldnames = list(next(iter(customers.values())).keys()) + ["transaction_total"]

    writer = csv.DictWriter(sys.stdout, fieldnames=fieldnames)
    writer.writeheader()

    for customer_id in sorted(customers):
        output = customers[customer_id].copy()
        output["transaction_total"] = totals.get(customer_id, 0)
        writer.writerow(output)

if __name__ == "__main__":
    main()
```
