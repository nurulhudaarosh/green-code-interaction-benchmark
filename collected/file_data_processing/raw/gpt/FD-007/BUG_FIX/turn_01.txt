```python
import csv
from datetime import datetime

INPUT_FILE = "events.csv"
OUTPUT_FILE = "user_summaries.csv"

def parse_time(value):
    return datetime.fromisoformat(value.strip())

events = {}

with open(INPUT_FILE, "r", newline="", encoding="utf-8") as file:
    reader = csv.DictReader(file)

    for row in reader:
        user = row["user"].strip()
        event = row["event"].strip().lower()
        timestamp = parse_time(row["timestamp"])

        if user not in events:
            events[user] = []

        events[user].append((timestamp, event))

summaries = []

for user in sorted(events):
    user_events = sorted(events[user], key=lambda x: x[0])
    sessions = []
    login_time = None

    for timestamp, event in user_events:
        if event == "login":
            if login_time is None:
                login_time = timestamp

        elif event == "logout":
            if login_time is not None and timestamp > login_time:
                sessions.append((login_time, timestamp))
                login_time = None

    merged = []

    for start, end in sessions:
        if not merged or start > merged[-1][1]:
            merged.append([start, end])
        elif start <= merged[-1][1]:
            if end > merged[-1][1]:
                merged[-1][1] = end

    active_seconds = sum(
        (end - start).total_seconds()
        for start, end in merged
    )

    summaries.append({
        "user": user,
        "sessions": len(merged),
        "active_seconds": int(active_seconds)
    })

with open(OUTPUT_FILE, "w", newline="", encoding="utf-8") as file:
    writer = csv.DictWriter(
        file,
        fieldnames=["user", "sessions", "active_seconds"]
    )
    writer.writeheader()
    writer.writerows(summaries)
```
