```python
import json
import sys
from collections import defaultdict


def analyze_ndjson(input_file, output_file):
    users = defaultdict(lambda: {
        "request_count": 0,
        "error_count": 0,
        "latencies": [],
        "endpoints": defaultdict(int)
    })

    malformed_count = 0

    with open(input_file, "r", encoding="utf-8") as file:
        for line in file:
            line = line.strip()

            if not line:
                continue

            try:
                data = json.loads(line)

                user = data["user"]
                request = data["request"]

                user_id = str(user["id"])
                endpoint = str(request["endpoint"])
                latency = float(request["latency"])
                status = request.get("status")

            except (json.JSONDecodeError, KeyError, TypeError, ValueError):
                malformed_count += 1
                continue

            users[user_id]["request_count"] += 1
            users[user_id]["latencies"].append(latency)
            users[user_id]["endpoints"][endpoint] += 1

            if status == "error":
                users[user_id]["error_count"] += 1

    result = {
        "malformed_count": malformed_count,
        "users": []
    }

    for user_id in sorted(users):
        user_data = users[user_id]

        max_count = max(user_data["endpoints"].values())

        most_requested_endpoint = min(
            endpoint
            for endpoint, count in user_data["endpoints"].items()
            if count == max_count
        )

        average_latency = (
            sum(user_data["latencies"]) / len(user_data["latencies"])
        )

        result["users"].append({
            "user_id": user_id,
            "request_count": user_data["request_count"],
            "error_count": user_data["error_count"],
            "average_latency": average_latency,
            "most_requested_endpoint": most_requested_endpoint
        })

    with open(output_file, "w", encoding="utf-8") as file:
        json.dump(
            result,
            file,
            indent=2,
            ensure_ascii=False,
            sort_keys=False
        )


if __name__ == "__main__":
    if len(sys.argv) != 3:
        sys.exit(1)

    analyze_ndjson(sys.argv[1], sys.argv[2])
```
