Here is a complete, production-grade Python script that implements the event pivot utility using `pandas`. It validates input records against required fields and types, dynamically pivots event types, fills missing counts with zeros, and sorts the output deterministically.

```python
from datetime import datetime
from typing import Any, Dict, List
import pandas as pd


def pivot_event_data(
    records: List[Dict[str, Any]],
    date_col: str = "date",
    region_col: str = "region",
    event_type_col: str = "event_type",
) -> pd.DataFrame:
    """Validates, aggregates, pivots, and sorts event records.

    Args:
        records: List of dictionary records representing raw events.
        date_col: Field name for event date (ISO format string YYYY-MM-DD or
          datetime).
        region_col: Field name for event region.
        event_type_col: Field name for event classification.

    Returns:
        pd.DataFrame: Pivoted DataFrame indexed by [date, region] with event
        type columns.
    """
    if not records:
        return pd.DataFrame()

    # 1. Input Validation
    required_keys = {date_col, region_col, event_type_col}
    validated_records = []

    for idx, rec in enumerate(records):
        if not isinstance(rec, dict):
            raise TypeError(f"Record at index {idx} must be a dictionary.")

        missing_keys = required_keys - rec.keys()
        if missing_keys:
            raise ValueError(
                f"Record at index {idx} missing required fields: {missing_keys}"
            )

        # Validate date format / parsing
        raw_date = rec[date_col]
        if isinstance(raw_date, str):
            try:
                parsed_date = datetime.strptime(raw_date, "%Y-%m-%d").date()
            except ValueError:
                raise ValueError(
                    f"Record at index {idx} has invalid date '{raw_date}'. Expected 'YYYY-MM-DD'."
                )
        elif isinstance(raw_date, datetime):
            parsed_date = raw_date.date()
        else:
            raise TypeError(
                f"Record at index {idx} date must be string or datetime, got {type(raw_date)}."
            )

        # Validate region and event type
        region = str(rec[region_col]).strip()
        event_type = str(rec[event_type_col]).strip()

        if not region or not event_type:
            raise ValueError(
                f"Record at index {idx} contains empty region or event_type."
            )

        validated_records.append({
            date_col: parsed_date,
            region_col: region,
            event_type_col: event_type,
            "count": 1,
        })

    # 2. Convert to DataFrame
    df = pd.DataFrame(validated_records)

    # 3. Pivot and Aggregate Counts (Fills missing combinations with 0)
    pivot_df = pd.pivot_table(
        df,
        values="count",
        index=[date_col, region_col],
        columns=[event_type_col],
        aggfunc="sum",
        fill_value=0,
    )

    # 4. Sort deterministic column names and index rows
    pivot_df = pivot_df.reindex(columns=sorted(pivot_df.columns))
    pivot_df = pivot_df.sort_index(ascending=[True, True])

    # Remove name of columns index for cleaner output structure
    pivot_df.columns.name = None
    pivot_df = pivot_df.reset_index()

    return pivot_df


# --- Example Usage ---
```
