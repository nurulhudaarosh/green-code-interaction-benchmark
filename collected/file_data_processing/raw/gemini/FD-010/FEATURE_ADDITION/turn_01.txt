import json
from typing import Any, Dict, List, Tuple


def process_product_data(
    json_data: str,
) -> Tuple[List[Dict[str, Any]], int]:
    """Reads nested product JSON, validates fields, flattens warehouse entries,

    preserves zero stock, tracks invalid counts, and sorts output rows.
    """
    products = json.loads(json_data)
    if not isinstance(products, list):
        raise ValueError("Expected root JSON element to be a list of products.")

    flattened_rows: List[Dict[str, Any]] = []
    invalid_count: int = 0

    for product in products:
        # Validate required top-level product structure
        if not isinstance(product, dict):
            invalid_count += 1
            continue

        product_id = product.get("id")
        product_name = product.get("name")
        warehouses = product.get("warehouses")

        if (
            product_id is None
            or product_name is None
            or not isinstance(warehouses, list)
        ):
            invalid_count += 1
            continue

        product_valid = True
        temp_rows = []

        # Validate and flatten warehouse entries
        for wh in warehouses:
            if not isinstance(wh, dict):
                product_valid = False
                break

            wh_id = wh.get("warehouse_id")
            stock = wh.get("stock")

            # Validate field presence (explicitly allowing stock == 0)
            if wh_id is None or stock is None or type(stock) not in (int, float):
                product_valid = False
                break

            temp_rows.append(
                {
                    "product_id": product_id,
                    "product_name": product_name,
                    "warehouse_id": wh_id,
                    "stock": stock,
                }
            )

        if not product_valid:
            invalid_count += 1
        else:
            flattened_rows.extend(temp_rows)

    # Sort rows by product_id primary, warehouse_id secondary
    flattened_rows.sort(key=lambda x: (x["product_id"], x["warehouse_id"]))

    return flattened_rows, invalid_count


# Example Usage
if __name__ == "__main__":
    sample_json = """
    [
        {
            "id": "P101",
            "name": "Wireless Mouse",
            "warehouses": [
                {"warehouse_id": "WH-A", "stock": 50},
                {"warehouse_id": "WH-B", "stock": 0}
            ]
        },
        {
            "id": "P102",
            "name": "Mechanical Keyboard",
            "warehouses": [
                {"warehouse_id": "WH-A", "stock": 12}
            ]
        },
        {
            "id": "P103",
            "name": "Invalid Product Missing Stock",
            "warehouses": [
                {"warehouse_id": "WH-C"}
            ]
        },
        {
            "id": "P104",
            "name": "Monitor",
            "warehouses": [
                {"warehouse_id": "WH-B", "stock": 5}
            ]
        }
    ]
    """

    rows, invalid_products = process_product_data(sample_json)

    print(f"Invalid Products Count: {invalid_products}\n")
    print("Flattened & Sorted Rows:")
    print(json.dumps(rows, indent=2))
