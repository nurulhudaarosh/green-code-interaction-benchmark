import json
from typing import Any, Dict, List, Tuple


def process_product_catalog(json_data: Any) -> Tuple[List[Dict[str, Any]], int]:
    """
    Parses, validates, flattens, and sorts product data from nested JSON.

    Skips malformed top-level elements, missing required fields, or invalid
    types without crashing or terminating processing. Keeps valid entries
    where stock is 0.

    Returns:
        Tuple[List[Dict[str, Any]], int]: (flattened_rows, invalid_product_count)
    """
    invalid_count = 0
    flattened_rows: List[Dict[str, Any]] = []

    # 1. Parse JSON string if input is raw text; catch JSON errors gracefully
    if isinstance(json_data, str):
        try:
            data = json.loads(json_data)
        except (json.JSONDecodeError, TypeError):
            return [], 1  # Entire input was malformed JSON
    else:
        data = json_data

    # 2. Ensure top-level structure is a list
    if not isinstance(data, list):
        return [], 1

    # 3. Process each product entry safely
    for product in data:
        try:
            # Must be a dictionary object
            if not isinstance(product, dict):
                invalid_count += 1
                continue

            product_id = product.get("id")
            product_name = product.get("name")
            warehouses = product.get("warehouses")

            # Validate required top-level fields
            if product_id is None or not product_name or not isinstance(warehouses, list):
                invalid_count += 1
                continue

            category = product.get("category", "Uncategorized")
            product_rows = []
            is_valid = True

            # Validate nested warehouse list
            for wh in warehouses:
                if not isinstance(wh, dict):
                    is_valid = False
                    break

                wh_location = wh.get("location")
                stock = wh.get("stock")

                # Validate required warehouse fields (stock can be 0, but must be int >= 0)
                if not wh_location or stock is None or not isinstance(stock, int) or stock < 0:
                    is_valid = False
                    break

                product_rows.append({
                    "product_id": product_id,
                    "product_name": product_name,
                    "category": category,
                    "warehouse_location": wh_location,
                    "stock": stock
                })

            if is_valid:
                flattened_rows.extend(product_rows)
            else:
                invalid_count += 1

        except Exception:
            # Catch unexpected structure/parsing issues per item and skip gracefully
            invalid_count += 1
            continue

    # 4. Sort resulting rows by product_id (asc) then warehouse_location (asc)
    flattened_rows.sort(key=lambda x: (x["product_id"], x["warehouse_location"]))

    return flattened_rows, invalid_count
