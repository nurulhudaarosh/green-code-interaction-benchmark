import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation
from typing import Any, Dict, Optional, Tuple


def _parse_timestamp(ts_str: str) -> datetime:
    """
    Parse an ISO 8601 timestamp (including optional trailing 'Z').
    Uses Python's standard library parser (no external dependencies).
    """
    cleaned = ts_str.strip()
    if cleaned.endswith("Z") or cleaned.endswith("z"):
        cleaned = cleaned[:-1] + "+00:00"
    return datetime.fromisoformat(cleaned)


def process_customer_events(
    input_file: str,
    output_file: str,
    customer_id_col: str = "customer_id",
    timestamp_col: str = "timestamp",
    status_col: str = "status",
    amount_col: str = "amount",
) -> Dict[str, Dict[str, Any]]:
    """
    Processes a customer events CSV file completely offline:
      1. Deduplicates (customer_id, timestamp) pairs, keeping the last occurrence.
      2. Retains only the chronologically latest event for each customer.
      3. Calculates count and total monetary amount grouped by status for retained records.
      4. Writes the final retained records to output_file, sorted deterministically by customer_id.

    Returns:
      A summary dictionary mapping status -> {"count": int, "total_amount": Decimal}
    """
    # ---------------------------------------------------------
    # Step 1: Read CSV and deduplicate (customer_id, timestamp)
    # Keeping the last occurrence by overwriting dictionary keys.
    # ---------------------------------------------------------
    deduped_events: Dict[Tuple[str, str], Tuple[int, Dict[str, str]]] = {}
    fieldnames = []

    with open(input_file, mode="r", encoding="utf-8", newline="") as infile:
        reader = csv.DictReader(infile)
        fieldnames = reader.fieldnames or []
        for line_idx, row in enumerate(reader):
            cust_id = row[customer_id_col].strip()
            ts_raw = row[timestamp_col].strip()
            deduped_events[(cust_id, ts_raw)] = (line_idx, row)

    # ---------------------------------------------------------
    # Step 2: Select the chronologically latest record per customer
    # Breaks timestamp ties using file appearance order (line_idx).
    # ---------------------------------------------------------
    # Mapping: customer_id -> (parsed_datetime, line_index, row_dict)
    latest_by_customer: Dict[str, Tuple[datetime, int, Dict[str, str]]] = {}

    for (cust_id, _), (line_idx, row) in deduped_events.items():
        parsed_dt = _parse_timestamp(row[timestamp_col])

        if cust_id not in latest_by_customer:
            latest_by_customer[cust_id] = (parsed_dt, line_idx, row)
        else:
            prev_dt, prev_idx, _ = latest_by_customer[cust_id]
            # Select newer timestamp; if timestamps match, take the later occurrence
            if (parsed_dt > prev_dt) or (parsed_dt == prev_dt and line_idx > prev_idx):
                latest_by_customer[cust_id] = (parsed_dt, line_idx, row)

    # ---------------------------------------------------------
    # Step 3: Compute aggregations (Count & Total Amount by Status)
    # ---------------------------------------------------------
    status_summary: Dict[str, Dict[str, Any]] = {}

    for _, (_, _, row) in latest_by_customer.items():
        status = row.get(status_col, "UNKNOWN").strip()
        raw_amt = row.get(amount_col, "0").strip()

        try:
            amt = Decimal(raw_amt)
        except (InvalidOperation, TypeError):
            amt = Decimal("0.00")

        if status not in status_summary:
            status_summary[status] = {"count": 0, "total_amount": Decimal("0.00")}

        status_summary[status]["count"] += 1
        status_summary[status]["total_amount"] += amt

    # ---------------------------------------------------------
    # Step 4: Write deterministic output sorted by customer_id
    # ---------------------------------------------------------
    sorted_customer_ids = sorted(latest_by_customer.keys())

    with open(output_file, mode="w", encoding="utf-8", newline="") as outfile:
        writer = csv.DictWriter(outfile, fieldnames=fieldnames)
        writer.writeheader()
        for cust_id in sorted_customer_ids:
            _, _, row = latest_by_customer[cust_id]
            writer.writerow(row)

    return status_summary


# =============================================================

if __name__ == "__main__":
    sample_csv_data = """customer_id,timestamp,status,amount

    input_path = "sample_events.csv"
    output_path = "latest_customer_events.csv"

    # Write test CSV

    # Process
    summary = process_customer_events(input_path, output_path)

    # Print aggregated results
    print("--- Status Summary (Retained Records) ---")
    for status, metrics in sorted(summary.items()):
        print(f"Status: {status:<10} | Count: {metrics['count']} | Total Amount: {metrics['total_amount']}")

    # Print output file contents
    print("\n--- Output CSV Contents ---")
    with open(output_path, "r", encoding="utf-8") as f:
        print(f.read())
