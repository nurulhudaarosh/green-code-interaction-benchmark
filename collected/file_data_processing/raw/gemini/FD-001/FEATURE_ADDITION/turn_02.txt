import csv
from collections import defaultdict
from datetime import datetime
from decimal import Decimal
from typing import Dict, List, Tuple, Any, Optional


def process_customer_events_with_summary(
    input_filepath: str, 
    output_filepath: str,
    summary_output_filepath: Optional[str] = None
) -> Dict[str, Dict[str, Any]]:
    """
    Processes customer event CSV records offline:
      1. Deduplicates (customer_id, timestamp) by keeping the last occurrence.
      2. Selects the chronologically latest record for each customer.
      3. Calculates count and total amount grouped by status on the retained records.
      4. Writes the retained customer records sorted deterministically by customer_id.
      5. Optionally writes a status summary CSV without altering customer-level output.
    """
    deduped_events: Dict[Tuple[str, str], dict] = {}
    fieldnames: List[str] = []

    # Step 1: Deduplicate (customer_id, timestamp) - keep last occurrence
    with open(input_filepath, mode="r", encoding="utf-8", newline="") as infile:
        reader = csv.DictReader(infile)
        fieldnames = reader.fieldnames or []
        for row in reader:
            key = (row["customer_id"].strip(), row["timestamp"].strip())
            deduped_events[key] = row

    # Step 2: Retain only the chronologically latest record per customer
    customer_latest: Dict[str, dict] = {}
    customer_latest_dt: Dict[str, datetime] = {}

    for row in deduped_events.values():
        cid = row["customer_id"].strip()
        ts_str = row["timestamp"].strip()
        try:
            dt = datetime.fromisoformat(ts_str)
        except ValueError:
            dt = datetime.strptime(ts_str, "%Y-%m-%d %H:%M:%S")

        if cid not in customer_latest_dt or dt > customer_latest_dt[cid]:
            customer_latest_dt[cid] = dt
            customer_latest[cid] = row

    # Step 3: Calculate status aggregation (count & total amount) on retained records
    status_summary_acc: Dict[str, Dict[str, Any]] = defaultdict(
        lambda: {"retained_count": 0, "total_amount": Decimal("0.00")}
    )

    for row in customer_latest.values():
        status = row.get("status", "UNKNOWN").strip()
        amount_raw = row.get("amount", "0").strip()
        amount = Decimal(amount_raw) if amount_raw else Decimal("0.00")

        status_summary_acc[status]["retained_count"] += 1
        status_summary_acc[status]["total_amount"] += amount

    # Step 4: Write customer-level output sorted deterministically by customer_id
    sorted_records = [customer_latest[cid] for cid in sorted(customer_latest.keys())]

    with open(output_filepath, mode="w", encoding="utf-8", newline="") as outfile:
        writer = csv.DictWriter(outfile, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(sorted_records)

    # Step 5: Format status summary
    formatted_summary = {
        status: {
            "retained_count": data["retained_count"],
            "total_amount": f"{data['total_amount']:.2f}"
        }
        for status, data in sorted(status_summary_acc.items())
    }

    # Optional: Write status summary to its own dedicated CSV
    if summary_output_filepath:
        summary_fieldnames = ["status", "retained_count", "total_amount"]
        with open(summary_output_filepath, mode="w", encoding="utf-8", newline="") as s_outfile:
            writer = csv.DictWriter(s_outfile, fieldnames=summary_fieldnames)
            writer.writeheader()
            for status, metrics in formatted_summary.items():
                writer.writerow({
                    "status": status,
                    "retained_count": metrics["retained_count"],
                    "total_amount": metrics["total_amount"]
                })

    return formatted_summary


if __name__ == "__main__":
    sample_csv = """customer_id,timestamp,status,amount

"""

    with open("input_events.csv", "w", encoding="utf-8") as f:
        f.write(sample_csv.strip())

    summary = process_customer_events_with_summary(
        input_filepath="input_events.csv",
        output_filepath="customer_records.csv",
        summary_output_filepath="status_summary.csv"
    )

    print("Status Summary (Retained Records Only):")
    print(f"{'Status':<12} {'Retained Count':<16} {'Total Amount':<12}")
    print("-" * 40)
    for status, metrics in summary.items():
        print(f"{status:<12} {metrics['retained_count']:<16} {metrics['total_amount']:<12}")
