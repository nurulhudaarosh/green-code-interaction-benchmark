import csv
from collections import defaultdict
from datetime import datetime
from decimal import Decimal
from typing import Any, Dict, List, Tuple


def process_customer_events(
    input_filepath: str,
    output_filepath: str,
    timestamp_format: str = "%Y-%m-%d %H:%M:%S",
) -> Dict[str, Dict[str, Any]]:
    """Process customer event CSV files.

    Steps performed:
    1. Removes duplicate (customer_id, timestamp) pairs, keeping the last occurrence.
    2. Selects the chronologically latest record for each customer.
    3. Calculates count and total amount grouped by status on retained records.
    4. Writes deterministic output sorted ascending by customer_id.

    Returns:
        Dict containing the status aggregation summary:
        {status: {'count': int, 'total_amount': Decimal}}
    """
    # Key: (customer_id, raw_timestamp) -> Value: row dictionary
    # Keeping the last occurrence handles deduplication of identical timestamps.
    deduped_records: Dict[Tuple[str, str], Dict[str, str]] = {}
    fieldnames: List[str] = []

    with open(input_filepath, mode="r", encoding="utf-8-sig", newline="") as infile:
        reader = csv.DictReader(infile)
        fieldnames = reader.fieldnames or []
        for row in reader:
            customer_id = row["customer_id"].strip()
            timestamp = row["timestamp"].strip()
            deduped_records[(customer_id, timestamp)] = row

    # Group valid unique records by customer_id
    customer_events = defaultdict(list)
    for (_, _), row in deduped_records.items():
        customer_id = row["customer_id"].strip()
        parsed_dt = datetime.strptime(row["timestamp"].strip(), timestamp_format)
        customer_events[customer_id].append((parsed_dt, row))

    # Pick the chronologically latest event per customer
    latest_per_customer: List[Dict[str, str]] = []
    for customer_id, events in customer_events.items():
        # Sort by parsed datetime ascending; the last item is the latest event
        events.sort(key=lambda x: x[0])
        latest_per_customer.append(events[-1][1])

    # Sort final retained records deterministically by customer_id
    latest_per_customer.sort(key=lambda x: x["customer_id"].strip())

    # Calculate count and total amount grouped by status
    status_summary: Dict[str, Dict[str, Any]] = defaultdict(
        lambda: {"count": 0, "total_amount": Decimal("0.00")}
    )

    for record in latest_per_customer:
        status = record.get("status", "UNKNOWN").strip()
        amount_str = record.get("amount", "0").strip()
        amount = Decimal(amount_str) if amount_str else Decimal("0.00")

        status_summary[status]["count"] += 1
        status_summary[status]["total_amount"] += amount

    # Write output to CSV
    with open(output_filepath, mode="w", encoding="utf-8", newline="") as outfile:
        writer = csv.DictWriter(outfile, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(latest_per_customer)

    return dict(status_summary)
