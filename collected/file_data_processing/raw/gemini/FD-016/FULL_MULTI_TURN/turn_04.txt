import pandas as pd


def aggregate_customer_transactions(
    customers_df: pd.DataFrame, transactions_df: pd.DataFrame
) -> pd.DataFrame:
    """Aggregates transactions by customer ID deterministically.

    - Excludes invalid/missing transaction IDs (None/NaN) - Excludes
    transactions referring to unknown customer IDs - Includes all master
    customers, defaulting zero-transaction customers to count=0 and amount=0.0
    - Sorts output strictly by customer_id ascending
    """
    # 1. Filter out null/invalid transaction IDs
    valid_txns = transactions_df.dropna(subset=["transaction_id"]).copy()

    # 2. Filter out transactions referencing unknown customer IDs (not in master)
    known_customer_ids = set(customers_df["customer_id"].dropna().unique())
    valid_txns = valid_txns[valid_txns["customer_id"].isin(known_customer_ids)]

    # 3. Aggregate valid transactions by customer_id
    aggregated_txns = (
        valid_txns.groupby("customer_id", as_index=False)
        .agg(
            total_spend=("amount", "sum"),
            transaction_count=("transaction_id", "count"),
        )
    )

    # 4. Deterministic Left Join starting from customer master
    result_df = pd.merge(
        customers_df, aggregated_txns, on="customer_id", how="left"
    )

    # 5. Deterministically assign 0 / 0.0 to zero-transaction customers
    result_df["total_spend"] = result_df["total_spend"].fillna(0.0)
    result_df["transaction_count"] = (
        result_df["transaction_count"].fillna(0).astype(int)
    )

    # 6. Sort by customer_id ascending and reset index for deterministic layout
    result_df = result_df.sort_values(
        by="customer_id", ascending=True
    ).reset_index(drop=True)

    return result_df


# Example Verification
if __name__ == "__main__":
    customers = pd.DataFrame(
        {
            "customer_id": [101, 102, 103, 104],
            "customer_name": ["Alice", "Bob", "Charlie", "Diana"],
        }
    )

    # Edge cases present:
    # - Customer 104 has 0 transactions
    # - Customer 999 is an unknown customer ID in transactions
    # - Transaction with None is an unknown/missing transaction ID
    transactions = pd.DataFrame(
        {
            "transaction_id": ["TXN001", "TXN002", "TXN003", None, "TXN005"],
            "customer_id": [101, 101, 999, 102, 103],
            "amount": [150.00, 50.00, 500.00, 200.00, 75.50],
        }
    )

    report = aggregate_customer_transactions(customers, transactions)
    print(report)
