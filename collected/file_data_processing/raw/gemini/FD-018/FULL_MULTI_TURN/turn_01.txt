import re
from typing import List, Dict, Any, Tuple

def normalize_text(text: str) -> str:
    """Normalizes string by lowering case, stripping whitespace, and removing special characters."""
    if not text:
        return ""
    text = text.lower()
    text = re.sub(r'[^a-z0-9\s]', '', text)
    return ' '.join(text.split())

def detect_duplicates(records: List[Dict[str, Any]]) -> List[List[Any]]:
    """
    Groups record IDs that share exact normalized title, description, and date.
    Returns a deterministic list of duplicate ID groups sorted by the earliest ID in each group.
    """
    grouped_records: Dict[Tuple[str, str, str], List[Any]] = {}

    for record in records:
        rec_id = record.get("id")
        norm_title = normalize_text(record.get("title", ""))
        norm_desc = normalize_text(record.get("description", ""))
        date = str(record.get("date", "")).strip()

        key = (norm_title, norm_desc, date)
        grouped_records.setdefault(key, []).append(rec_id)

    # Filter for groups with 2 or more records (duplicates)
    duplicate_groups = [
        sorted(group) for group in grouped_records.values() if len(group) > 1
    ]

    # Sort deterministically by the first ID of each duplicate group
    duplicate_groups.sort(key=lambda g: g[0])

    return duplicate_groups

# Example Usage
if __name__ == "__main__":
    sample_records = [
        {"id": 101, "title": " Acme Corp Report ", "description": "Q1 Financials...", "date": "2026-03-31"},
        {"id": 102, "title": "acme corp report", "description": "q1 financials", "date": "2026-03-31"},
        {"id": 103, "title": "Acme Corp Report!", "description": "Q1 Financials", "date": "2026-03-31"},
        {"id": 104, "title": "Acme Corp Report", "description": "Q2 Financials", "date": "2026-06-30"},
        {"id": 105, "title": "Unique Project", "description": "Details here", "date": "2026-01-15"}
    ]

    duplicates = detect_duplicates(sample_records)
    print("Duplicate ID Groups:", duplicates)
