Here is the complete **Expense Analyzer** Python script, refactored to include robust verification and dynamic sanitization for corrupted or malformed transaction records.

```python
import json
import os
import sys
import matplotlib.pyplot as plt

DATA_FILE = "expense_data.json"

# ==========================================
# 5. DATA PERSISTENCE & MALFORMED RECORD RECOVERY
# ==========================================
def validate_and_clean_record(record):
    """
    Validates individual transaction structure and fixes/discards malformed records.
    Returns sanitized record dict or None if unrecoverable.
    """
    if not isinstance(record, dict):
        return None
        
    desc = str(record.get("description", "Unspecified")).strip()
    category = str(record.get("category", "Uncategorized")).strip().title()
    
    # Attempt string-to-float conversion and enforce positive finite numbers
    try:
        raw_amount = record.get("amount")
        amount = float(raw_amount)
        if amount <= 0 or amount != amount:  # Check for <=0 and NaN
            return None
    except (ValueError, TypeError):
        return None

    return {
        "description": desc if desc else "Unspecified",
        "amount": round(amount, 2),
        "category": category if category else "Uncategorized"
    }

def load_data():
    """
    Loads JSON data with resilient error handling and cleans malformed transaction entries.
    """
    default_structure = {"users": {}, "expenses": {}, "budgets": {}}
    if not os.path.exists(DATA_FILE):
        return default_structure
        
    try:
        with open(DATA_FILE, "r") as f:
            data = json.load(f)
            
        if not isinstance(data, dict):
            return default_structure

        # Ensure top-level schema integrity
        for key in ["users", "expenses", "budgets"]:
            if key not in data or not isinstance(data[key], dict):
                data[key] = {}

        # Inspect and filter corrupted or malformed transaction records
        cleaned_expenses = {}
        malformed_count = 0
        
        for user, records in data["expenses"].items():
            if not isinstance(records, list):
                cleaned_expenses[user] = []
                continue
                
            valid_records = []
            for item in records:
                sanitized = validate_and_clean_record(item)
                if sanitized:
                    valid_records.append(sanitized)
                else:
                    malformed_count += 1
            cleaned_expenses[user] = valid_records
            
        data["expenses"] = cleaned_expenses

        if malformed_count > 0:
            print(f"\n[SYSTEM NOTICE] Cleaned up {malformed_count} malformed or corrupted transaction record(s) from persistent storage.")
            save_data(data)

        return data

    except (json.JSONDecodeError, PermissionError, IOError) as e:
        print(f"\n[WARNING] Could not parse persistent storage ({e}). Re-initializing storage.")
        return default_structure

def save_data(data):
    try:
        with open(DATA_FILE, "w") as f:
            json.dump(data, f, indent=4)
    except IOError as e:
        print(f"[ERROR] Failed to persist data: {e}")

# ==========================================
# 1. USER AUTHENTICATION
# ==========================================
def register(data):
    print("\n--- User Registration ---")
    username = input("Enter username: ").strip().lower()
    if not username:
        print("Username cannot be empty.")
        return None
    if username in data["users"]:
        print("Username already exists!")
        return None
    password = input("Enter password: ").strip()
    
    data["users"][username] = password
    data["expenses"][username] = []
    data["budgets"][username] = {}
    save_data(data)
    print("Registration successful! You can now log in.")
    return username

def login(data):
    print("\n--- User Login ---")
    username = input("Enter username: ").strip().lower()
    password = input("Enter password: ").strip()
    
    if data["users"].get(username) == password:
        print(f"\nWelcome back, {username}!")
        return username
    else:
        print("Invalid username or password.")
        return None

# ==========================================
# 2. EXPENSE RECORDING (QUICK ADD WITH INPUT SANITIZATION)
# ==========================================
def add_expense(data, username):
    print("\n--- Quick Add Expense ---")
    desc = input("Description (e.g., Lunch, Bus fare): ").strip()
    try:
        raw_amt = input("Amount: ").strip()
        amount = float(raw_amt)
        if amount <= 0:
            print("Invalid input: Amount must be greater than zero.")
            return
    except ValueError:
        print("Invalid input: Amount must be a valid numeric value.")
        return
    
    category = input("Category (e.g., Food, Transport, Rent, Entertainment): ").strip().title()
    
    raw_record = {
        "description": desc,
        "amount": amount,
        "category": category
    }
    
    sanitized_record = validate_and_clean_record(raw_record)
    if not sanitized_record:
        print("Failed to record expense due to invalid or malformed data format.")
        return

    if username not in data["expenses"]:
        data["expenses"][username] = []

    data["expenses"][username].append(sanitized_record)
    save_data(data)
    print(f"Added expense: '{sanitized_record['description']}' - ${sanitized_record['amount']:.2f} under [{sanitized_record['category']}]")
    
    check_budget_status(data, username, sanitized_record["category"])

# ==========================================
# 3. BUDGETING & LIMIT TRACKING
# ==========================================
def set_budget(data, username):
    print("\n--- Set Category Budget ---")
    category = input("Enter category name to set budget for: ").strip().title()
    if not category:
        print("Category name cannot be empty.")
        return
    try:
        limit = float(input(f"Enter monthly budget limit for {category}: "))
        if limit <= 0:
            print("Limit must be greater than zero.")
            return
    except ValueError:
        print("Invalid limit entered.")
        return

    if username not in data["budgets"]:
        data["budgets"][username] = {}

    data["budgets"][username][category] = round(limit, 2)
    save_data(data)
    print(f"Budget of ${limit:.2f} set for category '{category}'.")

def check_budget_status(data, username, category):
    user_budgets = data["budgets"].get(username, {})
    if category in user_budgets:
        limit = user_budgets[category]
        total_spent = sum(e["amount"] for e in data["expenses"].get(username, []) if e["category"] == category)
        
        if total_spent > limit:
            print(f"\n⚠️  [ALERT] You have EXCEEDED your budget for '{category}'!")
            print(f"    Spent: ${total_spent:.2f} | Limit: ${limit:.2f} | Over by: ${total_spent - limit:.2f}")
        elif total_spent >= limit * 0.85:
            print(f"\n⚠️  [WARNING] You have used {(total_spent / limit) * 100:.1f}% of your budget for '{category}'.")
            print(f"    Spent: ${total_spent:.2f} / ${limit:.2f}")

# ==========================================
# 4. DASHBOARD & VISUALIZATIONS
# ==========================================
def show_dashboard(data, username):
    user_expenses = data["expenses"].get(username, [])
    user_budgets = data["budgets"].get(username, {})
    
    print("\n==========================================")
    print(f"          EXPENSE DASHBOARD ({username.upper()})")
    print("==========================================")
    
    total_spent = sum(e["amount"] for e in user_expenses)
    print(f"Total Expenses Recorded: ${total_spent:.2f}\n")
    
    if not user_expenses:
        print("No expenses recorded yet.")
        return

    cat_totals = {}
    for e in user_expenses:
        cat = e["category"]
        cat_totals[cat] = cat_totals.get(cat, 0.0) + e["amount"]

    print(f"{'Category':<18} | {'Spent':<10} | {'Budget':<10} | {'Status'}")
    print("-" * 55)
    for cat, spent in cat_totals.items():
        budget_str = f"${user_budgets[cat]:.2f}" if cat in user_budgets else "N/A"
        if cat in user_budgets:
            status = "OVER BUDGET!" if spent > user_budgets[cat] else "OK"
        else:
            status = "-"
        print(f"{cat:<18} | ${spent:<9.2f} | {budget_str:<10} | {status}")
    print("-" * 55)

    plot_choice = input("\nWould you like to generate a graphical visual report? (y/n): ").strip().lower()
    if plot_choice == 'y':
        generate_charts(cat_totals)

def generate_charts(cat_totals):
    categories = list(cat_totals.keys())
    amounts = list(cat_totals.values())

    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

    ax1.pie(amounts, labels=categories, autopct='%1.1f%%', startangle=140)
    ax1.set_title("Expense Breakdown by Category")

    ax2.bar(categories, amounts, color='skyblue')
    ax2.set_xlabel("Categories")
    ax2.set_ylabel("Amount Spent ($)")
    ax2.set_title("Category Spending Comparison")
    plt.xticks(rotation=45)

    plt.tight_layout()
    plt.show()

# ==========================================
# MAIN APPLICATION LOOP
# ==========================================
def main():
    data = load_data()
    current_user = None

    while True:
        if not current_user:
            print("\n=== EXPENSE ANALYZER ===")
            print("1. Login")
            print("2. Register")
            print("3. Exit")
            choice = input("Select an option (1-3): ").strip()

            if choice == "1":
                current_user = login(data)
            elif choice == "2":
                current_user = register(data)
            elif choice == "3":
                print("Goodbye!")
                sys.exit()
            else:
                print("Invalid selection.")
        else:
            print(f"\n=== MAIN MENU (User: {current_user}) ===")
            print("1. Quick Add Expense")
            print("2. Set Category Budget")
            print("3. View Dashboard & Visualizations")
            print("4. Logout")
            choice = input("Select an option (1-4): ").strip()

            if choice == "1":
                add_expense(data, current_user)
            elif choice == "2":
                set_budget(data, current_user)
            elif choice == "3":
                show_dashboard(data, current_user)
            elif choice == "4":
                print(f"Logged out from {current_user}.")
                current_user = None
            else:
                print("Invalid selection.")

if __name__ == "__main__":
    main()
```
