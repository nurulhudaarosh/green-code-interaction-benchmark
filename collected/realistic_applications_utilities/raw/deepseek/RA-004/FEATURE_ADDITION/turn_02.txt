# ---------- Extension handling ----------
def _normalize_for_ext(name: str) -> str:
    try:
        return unicodedata.normalize("NFKC", name).casefold()
    except (TypeError, ValueError):
        return name.casefold() if isinstance(name, str) else ""


def get_extension(filename: str) -> str:
    if not isinstance(filename, str):
        return ""
    name = filename.strip()
    if not name:
        return ""
    if "/" in name or "\\" in name or os.sep in name:
        name = os.path.basename(name.replace("\\", "/"))
        if not name:
            return ""
    if set(name) == {"."}:
        return ""
    if name.startswith(".") and name.count(".") == 1:
        return ""
    if "." not in name:
        return ""
    if name.endswith("."):
        return ""
    folded = _normalize_for_ext(name)
    for ext in MULTIPART_EXTENSIONS:
        if folded.endswith(ext):
            return ext
    _, _, tail = folded.rpartition(".")
    if not tail or set(tail) == {"."}:
        return ""
    return "." + tail


def get_category(filename_or_ext: str) -> str:
    if not isinstance(filename_or_ext, str):
        return "Others"
    s = filename_or_ext.strip()
    if not s:
        return "Others"
    if s.startswith(".") and "/" not in s and "\\" not in s and s.count(".") >= 1:
        ext = _normalize_for_ext(s)
    else:
        ext = get_extension(s)
    if not ext:
        return "Others"
    return _EXT_TO_CATEGORY.get(ext, "Others")


# ---------- Size helpers ----------
def _coerce_int(value):
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return value
    if isinstance(value, float):
        if math.isnan(value) or math.isinf(value):
            return None
        if value.is_integer():
            return int(value)
    return None


def get_size_bucket(size_bytes, buckets=None) -> str:
    n = _coerce_int(size_bytes)
    if n is None or n < 0:
        return "Unknown"
    buckets = buckets or DEFAULT_SIZE_BUCKETS
    for name, lo, hi in buckets:
        if n >= lo and (hi is None or n < hi):
            return name
    return "Unknown"


def get_size_bucket_range(size_bytes, buckets=None):
    n = _coerce_int(size_bytes)
    if n is None or n < 0:
        return ("Unknown", None, None)
    buckets = buckets or DEFAULT_SIZE_BUCKETS
    for name, lo, hi in buckets:
        if n >= lo and (hi is None or n < hi):
            return (name, lo, hi)
    return ("Unknown", None, None)


def _parse_one_size(token: str) -> int:
    s = token.strip()
    if not s:
        raise ValueError("empty size token")
    unit = s[-1].upper()
    mult = {"K": KIB, "M": MIB, "G": GIB, "T": TIB}.get(unit)
    if mult is not None:
        num = s[:-1]
    else:
        mult = 1
        num = s
    if not num:
        raise ValueError(f"bad size token: {token!r}")
    try:
        val = int(float(num) * mult)
    except (ValueError, TypeError):
        raise ValueError(f"bad size token: {token!r}")
    if val < 0:
        raise ValueError(f"negative size: {token!r}")
    return val


def parse_size(token):
    if token is None:
        return None
    if not isinstance(token, str) or not token.strip():
        return None
    return _parse_one_size(token)


def parse_size_thresholds(spec):
    if spec is None:
        return None
    parts = [p.strip() for p in spec.split(",") if p.strip()]
    if not parts:
        return None

    values = [_parse_one_size(p) for p in parts]

    if values[0] <= 0:
        raise ValueError("first threshold must be > 0")
    for a, b in zip(values, values[1:]):
        if b <= a:
            raise ValueError("thresholds must be strictly increasing")

    default_names = [b[0] for b in DEFAULT_SIZE_BUCKETS]
    if len(values) + 1 == len(default_names):
        names = default_names
    else:
        names = [f"Bucket{i}" for i in range(len(values) + 1)]

    buckets = []
    prev = 0
    for name, hi in zip(names, values):
        buckets.append((name, prev, hi))
        prev = hi
    buckets.append((names[-1], prev, None))
    return buckets


def decorate_bucket_name(name, lo, hi, detail, order) -> str:
    if detail == "bucket":
        return name
    if detail == "prefixed":
        return f"{order}-{name}"
    if detail == "human":
        if lo is None or hi is None:
            span = "open"
        else:
            span = f"{human_size(lo)}-{human_size(hi)}"
        return f"{name} ({span})"
    if detail == "raw":
        if lo is None or hi is None:
            span = "open"
        elif hi is None:
            span = f"{lo}-inf"
        else:
            span = f"{lo}-{hi - 1}"
        return f"{name} ({span})"
    return name


def _bucket_order_map(buckets) -> dict:
    return {name: i + 1 for i, (name, _, _) in enumerate(buckets)}


# ---------- Timestamp handling ----------
def parse_timestamp(value):
    if value is None:
        return None
    if isinstance(value, datetime):
        return _clamp_year(value)
    if not isinstance(value, str):
        return None
    s = value.strip()
    if not s:
        return None
    if any(ord(ch) < 0x20 for ch in s):
        return None
    if s.count("Z") + s.count("z") > 1:
        return None
    if s.count("+") > 1:
        return None

    candidate = s.replace("Z", "+00:00").replace("z", "+00:00")
    dt = None
    try:
        dt = datetime.fromisoformat(candidate)
    except (ValueError, TypeError):
        dt = None

    if dt is None:
        for fmt in _ALT_DATE_FORMATS:
            try:
                dt = datetime.strptime(s, fmt)
                break
            except (ValueError, TypeError):
                continue

    if dt is None:
        return None
    return _clamp_year(dt)


def _clamp_year(dt):
    if _MIN_YEAR <= dt.year <= _MAX_YEAR:
        return dt
    return None


def is_valid_manifest_entry(entry) -> bool:
    if isinstance(entry, str):
        return bool(entry.strip())
    if isinstance(entry, dict):
        dest = entry.get("dest")
        return isinstance(dest, str) and bool(dest.strip())
    return False


# ---------- File utilities ----------
def unique_destination(dest_path: Path) -> Path:
    if not dest_path.exists():
        return dest_path
    stem = dest_path.stem
    suffix = dest_path.suffix
    parent = dest_path.parent
    counter = 1
    while True:
        candidate = parent / f"{stem} ({counter}){suffix}"
        if not candidate.exists():
            return candidate
        counter += 1


def human_size(n) -> str:
    if n is None:
        return "?"
    if isinstance(n, bool):
        return "?"
    if isinstance(n, float):
        if math.isnan(n) or math.isinf(n):
            return "?"
        if n.is_integer():
            n = int(n)
    if not isinstance(n, int):
        return "?"
    if n < 0:
        return "?"
    units = ["B", "KiB", "MiB", "GiB", "TiB", "PiB", "EiB"]
    size = float(n)
    for u in units:
        if size < 1024 or u == units[-1]:
            if u == "B":
                return f"{int(size)} {u}"
            return f"{size:.1f} {u}"
        size /= 1024


def _safe_stat_size(path: Path) -> int:
    try:
        return path.stat().st_size
    except OSError:
        return 0


def _quarantine_bad_entries(folder: Path, bad: dict) -> None:
    if not bad:
        return
    try:
        path = folder / MANIFEST_BAD_NAME
        existing = {}
        if path.exists():
            try:
                with open(path, "r", encoding="utf-8") as f:
                    existing = json.load(f)
                if not isinstance(existing, dict):
                    existing = {}
            except (json.JSONDecodeError, OSError):
                existing = {}
        existing.update(bad)
        with open(path, "w", encoding="utf-8") as f:
            json.dump(existing, f, indent=2)
        log(f"Quarantined {len(bad)} malformed entry(ies) -> {MANIFEST_BAD_NAME}")
    except OSError as e:
        log(f"Warning: could not write quarantine file: {e}")


def load_manifest(folder: Path) -> dict:
    manifest_path = folder / MANIFEST_NAME
    if not manifest_path.exists():
        return {}

    try:
        with open(manifest_path, "r", encoding="utf-8") as f:
            raw = json.load(f)
    except (json.JSONDecodeError, OSError) as e:
        log(f"Warning: could not read manifest ({e}). Starting fresh.")
        return {}

    if isinstance(raw, dict) and "entries" in raw and isinstance(raw["entries"], dict):
        entries = raw["entries"]
    elif isinstance(raw, dict):
        entries = raw
    else:
        log("Warning: manifest has unexpected structure. Ignoring.")
        return {}

    cleaned = {}
    bad = {}
    for original, entry in entries.items():
        if not isinstance(original, str) or not original.strip():
            bad[repr(original)] = entry
            continue
        if not is_valid_manifest_entry(entry):
            bad[original] = entry
            continue

        if isinstance(entry, str):
            cleaned[original] = {"dest": entry, "timestamp": None, "size": None}
        else:
            ts_raw = entry.get("timestamp")
            ts = parse_timestamp(ts_raw)
            if ts_raw is not None and ts is None:
                log(f"Warning: malformed timestamp for {original!r}: {ts_raw!r}")
                bad.setdefault(original, {
                    "dest": entry["dest"],
                    "timestamp": ts_raw,
                    "reason": "malformed_timestamp",
                })
            size = entry.get("size")
            if _coerce_int(size) is None or size < 0:
                size = None
            cleaned[original] = {
                "dest": entry["dest"],
                "timestamp": ts.isoformat() if ts else None,
                "size": size,
                "bucket": entry.get("bucket"),
                "bucket_detail": entry.get("bucket_detail"),
                "size_range": entry.get("size_range"),
            }

    _quarantine_bad_entries(folder, bad)
    return cleaned


def save_manifest(folder: Path, manifest: dict) -> None:
    manifest_path = folder / MANIFEST_NAME
    payload = {
        "version": MANIFEST_VERSION,
        "saved_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "entries": manifest,
    }
    with open(manifest_path, "w", encoding="utf-8") as f:
        json.dump(payload, f, indent=2)


def log(message: str) -> None:
    timestamp = datetime.now().strftime("%H:%M:%S")
    print(f"[{timestamp}] {message}")


# ---------- Layout planning ----------
def plan_relative_dir(file_path: Path, layout: str, size_buckets,
                      size_detail: str = "bucket") -> Path:
    if layout not in LAYOUTS:
        raise ValueError(f"Unknown layout: {layout}")

    buckets = size_buckets or DEFAULT_SIZE_BUCKETS
    size = _safe_stat_size(file_path)
    name, lo, hi = get_size_bucket_range(size, buckets)
    orders = _bucket_order_map(buckets)
    order = orders.get(name, 0)
    size_dir = decorate_bucket_name(name, lo, hi, size_detail, order)
    type_dir = get_category(file_path.name)

    if layout == "type":
        return Path(type_dir)
    if layout == "size":
        return Path(size_dir)
    if layout == "type-size":
        return Path(type_dir) / size_dir
    if layout == "size-type":
        return Path(size_dir) / type_dir
    return Path("Others")


# ---------- Core operations ----------
def scan_files(source: Path):
    for entry in source.iterdir():
        if not entry.is_file():
            continue
        if entry.name in (MANIFEST_NAME, MANIFEST_BAD_NAME):
            continue
        yield entry


def organize(source: Path, copy: bool = False, dry_run: bool = False,
             layout: str = "type", size_buckets=None,
             size_detail: str = "bucket",
             min_size=None, max_size=None) -> dict:
    if not source.exists() or not source.is_dir():
        raise NotADirectoryError(f"Not a valid directory: {source}")
    if layout not in LAYOUTS:
        raise ValueError(f"Unknown layout: {layout}")
    if size_detail not in SIZE_DETAILS:
        raise ValueError(f"Unknown size detail: {size_detail}")
    if min_size is not None and min_size < 0:
        raise ValueError("--min-size must be >= 0")
    if max_size is not None and max_size < 0:
        raise ValueError("--max-size must be >= 0")
    if (min_size is not None and max_size is not None
            and max_size < min_size):
        raise ValueError("--max-size must be >= --min-size")

    moved = {}
    stats = defaultdict(int)
    skipped_by_size = 0

    files = list(scan_files(source))
    if not files:
        log("No files found to organize.")
        return moved

    log(f"Found {len(files)} file(s) in '{source}'")
    log(f"Mode: {'COPY' if copy else 'MOVE'}{' (DRY RUN)' if dry_run else ''}")
    log(f"Layout: {layout}  (size-detail: {size_detail})")
    if min_size is not None or max_size is not None:
        lo_s = human_size(min_size) if min_size is not None else "0"
        hi_s = human_size(max_size) if max_size is not None else "inf"
        log(f"Size filter: [{lo_s}, {hi_s})")
    log("-" * 50)

    buckets = size_buckets or DEFAULT_SIZE_BUCKETS

    for file_path in files:
        size = _safe_stat_size(file_path)

        if min_size is not None and size < min_size:
            log(f"SKIP (below min size): {file_path.name}  [{human_size(size)}]")
            skipped_by_size += 1
            continue
        if max_size is not None and size >= max_size:
            log(f"SKIP (above max size): {file_path.name}  [{human_size(size)}]")
            skipped_by_size += 1
            continue

        name, lo, hi = get_size_bucket_range(size, buckets)

        rel_dir = plan_relative_dir(file_path, layout, buckets, size_detail)
        target_dir = source / rel_dir
        dest_path = unique_destination(target_dir / file_path.name)

        if lo is None:
            range_str = None
        elif hi is None:
            range_str = f"[{lo}, inf)"
        else:
            range_str = f"[{lo}, {hi})"

        try:
            if dry_run:
                log(f"[DRY] {file_path.name}  ->  {rel_dir}/  "
                    f"[{name}, {human_size(size)}]")
            else:
                target_dir.mkdir(parents=True, exist_ok=True)
                if copy:
                    shutil.copy2(file_path, dest_path)
                else:
                    shutil.move(str(file_path), str(dest_path))
                log(f"{file_path.name}  ->  {rel_dir}/{dest_path.name}  "
                    f"[{name}, {human_size(size)}]")
                moved[str(file_path)] = {
                    "dest": str(dest_path),
                    "timestamp": datetime.now(timezone.utc).isoformat(timespec="seconds"),
                    "size": size,
                    "bucket": name,
                    "bucket_detail": size_detail,
                    "size_range": range_str,
                    "layout": layout,
                }
            stats[f"{rel_dir}"] += 1
        except (OSError, shutil.Error) as e:
            log(f"ERROR moving '{file_path.name}': {e}")

    log("-" * 50)
    log("Summary:")
    for key, count in sorted(stats.items()):
        log(f"  {key:<40} {count} file(s)")
    if skipped_by_size:
        log(f"  (skipped by size filter: {skipped_by_size})")

    return moved


def undo(source: Path) -> None:
    manifest = load_manifest(source)
    if not manifest:
        log("No valid manifest entries found. Nothing to undo.")
        return

    log(f"Undoing {len(manifest)} operation(s)...")
    errors = 0
    skipped = 0

    for original, entry in manifest.items():
        current = entry.get("dest")
        ts = parse_timestamp(entry.get("timestamp"))

        if not isinstance(current, str) or not current.strip():
            log(f"SKIP (malformed dest): {original!r}")
            skipped += 1
            continue

        original_path = Path(original)
        current_path = Path(current)

        if entry.get("timestamp") and ts is None:
            log(f"Note: malformed timestamp for '{current_path.name}' "
                f"(proceeding anyway)")
        elif ts is not None:
            log(f"  (from {ts.strftime('%Y-%m-%d %H:%M:%S')})")

        if not current_path.exists():
            log(f"SKIP (missing): {current_path}")
            skipped += 1
            continue

        if original_path.exists():
            log(f"SKIP (target exists): {original_path}")
            errors += 1
            continue

        try:
            original_path.parent.mkdir(parents=True, exist_ok=True)
            shutil.move(str(current_path), str(original_path))
            log(f"Restored: {current_path.name}  ->  {original_path.parent.name}/")
        except (OSError, shutil.Error) as e:
            log(f"ERROR restoring '{current_path.name}': {e}")
            errors += 1

    _prune_empty_dirs(source)

    manifest_path = source / MANIFEST_NAME
    if manifest_path.exists():
        manifest_path.unlink()

    if errors == 0 and skipped == 0:
        log("Undo complete.")
    else:
        log(f"Undo complete with {errors} error(s) and {skipped} skipped.")


def _prune_empty_dirs(root: Path) -> None:
    for dirpath, dirnames, filenames in os.walk(root, topdown=False):
        p = Path(dirpath)
        if p == root:
            continue
        try:
            if p.is_dir() and not any(p.iterdir()):
                p.rmdir()
                log(f"Removed empty folder: {p.relative_to(root)}/")
        except OSError:
            pass


# ---------- Self-test ----------
def _self_test() -> None:
    ext_cases = [
        ("report.pdf", ".pdf", "Documents"),
        ("REPORT.PDF", ".pdf", "Documents"),
