#!/usr/bin/env python3
"""
Configuration Validation System — single-block version.

Includes:
  * Path resolution fixes (flat dotted keys, nesting, list indices, MISSING sentinel)
  * Boolean-vs-int type handling fixes
  * Cross-field dependency rules:
      DependsOn, RequiredIf, ForbiddenIf, MutuallyExclusive,
      RequiresAll, RequiresAny, Equals, GreaterThan
"""

from __future__ import annotations

import ipaddress
import json
import os
import re
from dataclasses import dataclass, field
from enum import Enum
from pathlib import Path
from typing import Any, Callable, List, Optional, Union
from urllib.parse import urlparse


# ============================================================
# Path resolution (FIX 1)
# ============================================================
MISSING = object()


def resolve_path(config: Any, dotted: str) -> Any:
    if not dotted:
        return MISSING
    if isinstance(config, dict) and dotted in config:
        return config[dotted]
    cur: Any = config
    parts = dotted.split(".")
    i = 0
    while i < len(parts):
        if isinstance(cur, dict):
            matched = False
            for j in range(len(parts), i, -1):
                candidate = ".".join(parts[i:j])
                if candidate in cur:
                    cur = cur[candidate]
                    i = j
                    matched = True
                    break
            if matched:
                continue
            part = parts[i]
            if part in cur:
                cur = cur[part]
                i += 1
                continue
            return MISSING
        if isinstance(cur, (list, tuple)):
            part = parts[i]
            if not part.lstrip("-").isdigit():
                return MISSING
            idx = int(part)
            if -len(cur) <= idx < len(cur):
                cur = cur[idx]
                i += 1
                continue
            return MISSING
        return MISSING
    return cur


def has_path(config: Any, dotted: str) -> bool:
    return resolve_path(config, dotted) is not MISSING


def is_set(value: Any) -> bool:
    return value is not MISSING and value not in (None, "", [], {})


# ============================================================
# Result types
# ============================================================
class Severity(Enum):
    ERROR = "error"
    WARNING = "warning"
    INFO = "info"


@dataclass
class ValidationIssue:
    path: str
    message: str
    severity: Severity = Severity.ERROR
    rule: str = ""
    value: Any = None

    def __str__(self) -> str:
        shown = "<missing>" if self.value is MISSING else repr(self.value)
        return f"[{self.severity.value.upper()}] {self.path}: {self.message} (rule={self.rule}, value={shown})"


@dataclass
class ValidationResult:
    issues: List[ValidationIssue] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not any(i.severity == Severity.ERROR for i in self.issues)

    def add(self, issue: ValidationIssue) -> None:
        self.issues.append(issue)

    def errors(self) -> List[ValidationIssue]:
        return [i for i in self.issues if i.severity == Severity.ERROR]

    def warnings(self) -> List[ValidationIssue]:
        return [i for i in self.issues if i.severity == Severity.WARNING]

    def report(self) -> str:
        if not self.issues:
            return "Configuration is valid."
        head = (f"Validation finished: {len(self.errors())} error(s), "
                f"{len(self.warnings())} warning(s)")
        return "\n".join([head] + [str(i) for i in self.issues])


# ============================================================
# Rule engine — single-field rules
# ============================================================
class Rule:
    name = "rule"

    def check(self, path: str, value: Any, config: dict) -> Optional[ValidationIssue]:
        raise NotImplementedError


class Required(Rule):
    name = "required"

    def check(self, path, value, config):
        if value is MISSING or value is None or value == "":
            return ValidationIssue(path, "is required", rule=self.name,
                                   value=None if value is MISSING else value)
        return None


class TypeRule(Rule):
    """FIX 2: correct bool-vs-int semantics."""
    name = "type"

    def __init__(self, expected: Union[type, tuple], *,
                 allow_bool: bool = False, allow_int: bool = False):
        self.allow_bool = allow_bool
        self.allow_int = allow_int
        self.expected: tuple = (expected,) if isinstance(expected, type) else tuple(expected)

    def _ok(self, value: Any) -> bool:
        if isinstance(value, bool):
            if bool in self.expected:
                return True
            if self.allow_bool and (int in self.expected or float in self.expected):
                return True
            return False
        if isinstance(value, int) and not isinstance(value, bool):
            if int in self.expected or float in self.expected:
                return True
            if bool in self.expected and self.allow_int and value in (0, 1):
                return True
            return False
        return isinstance(value, self.expected)

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        if self._ok(value):
            return None
        names = "/".join(t.__name__ for t in self.expected)
        return ValidationIssue(path, f"must be {names}, got {type(value).__name__}",
                               rule=self.name, value=value)


class Range(Rule):
    name = "range"

    def __init__(self, min_val=None, max_val=None, inclusive=True):
        self.min_val, self.max_val, self.inclusive = min_val, max_val, inclusive

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        if isinstance(value, bool) or not isinstance(value, (int, float)):
            return None
        if self.min_val is not None:
            bad = value < self.min_val if self.inclusive else value <= self.min_val
            if bad:
                op = ">=" if self.inclusive else ">"
                return ValidationIssue(path, f"must be {op} {self.min_val}, got {value}",
                                       rule=self.name, value=value)
        if self.max_val is not None:
            bad = value > self.max_val if self.inclusive else value >= self.max_val
            if bad:
                op = "<=" if self.inclusive else "<"
                return ValidationIssue(path, f"must be {op} {self.max_val}, got {value}",
                                       rule=self.name, value=value)
        return None


class Length(Rule):
    name = "length"

    def __init__(self, min_len=None, max_len=None, exact=None):
        self.min_len, self.max_len, self.exact = min_len, max_len, exact

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        try:
            n = len(value)
        except TypeError:
            return None
        if self.exact is not None and n != self.exact:
            return ValidationIssue(path, f"length must be exactly {self.exact}, got {n}",
                                   rule=self.name, value=value)
        if self.min_len is not None and n < self.min_len:
            return ValidationIssue(path, f"length must be >= {self.min_len}, got {n}",
                                   rule=self.name, value=value)
        if self.max_len is not None and n > self.max_len:
            return ValidationIssue(path, f"length must be <= {self.max_len}, got {n}",
                                   rule=self.name, value=value)
        return None


class Pattern(Rule):
    name = "pattern"

    def __init__(self, regex: str, flags: int = 0):
        self.regex = re.compile(regex, flags)

    def check(self, path, value, config):
        if value is MISSING or value is None or not isinstance(value, str):
            return None
        if not self.regex.fullmatch(value):
            return ValidationIssue(path, f"must match pattern {self.regex.pattern!r}",
                                   rule=self.name, value=value)
        return None


class Choices(Rule):
    name = "choices"

    def __init__(self, options: List[Any], case_sensitive: bool = True):
        self.options = options
        self.case_sensitive = case_sensitive

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        opts, v = self.options, value
        if not self.case_sensitive and isinstance(value, str):
            opts = [o.lower() if isinstance(o, str) else o for o in self.options]
            v = value.lower()
        for o in opts:
            if isinstance(o, bool) or isinstance(v, bool):
                if o is v:
                    return None
            elif o == v:
                return None
        return ValidationIssue(path, f"must be one of {self.options}, got {value!r}",
                               rule=self.name, value=value)


class Custom(Rule):
    name = "custom"

    def __init__(self, func: Callable[[Any, dict], bool], message: str = "failed custom check"):
        self.func, self.message = func, message

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        try:
            if not self.func(value, config):
                return ValidationIssue(path, self.message, rule=self.name, value=value)
        except Exception as e:
            return ValidationIssue(path, f"custom check raised {e!r}",
                                   rule=self.name, value=value)
        return None


class URL(Rule):
    name = "url"

    def __init__(self, schemes: Optional[List[str]] = None, require_host: bool = True):
        self.schemes = schemes or ["http", "https"]
        self.require_host = require_host

    def check(self, path, value, config):
        if value is MISSING or value is None or not isinstance(value, str):
            return None
        try:
            parsed = urlparse(value)
        except Exception as e:
            return ValidationIssue(path, f"invalid URL: {e}", rule=self.name, value=value)
        if parsed.scheme not in self.schemes:
            return ValidationIssue(path, f"URL scheme must be one of {self.schemes}",
                                   rule=self.name, value=value)
        if self.require_host and not parsed.hostname:
            return ValidationIssue(path, "URL must have a host",
                                   rule=self.name, value=value)
        return None


class Email(Rule):
    name = "email"
    _re = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")

    def check(self, path, value, config):
        if value is MISSING or value is None or not isinstance(value, str):
            return None
        if not self._re.match(value):
            return ValidationIssue(path, "must be a valid email",
                                   rule=self.name, value=value)
        return None


class Hostname(Rule):
    name = "hostname"
    _re = re.compile(r"^(?=.{1,253}$)([a-zA-Z0-9](-?[a-zA-Z0-9])*)(\.[a-zA-Z0-9](-?[a-zA-Z0-9])*)*$")

    def check(self, path, value, config):
        if value is MISSING or value is None or not isinstance(value, str):
            return None
        if not self._re.match(value):
            return ValidationIssue(path, "must be a valid hostname",
                                   rule=self.name, value=value)
        return None


class IPAddress(Rule):
    name = "ip"

    def __init__(self, version: Optional[int] = None):
        self.version = version

    def check(self, path, value, config):
        if value is MISSING or value is None or not isinstance(value, str):
            return None
        try:
            ip = ipaddress.ip_address(value)
        except ValueError:
            return ValidationIssue(path, "must be a valid IP address",
                                   rule=self.name, value=value)
        if self.version and ip.version != self.version:
            return ValidationIssue(path, f"must be IPv{self.version}",
                                   rule=self.name, value=value)
        return None


class Port(Rule):
    name = "port"

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        if isinstance(value, bool) or not isinstance(value, int):
            return ValidationIssue(path, "port must be an integer",
                                   rule=self.name, value=value)
        if not (1 <= value <= 65535):
            return ValidationIssue(path, "port must be between 1 and 65535",
                                   rule=self.name, value=value)
        return None


class NoExtraKeys(Rule):
    name = "no_extra_keys"

    def __init__(self, allowed: List[str]):
        self.allowed = set(allowed)

    def check(self, path, value, config):
        if not isinstance(value, dict):
            return None
        extras = set(value) - self.allowed
        if extras:
            return ValidationIssue(path, f"unexpected keys: {sorted(extras)}",
                                   rule=self.name, value=sorted(extras))
        return None


# ============================================================
# Cross-field rules
# ============================================================
def _eval_predicate(pred, config: dict) -> bool:
    if pred is None:
        return False
    if callable(pred):
        try:
            return bool(pred(config))
        except Exception:
            return False
    if isinstance(pred, str):
        return is_set(resolve_path(config, pred))
    return bool(pred)


def _describe(pred) -> str:
    return f"'{pred}' is set" if isinstance(pred, str) else "predicate holds"


class DependsOn(Rule):
    name = "depends_on"

    def __init__(self, other_path: str):
        self.other_path = other_path

    def check(self, path, value, config):
        if not is_set(value):
            return None
        if not is_set(resolve_path(config, self.other_path)):
            return ValidationIssue(path, f"requires '{self.other_path}' to be set",
                                   rule=self.name, value=value)
        return None


class RequiredIf(Rule):
    name = "required_if"

    def __init__(self, predicate, message: Optional[str] = None):
        self.predicate = predicate
        self.message = message

    def check(self, path, value, config):
        if not _eval_predicate(self.predicate, config):
            return None
        if not is_set(value):
            msg = self.message or f"is required when {_describe(self.predicate)}"
            return ValidationIssue(path, msg, rule=self.name, value=value)
        return None


class ForbiddenIf(Rule):
    name = "forbidden_if"

    def __init__(self, predicate, message: Optional[str] = None):
        self.predicate = predicate
        self.message = message

    def check(self, path, value, config):
        if not is_set(value):
            return None
        if _eval_predicate(self.predicate, config):
            msg = self.message or f"must not be set when {_describe(self.predicate)}"
            return ValidationIssue(path, msg, rule=self.name, value=value)
        return None


class MutuallyExclusive(Rule):
    name = "mutually_exclusive"

    def __init__(self, others: List[str]):
        self.others = others

    def check(self, path, value, config):
        if not is_set(value):
            return None
        for other in self.others:
            if is_set(resolve_path(config, other)):
                return ValidationIssue(path, f"mutually exclusive with '{other}'",
                                       rule=self.name, value=value)
        return None


class RequiresAll(Rule):
    name = "requires_all"

    def __init__(self, others: List[str]):
        self.others = others

    def check(self, path, value, config):
        if not is_set(value):
            return None
        missing = [o for o in self.others if not is_set(resolve_path(config, o))]
        if missing:
            return ValidationIssue(path, f"requires all of {missing} to be set",
                                   rule=self.name, value=value)
        return None


class RequiresAny(Rule):
    name = "requires_any"

    def __init__(self, others: List[str]):
        self.others = others

    def check(self, path, value, config):
        if not is_set(value):
            return None
        if not any(is_set(resolve_path(config, o)) for o in self.others):
            return ValidationIssue(path, f"requires at least one of {self.others}",
                                   rule=self.name, value=value)
        return None


class Equals(Rule):
    name = "equals"

    def __init__(self, other_path: str):
        self.other_path = other_path

    def check(self, path, value, config):
        if not is_set(value):
            return None
        other = resolve_path(config, self.other_path)
        if not is_set(other):
            return ValidationIssue(path, f"equals check needs '{self.other_path}' to be set",
                                   rule=self.name, value=value)
        if isinstance(value, bool) or isinstance(other, bool):
            same = value is other
        else:
            same = value == other
        if not same:
            return ValidationIssue(path, f"must equal '{self.other_path}' ({other!r})",
                                   rule=self.name, value=value)
        return None


class GreaterThan(Rule):
    name = "greater_than"

    def __init__(self, other_path: str):
        self.other_path = other_path

    def check(self, path, value, config):
        if value is MISSING or value is None:
            return None
        if isinstance(value, bool) or not isinstance(value, (int, float)):
            return None
        other = resolve_path(config, self.other_path)
        if not is_set(other) or isinstance(other, bool) or not isinstance(other, (int, float)):
            return None
        if value <= other:
            return ValidationIssue(path, f"must be > '{self.other_path}' ({other})",
                                   rule=self.name, value=value)
        return None

@dataclass
class FieldSchema:
    path: str
    rules: List[Rule] = field(default_factory=list)
    optional: bool = True

    def validate(self, config: dict, result: ValidationResult) -> Any:
        value = resolve_path(config, self.path)
        present = value is not MISSING
        for rule in self.rules:
            if isinstance(rule, Required):
                if not present and self.optional:
                    continue
            elif isinstance(rule, (RequiredIf, ForbiddenIf)):
                pass
            elif not present:
                continue
            issue = rule.check(self.path, value, config)
            if issue:
                result.add(issue)
        return value


@dataclass
class ConfigSchema:
    fields: List[FieldSchema] = field(default_factory=list)

    def field(self, path: str, *rules: Rule, optional: bool = True) -> "ConfigSchema":
        self.fields.append(FieldSchema(path=path, rules=list(rules), optional=optional))
        return self

    def validate(self, config: dict) -> ValidationResult:
        result = ValidationResult()
        for f in self.fields:
            f.validate(config, result)
        return result

# Example schema
# ============================================================
def build_app_schema() -> ConfigSchema:
    s = ConfigSchema()

    s.field("app.name", Required(), TypeRule(str), Length(min_len=1, max_len=64),
            optional=False)
    s.field("app.env", Required(), Choices(["dev", "staging", "prod"]),
            optional=False)
    s.field("app.debug", TypeRule(bool))
    s.field("app.retries", TypeRule(int))
    s.field("app.version", Pattern(r"\d+\.\d+\.\d+"))

    s.field("server.host", TypeRule(str), Hostname())
    s.field("server.port", Required(), Port(), optional=False)
    s.field("server.workers", TypeRule(int), Range(min_val=1, max_val=128))
    s.field("server.timeout", TypeRule((int, float)), Range(min_val=0.1, max_val=300))
    s.field("server.protocol", Choices(["http", "https"], case_sensitive=False))
    s.field("server.keepalive", TypeRule(int, allow_bool=True), Range(min_val=0, max_val=1))

    s.field("server.tls_cert",
            RequiredIf(lambda c: c.get("server", {}).get("protocol", "").lower() == "https"),
            TypeRule(str), Length(min_len=1))
    s.field("server.tls_key",
            RequiredIf(lambda c: c.get("server", {}).get("protocol", "").lower() == "https"),
            TypeRule(str))
    s.field("server.tls_cert",
            ForbiddenIf(lambda c: c.get("server", {}).get("protocol", "").lower() != "https"))
    s.field("server.tls_key",
            ForbiddenIf(lambda c: c.get("server", {}).get("protocol", "").lower() != "https"))

    s.field("database.url", Required(), TypeRule(str),
            URL(schemes=["postgres", "mysql", "sqlite"]), optional=False)
    s.field("database.pool_size", TypeRule(int), Range(min_val=1, max_val=100))
    s.field("database.ssl", TypeRule(bool))
    s.field("database.ssl_ca", RequiredIf("database.ssl"), TypeRule(str))
    s.field("database.replica_url", URL(schemes=["postgres", "mysql"]))
    s.field("database.replica_url",
            DependsOn("database.url"),
            Custom(lambda v, c: v != resolve_path(c, "database.url"),
                   "replica URL must differ from primary URL"))

    s.field("cache.enabled", TypeRule(bool))
    s.field("cache.host", DependsOn("cache.enabled"), IPAddress(version=4))
    s.field("cache.port", DependsOn("cache.enabled"), Port())

    s.field("auth.mode", Choices(["none", "basic", "jwt", "oauth"]))
    s.field("auth.secret", DependsOn("auth.mode"), Length(min_len=16))
    s.field("auth.basic_user", MutuallyExclusive(["auth.jwt_token", "auth.oauth_client_id"]))
    s.field("auth.jwt_token", MutuallyExclusive(["auth.basic_user", "auth.oauth_client_id"]))
    s.field("auth.oauth_client_id", MutuallyExclusive(["auth.basic_user", "auth.jwt_token"]))
    s.field("auth.oauth_client_id", RequiresAll(["auth.oauth_client_secret"]))
    s.field("auth.oauth_client_secret", RequiresAll(["auth.oauth_client_id"]))
    s.field("auth.jwt_token", RequiresAny(["auth.jwt_secret", "auth.jwt_public_key"]))

    s.field("server.tls_min_version", Choices(["1.0", "1.1", "1.2", "1.3"]))
    s.field("server.tls_max_version", Choices(["1.0", "1.1", "1.2", "1.3"]))
    s.field("server.tls_max_version",
            Custom(lambda v, c: (
                (not is_set(v)) or (not is_set(resolve_path(c, "server.tls_min_version")))
                or float(v) >= float(resolve_path(c, "server.tls_min_version"))
            ), "tls_max_version must be >= tls_min_version"))

    s.field("observability.metrics_port", Port(), GreaterThan("server.port"))

    s.field("notifications.email", Email())
    s.field("notifications.webhook", URL(schemes=["https"]))
    s.field("notifications.slack_webhook",
            RequiredIf("notifications.webhook"), URL(schemes=["https"]))

    s.field("app", NoExtraKeys(["name", "env", "debug", "retries", "version",
                                "allow_privileged_ports"]))
    return s


# ============================================================
# Env loader
# ============================================================
def _coerce(v: str) -> Any:
    try:
        return json.loads(v)
    except Exception:
        return v


def set_nested(cfg: dict, dotted: str, value: Any) -> None:
    parts = dotted.split(".")
    cur = cfg
    for p in parts[:-1]:
        nxt = cur.get(p)
        if not isinstance(nxt, dict):
            nxt = {}
            cur[p] = nxt
        cur = nxt
    cur[parts[-1]] = value


def load_config(path: Optional[str] = None, env_prefix: str = "APP_") -> dict:
    cfg: dict = {}
    if path and Path(path).exists():
        with open(path) as f:
            cfg = json.load(f)
    for key, val in os.environ.items():
        if not key.startswith(env_prefix):
            continue
        dotted = key[len(env_prefix):].lower().replace("__", ".")
        set_nested(cfg, dotted, _coerce(val))
    return cfg

# Regression tests

def test_path_resolution():
    assert resolve_path({"server.port": 80}, "server.port") == 80
    assert resolve_path({"server": {"port": 80}}, "server.port") == 80
    assert resolve_path({"servers": [{"host": "a"}, {"host": "b"}]},
                        "servers.0.host") == "a"
    assert resolve_path({"servers": [{"host": "a"}, {"host": "b"}]},
                        "servers.-1.host") == "b"
    assert resolve_path({"a": {"b": {"c": 0}}}, "a.b.c") == 0
    assert resolve_path({"a": {"b": {"c": False}}}, "a.b.c") is False
    assert resolve_path({"server": {}}, "server.port") is MISSING
    assert resolve_path({"a.b": {"c.d": 1}}, "a.b.c.d") == 1


def test_bool_handling():
    assert TypeRule(int).check("x", True, {}) is not None
    assert TypeRule(int).check("x", 1, {}) is None
    assert TypeRule(float).check("x", False, {}) is not None
    assert TypeRule(bool).check("x", 1, {}) is not None
    assert TypeRule(bool, allow_int=True).check("x", 1, {}) is None
    assert TypeRule(bool, allow_int=True).check("x", 2, {}) is not None
    assert TypeRule(int, allow_bool=True).check("x", True, {}) is None
    assert Port().check("x", True, {}) is not None
    assert Range(min_val=2).check("x", True, {}) is None
    assert Choices([1]).check("x", True, {}) is not None
    assert Choices([True]).check("x", 1, {}) is not None


def test_cross_field():
    r = ConfigSchema().field("cache.host", DependsOn("cache.enabled"))
    assert r.validate({"cache": {"host": "10.0.0.1"}}).ok is False
    assert r.validate({"cache": {"host": "10.0.0.1", "enabled": True}}).ok is True

    r = ConfigSchema().field("database.ssl_ca", RequiredIf("database.ssl"))
    assert r.validate({"database": {"ssl": True}}).ok is False
    assert r.validate({"database": {"ssl": True, "ssl_ca": "ca.pem"}}).ok is True
    assert r.validate({"database": {"ssl": False}}).ok is True

    r = ConfigSchema().field("x", ForbiddenIf("y"))
    assert r.validate({"y": True, "x": 1}).ok is False
    assert r.validate({"y": False, "x": 1}).ok is True

    r = ConfigSchema().field("a", MutuallyExclusive(["b"]))
    assert r.validate({"a": 1, "b": 2}).ok is False
    assert r.validate({"a": 1}).ok is True

    r = ConfigSchema().field("a", RequiresAll(["b", "c"]))
    assert r.validate({"a": 1, "b": 2}).ok is False
    assert r.validate({"a": 1, "b": 2, "c": 3}).ok is True

    r = ConfigSchema().field("a", RequiresAny(["b", "c"]))
    assert r.validate({"a": 1}).ok is False
    assert r.validate({"a": 1, "b": 2}).ok is True

    r = ConfigSchema().field("a", Equals("b"))
    assert r.validate({"a": 1, "b": 2}).ok is False
    assert r.validate({"a": 1, "b": 1}).ok is True
    assert r.validate({"a": True, "b": 1}).ok is False

    r = ConfigSchema().field("a", GreaterThan("b"))
    assert r.validate({"a": 1, "b": 2}).ok is False
    assert r.validate({"a": 3, "b": 2}).ok is True
