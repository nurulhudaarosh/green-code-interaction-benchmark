def summarize_logs(lines):
    """
    Summarize logs and detect repeated errors.

    Valid format:
        LEVEL: message

    Valid levels:
        INFO, WARNING, ERROR
    """
    summary = {
        "info": 0,
        "warning": 0,
        "error": 0,
        "errors": [],
        "repeated_errors": []
    }

    if not isinstance(lines, list):
        return summary

    error_counts = {}
    error_messages = {}

    for line in lines:
        # Ignore non-string and empty lines
        if not isinstance(line, str) or not line.strip():
            continue

        # Split only at the first colon
        parts = line.split(":", 1)

        # Ignore malformed lines
        if len(parts) != 2:
            continue

        level = parts[0].strip().lower()
        message = parts[1].strip()

        # Ignore missing level or message
        if not level or not message:
            continue

        if level == "info":
            summary["info"] += 1

        elif level == "warning":
            summary["warning"] += 1

        elif level == "error":
            summary["error"] += 1
            summary["errors"].append(message)

            # Normalize whitespace and case for comparison
            normalized = " ".join(message.lower().split())

            error_counts[normalized] = error_counts.get(normalized, 0) + 1
            error_messages[normalized] = message

    # Detect errors occurring more than once
    for normalized, count in error_counts.items():
        if count > 1:
            summary["repeated_errors"].append(error_messages[normalized])

    return summary


if __name__ == "__main__":
    logs = [
        "INFO: Application started",
        "ERROR: Database connection failed",
        "WARNING: Low disk space",
        "ERROR: Database connection failed",
        "ERROR:   Database   connection   failed  ",
        "INFO: User logged in",
        "MALFORMED LINE",
        "",
        "ERROR:",
        ": Missing message"
    ]

    result = summarize_logs(logs)
    print(result)
