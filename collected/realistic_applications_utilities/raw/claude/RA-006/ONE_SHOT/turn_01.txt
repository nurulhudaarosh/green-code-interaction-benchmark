from __future__ import annotations
import re
from dataclasses import dataclass, field
from typing import Any, Callable

MISSING = object()


@dataclass
class ValidationResult:
    errors: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not self.errors

    def __str__(self) -> str:
        return "OK" if self.ok else "\n".join(f"- {e}" for e in self.errors)


def get_path(cfg: dict, path: str) -> Any:
    """Fetch nested value using dotted path, e.g. 'db.port'."""
    cur: Any = cfg
    for part in path.split("."):
        if not isinstance(cur, dict) or part not in cur:
            return MISSING
        cur = cur[part]
    return cur


# ---- Rule factories: each returns fn(cfg) -> list[str] of errors ----

def required(path: str):
    return lambda c: [f"'{path}' is required"] if get_path(c, path) is MISSING else []


def type_is(path: str, *types: type):
    def rule(c):
        v = get_path(c, path)
        if v is MISSING:
            return []
        # bool is a subclass of int; reject it unless bool explicitly allowed
        if isinstance(v, bool) and bool not in types:
            return [f"'{path}' must be {_names(types)}, got bool"]
        if not isinstance(v, types):
            return [f"'{path}' must be {_names(types)}, got {type(v).__name__}"]
        return []
    return rule


def in_range(path: str, lo=None, hi=None):
    def rule(c):
        v = get_path(c, path)
        if v is MISSING or not isinstance(v, (int, float)) or isinstance(v, bool):
            return []
        if lo is not None and v < lo:
            return [f"'{path}' must be >= {lo}, got {v}"]
        if hi is not None and v > hi:
            return [f"'{path}' must be <= {hi}, got {v}"]
        return []
    return rule


def length_between(path: str, min_len=0, max_len=None):
    def rule(c):
        v = get_path(c, path)
        if v is MISSING or not hasattr(v, "__len__"):
            return []
        if len(v) < min_len:
            return [f"'{path}' length must be >= {min_len}"]
        if max_len is not None and len(v) > max_len:
            return [f"'{path}' length must be <= {max_len}"]
        return []
    return rule


def one_of(path: str, allowed: set):
    def rule(c):
        v = get_path(c, path)
        if v is MISSING or v in allowed:
            return []
        return [f"'{path}' must be one of {sorted(map(str, allowed))}, got {v!r}"]
    return rule


def matches(path: str, pattern: str, desc: str = ""):
    rx = re.compile(pattern)
    def rule(c):
        v = get_path(c, path)
        if v is MISSING or (isinstance(v, str) and rx.fullmatch(v)):
            return []
        return [f"'{path}' must match {desc or pattern}, got {v!r}"]
    return rule


def requires(path: str, *others: str):
    """If `path` is set, all `others` must be set too."""
    def rule(c):
        if get_path(c, path) is MISSING:
            return []
        return [f"'{path}' requires '{o}'" for o in others if get_path(c, o) is MISSING]
    return rule


def mutually_exclusive(*paths: str):
    def rule(c):
        present = [p for p in paths if get_path(c, p) is not MISSING]
        return [f"Only one of {list(paths)} may be set, found {present}"] if len(present) > 1 else []
    return rule


def at_least_one_of(*paths: str):
    def rule(c):
        if any(get_path(c, p) is not MISSING for p in paths):
            return []
        return [f"At least one of {list(paths)} must be set"]
    return rule


def custom(message: str, predicate: Callable[[dict], bool]):
    def rule(c):
        try:
            return [] if predicate(c) else [message]
        except Exception as e:  # predicate hit missing/invalid data
            return [f"{message} (check failed: {e})"]
    return rule


def _names(types):
    return " or ".join(t.__name__ for t in types)


# ---- Validator ----

class ConfigValidator:
    def __init__(self, rules: list[Callable[[dict], list[str]]]):
        self.rules = rules

    def validate(self, cfg: dict) -> ValidationResult:
        result = ValidationResult()
        for rule in self.rules:
            result.errors.extend(rule(cfg))
        return result


# ---- Example rule set (replace with your actual rules) ----

RULES = [
    required("app.name"),
    type_is("app.name", str),
    length_between("app.name", 1, 64),
    required("app.env"),
    one_of("app.env", {"dev", "staging", "prod"}),

    required("server.host"),
    type_is("server.host", str),
    required("server.port"),
    type_is("server.port", int),
    in_range("server.port", 1, 65535),

    type_is("db.url", str),
    matches("db.url", r"(postgresql|mysql|sqlite)://.+", "a supported DB URL"),
    in_range("db.pool_size", 1, 100),

    requires("tls.enabled", "tls.cert_path", "tls.key_path"),
    mutually_exclusive("auth.api_key", "auth.oauth"),
    at_least_one_of("auth.api_key", "auth.oauth"),

    custom("prod must have tls.enabled = true",
           lambda c: get_path(c, "app.env") != "prod" or get_path(c, "tls.enabled") is True),
    custom("debug must be false in prod",
           lambda c: get_path(c, "app.env") != "prod" or get_path(c, "app.debug") in (False, MISSING)),
]


if __name__ == "__main__":
    good = {
        "app": {"name": "svc", "env": "prod", "debug": False},
        "server": {"host": "0.0.0.0", "port": 8080},
        "db": {"url": "postgresql://u:p@localhost/db", "pool_size": 10},

        "auth": {"api_key": "abc"},
    }
    bad = {
        "app": {"name": "", "env": "qa", "debug": True},
        "server": {"host": "localhost", "port": 70000},
        "db": {"url": "oracle://x", "pool_size": 0},
        "tls": {"enabled": True},
        "auth": {"api_key": "a", "oauth": {}},
    }

    v = ConfigValidator(RULES)
    print("good:", v.validate(good))
    print("bad:\n", v.validate(bad))
