def check_inventory_status(inventory: dict, sku: str, reorder_level: int = 10) -> str:
    """
    Determine stock status for a SKU, handling edge cases:
    - empty/missing inventory data
    - unknown SKU
    - zero stock
    - boundary reorder levels (stock exactly at reorder_level)
    """
    # 1. Empty or missing data
    if not inventory:
        return "ERROR: No inventory data available."

    # 2. Unknown SKU
    if sku not in inventory:
        return f"ERROR: SKU '{sku}' not found in inventory."

    stock = inventory[sku]

    # Defensive check for bad values (e.g., None, negative stock)
    if stock is None or not isinstance(stock, (int, float)):
        return f"ERROR: Invalid stock value for SKU '{sku}'."
    if stock < 0:
        return f"ERROR: Negative stock ({stock}) for SKU '{sku}' — data issue."

    # 3. Zero stock
    if stock == 0:
        return f"OUT_OF_STOCK: SKU '{sku}' has 0 units."

    # 4. Boundary reorder level (stock exactly equals reorder_level)
    if stock == reorder_level:
        return f"REORDER_NOW: SKU '{sku}' at boundary ({stock} == reorder level {reorder_level})."

    # Below reorder level
    if stock < reorder_level:
        return f"LOW_STOCK: SKU '{sku}' has {stock} units (below reorder level {reorder_level})."

    # Healthy stock
    return f"OK: SKU '{sku}' has {stock} units (above reorder level {reorder_level})."
