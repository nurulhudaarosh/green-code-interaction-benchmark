from pathlib import Path
from datetime import datetime
import shutil


class DocumentOrganizer:
    """
    Organizes documents by file extension and validates date metadata.

    Handles:
    - Missing or malformed file extensions
    - Upper/lower-case extensions
    - Files with no extension
    - Malformed dates
    - Missing dates
    - Duplicate filenames
    - Invalid source directories
    - Dry-run mode
    """

    CATEGORIES = {
        ".pdf": "PDF",
        ".doc": "Documents",
        ".docx": "Documents",
        ".txt": "Documents",
        ".rtf": "Documents",
        ".odt": "Documents",
        ".xls": "Spreadsheets",
        ".xlsx": "Spreadsheets",
        ".csv": "Spreadsheets",
        ".ods": "Spreadsheets",
        ".ppt": "Presentations",
        ".pptx": "Presentations",
        ".odp": "Presentations",
        ".jpg": "Images",
        ".jpeg": "Images",
        ".png": "Images",
        ".gif": "Images",
        ".bmp": "Images",
        ".svg": "Images",
        ".webp": "Images",
        ".zip": "Archives",
        ".rar": "Archives",
        ".7z": "Archives",
        ".tar": "Archives",
        ".gz": "Archives",
    }

    def __init__(self, source_directory, destination_directory=None):
        self.source_directory = Path(source_directory)
        self.destination_directory = (
            Path(destination_directory)
            if destination_directory
            else self.source_directory
        )

    def normalize_extension(self, file_path):
        """
        Safely extract and normalize a file extension.

        Examples:
            report.PDF -> .pdf
            report.docx -> .docx
            report -> ""
            report. -> ""
        """
        suffix = file_path.suffix.strip().lower()

        if not suffix or suffix == ".":
            return ""

        # Reject suspicious/malformed extensions.
        if not suffix.startswith("."):
            return ""

        if len(suffix) == 1:
            return ""

        if not suffix[1:].replace("_", "").replace("-", "").isalnum():
            return ""

        return suffix

    def get_category(self, file_path):
        """Return a category based on the normalized extension."""
        extension = self.normalize_extension(file_path)

        if not extension:
            return "Unknown"

        return self.CATEGORIES.get(extension, "Other")

    def parse_date(self, date_value):
        """
        Parse a date safely.

        Accepted formats:
            YYYY-MM-DD
            YYYY/MM/DD
            DD-MM-YYYY
            DD/MM/YYYY

        Malformed or missing dates return None.
        """
        if date_value is None:
            return None

        if not isinstance(date_value, str):
            return None

        date_value = date_value.strip()

        if not date_value:
            return None

        formats = (
            "%Y-%m-%d",
            "%Y/%m/%d",
            "%d-%m-%Y",
            "%d/%m/%Y",
        )

        for date_format in formats:
            try:
                return datetime.strptime(
                    date_value,
                    date_format
                ).date()
            except ValueError:
                continue

        return None

    def validate_date(self, date_value):
        """
        Return True only when the supplied date is valid.
        """
        return self.parse_date(date_value) is not None

    def get_unique_destination(self, destination):
        """Avoid overwriting existing files."""
        if not destination.exists():
            return destination

        counter = 1

        while True:
            candidate = (
                destination.parent
                / f"{destination.stem}_{counter}{destination.suffix}"
            )

            if not candidate.exists():
                return candidate

            counter += 1

    def organize(self, file_dates=None, dry_run=False):
        """
        Organize files.

        file_dates is an optional dictionary:
            {
                "report.pdf": "2026-09-01",
                "notes.docx": "01/09/2026"
            }

        Malformed dates are not used for organization.
        """
        summary = {
            "moved": 0,
            "skipped": 0,
            "errors": 0,
            "invalid_dates": 0,
            "by_category": {},
            "details": [],
        }

        if not self.source_directory.exists():
            summary["errors"] += 1
            summary["details"].append(
                f"Source directory does not exist: "
                f"{self.source_directory}"
            )
            return summary

        if not self.source_directory.is_dir():
            summary["errors"] += 1
            summary["details"].append(
                f"Source path is not a directory: "
                f"{self.source_directory}"
            )
            return summary

        if file_dates is None:
            file_dates = {}

        try:
            files = list(self.source_directory.iterdir())
        except OSError as exc:
            summary["errors"] += 1
            summary["details"].append(
                f"Could not read directory: {exc}"
            )
            return summary

        for file_path in files:
            if not file_path.is_file():
                continue

            if file_path.name.startswith("."):
                summary["skipped"] += 1
                summary["details"].append(
                    f"Skipped hidden file: {file_path.name}"
                )
                continue

            extension = self.normalize_extension(file_path)

            if not extension:
                category = "Unknown"
                summary["details"].append(
                    f"Missing/malformed extension: {file_path.name}"
                )
            else:
                category = self.get_category(file_path)

            # Validate optional date metadata.
            if file_path.name in file_dates:
                raw_date = file_dates[file_path.name]
                parsed_date = self.parse_date(raw_date)

                if raw_date and parsed_date is None:
                    summary["invalid_dates"] += 1
                    summary["details"].append(
                        f"Malformed date for {file_path.name}: "
                        f"{raw_date}"
                    )

            category_directory = (
                self.destination_directory / category
            )

            destination = (
                category_directory / file_path.name
            )

            try:
                destination = self.get_unique_destination(
                    destination
                )

                if file_path.resolve() == destination.resolve():
                    summary["skipped"] += 1
                    summary["details"].append(
                        f"Already organized: {file_path.name}"
                    )
                    continue

                if dry_run:
                    summary["details"].append(
                        f"Would move: {file_path.name} -> "
                        f"{category}/{destination.name}"
                    )
                else:
                    category_directory.mkdir(
                        parents=True,
                        exist_ok=True
                    )

                    shutil.move(
                        str(file_path),
                        str(destination)
                    )

                    summary["details"].append(
                        f"Moved: {file_path.name} -> "
                        f"{category}/{destination.name}"
                    )

                summary["moved"] += 1
                summary["by_category"][category] = (
                    summary["by_category"].get(category, 0) + 1
                )

            except (OSError, shutil.Error) as exc:
                summary["errors"] += 1
                summary["details"].append(
                    f"Error processing {file_path.name}: {exc}"
                )

        return summary


def organize_documents(
    source_directory,
    destination_directory=None,
    file_dates=None,
    dry_run=False
):
    """Convenience wrapper for DocumentOrganizer."""
    organizer = DocumentOrganizer(
        source_directory,
        destination_directory
    )

    return organizer.organize(
        file_dates=file_dates,
        dry_run=dry_run
    )


def print_summary(summary):
    """Display the final organization summary."""
    print("\nDocument Organizer Summary")
    print("=" * 35)

    print(f"Files moved:    {summary['moved']}")
    print(f"Files skipped:  {summary['skipped']}")
    print(f"Errors:         {summary['errors']}")
    print(f"Invalid dates:  {summary['invalid_dates']}")

    if summary["by_category"]:
        print("\nFiles by category:")
        for category, count in sorted(
            summary["by_category"].items()
        ):
            print(f"  {category}: {count}")

    if summary["details"]:
        print("\nDetails:")
        for detail in summary["details"]:
            print(f"  - {detail}")


if __name__ == "__main__":
    source_directory = "documents"

    # Optional date metadata.
    # Invalid dates are detected and reported safely.
    file_dates = {
        "report.pdf": "2026-09-15",
        "notes.docx": "15/09/2026",
        "old.txt": "not-a-date",
        "broken.pdf": "2026-99-99",
    }

    # True = preview only, False = actually organize files.
    dry_run = False

    result = organize_documents(
        source_directory=source_directory,
        file_dates=file_dates,
        dry_run=dry_run
    )

    print_summary(result)
