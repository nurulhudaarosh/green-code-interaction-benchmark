def validate_config(config):
    """
    Validate configuration against common configuration-validation rules.

    Rules:
    - Config must be a dictionary.
    - Required fields must exist and be non-empty.
    - Unknown fields are rejected.
    - String fields cannot be blank after stripping whitespace.
    - Numeric fields must have valid types and ranges.
    - Boolean fields must be actual booleans.
    - List fields must contain valid values.
    - Nested configuration is validated where applicable.

    Returns:
        (True, []) when valid
        (False, [error messages]) when invalid
    """

    if not isinstance(config, dict):
        return False, ["Configuration must be a dictionary."]

    errors = []

    allowed_fields = {
        "name",
        "environment",
        "debug",
        "port",
        "host",
        "timeout",
        "retries",
        "log_level",
        "allowed_origins",
        "database",
    }

    required_fields = {
        "name",
        "environment",
        "port",
    }

    # Unknown fields
    for key in config:
        if key not in allowed_fields:
            errors.append(f"Unknown configuration field: {key}")

    # Required fields
    for field in required_fields:
        if field not in config:
            errors.append(f"Missing required field: {field}")

    # String validation
    string_fields = {
        "name",
        "environment",
        "host",
        "log_level",
    }

    for field in string_fields:
        if field in config:
            value = config[field]

            if not isinstance(value, str):
                errors.append(f"{field} must be a string.")
            elif not value.strip():
                errors.append(f"{field} cannot be empty or whitespace.")

    # Environment validation
    if "environment" in config and isinstance(config["environment"], str):
        valid_environments = {"development", "testing", "staging", "production"}

        if config["environment"].strip().lower() not in valid_environments:
            errors.append(
                "environment must be one of: "
                "development, testing, staging, production."
            )

    # Boolean validation
    if "debug" in config and not isinstance(config["debug"], bool):
        errors.append("debug must be a boolean.")

    # Port validation
    if "port" in config:
        port = config["port"]

        if isinstance(port, bool) or not isinstance(port, int):
            errors.append("port must be an integer.")
        elif not 1 <= port <= 65535:
            errors.append("port must be between 1 and 65535.")

    # Timeout validation
    if "timeout" in config:
        timeout = config["timeout"]

        if isinstance(timeout, bool) or not isinstance(timeout, (int, float)):
            errors.append("timeout must be a number.")
        elif timeout <= 0:
            errors.append("timeout must be greater than 0.")

    # Retry validation
    if "retries" in config:
        retries = config["retries"]

        if isinstance(retries, bool) or not isinstance(retries, int):
            errors.append("retries must be an integer.")
        elif retries < 0:
            errors.append("retries cannot be negative.")

    # Log level validation
    if "log_level" in config and isinstance(config["log_level"], str):
        valid_levels = {"debug", "info", "warning", "error", "critical"}

        if config["log_level"].strip().lower() not in valid_levels:
            errors.append(
                "log_level must be one of: "
                "debug, info, warning, error, critical."
            )

    # Allowed origins validation
    if "allowed_origins" in config:
        origins = config["allowed_origins"]

        if not isinstance(origins, list):
            errors.append("allowed_origins must be a list.")
        else:
            for index, origin in enumerate(origins):
                if not isinstance(origin, str):
                    errors.append(
                        f"allowed_origins[{index}] must be a string."
                    )
                elif not origin.strip():
                    errors.append(
                        f"allowed_origins[{index}] cannot be empty or whitespace."
                    )

    # Database configuration validation
    if "database" in config:
        database = config["database"]

        if not isinstance(database, dict):
            errors.append("database must be a dictionary.")
        else:
            allowed_database_fields = {
                "host",
                "port",
                "name",
                "username",
                "password",
            }

            for key in database:
                if key not in allowed_database_fields:
                    errors.append(
                        f"Unknown database configuration field: {key}"
                    )

            for field in ("host", "name", "username"):
                if field in database:
                    value = database[field]

                    if not isinstance(value, str):
                        errors.append(
                            f"database.{field} must be a string."
                        )
                    elif not value.strip():
                        errors.append(
                            f"database.{field} cannot be empty or whitespace."
                        )

            if "password" in database and not isinstance(
                database["password"], str
            ):
                errors.append("database.password must be a string.")

            if "port" in database:
                db_port = database["port"]

                if isinstance(db_port, bool) or not isinstance(db_port, int):
                    errors.append("database.port must be an integer.")
                elif not 1 <= db_port <= 65535:
                    errors.append(
                        "database.port must be between 1 and 65535."
                    )

    # Cross-field validation
    if (
        config.get("environment") == "production"
        and config.get("debug") is True
    ):
        errors.append("debug cannot be enabled in production.")

    if (
        "retries" in config
        and "timeout" in config
        and isinstance(config["retries"], int)
        and isinstance(config["timeout"], (int, float))
        and not isinstance(config["timeout"], bool)
        and config["retries"] > 0
        and config["timeout"] <= 0
    ):
        errors.append(
            "timeout must be greater than 0 when retries are enabled."
        )

    return len(errors) == 0, errors


def validate_and_raise(config):
    """Validate configuration and raise ValueError if invalid."""
    valid, errors = validate_config(config)

    if not valid:
        raise ValueError("Invalid configuration:\n- " + "\n- ".join(errors))

    return True

    config = {

    valid, errors = validate_config(config)

            print(f"- {error}")
