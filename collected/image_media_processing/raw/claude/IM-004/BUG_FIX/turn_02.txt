```python
import os
from pathlib import Path
from PIL import Image
import numpy as np


def preprocess_images(input_dir, output_dir, flat_fill=0, eps=1e-6):
    """
    Grayscale and min-max normalize every image in input_dir to the full
    0-255 range, saving results as PNG in output_dir.

    Args:
        input_dir: directory of source images.
        output_dir: directory to write normalized PNGs.
        flat_fill: pixel value (0-255) used for uniformly-colored images,
                   since there's no dynamic range to stretch.
        eps: threshold below which (max - min) is treated as zero, to
             avoid blowing up near-flat images due to floating point noise.

    Returns:
        dict: {filename: {"min": original_min, "max": original_max}}
    """
    input_dir = Path(input_dir)
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    valid_ext = {".png", ".jpg", ".jpeg", ".bmp", ".tif", ".tiff", ".webp"}
    stats = {}

    for file in sorted(input_dir.iterdir()):
        if not file.is_file() or file.suffix.lower() not in valid_ext:
            continue

        try:
            img = Image.open(file).convert("L")
        except Exception as e:
            print(f"Skipping {file.name}: {e}")
            continue

        arr = np.asarray(img, dtype=np.float64)
        orig_min = float(arr.min())
        orig_max = float(arr.max())
        span = orig_max - orig_min

        if span <= eps:
            # Uniformly-colored (flat) image: no dynamic range to stretch.
            # Fill with a constant instead of dividing by ~0.
            norm = np.full_like(arr, fill_value=flat_fill)
        else:
            norm = (arr - orig_min) / span * 255.0

        norm = np.clip(norm, 0, 255).astype(np.uint8)

        out_path = output_dir / (file.stem + ".png")
        Image.fromarray(norm, mode="L").save(out_path, format="PNG")

        stats[file.name] = {"min": orig_min, "max": orig_max}

    return stats
```
