```python
"""
Recursively generate aspect-ratio-preserving thumbnails, mirroring the
input directory tree into a cache directory.
"""

import os
from pathlib import Path
from PIL import Image

IMAGE_EXTENSIONS = {".jpg", ".jpeg", ".png", ".gif", ".bmp", ".tiff", ".webp"}
JPEG_EXTENSIONS = {".jpg", ".jpeg"}


def generate_thumbnails(input_dir, cache_dir, max_dim=128):
    """
    Walk input_dir recursively and create a thumbnail (longest side
    <= max_dim, aspect ratio preserved) for every image file found,
    writing each one to the same relative path under cache_dir.

    Thumbnails whose file already exists and is newer than the source
    image are skipped, so repeated calls are cheap.

    Args:
        input_dir: Root directory to scan for images.
        cache_dir: Root directory to mirror thumbnails into.
        max_dim: Max width/height in pixels for the longest side.

    Returns:
        List of (source_path, thumbnail_path) tuples for thumbnails
        that were newly created or refreshed.
    """
    input_dir = Path(input_dir)
    cache_dir = Path(cache_dir)

    if not input_dir.is_dir():
        raise NotADirectoryError(f"Input directory not found: {input_dir}")

    generated = []

    for root, _dirs, files in os.walk(input_dir):
        root = Path(root)
        rel_dir = root.relative_to(input_dir)
        dest_dir = cache_dir / rel_dir

        # Mirror the directory itself, even if it turns out to contain
        # no images (or no files at all).
        dest_dir.mkdir(parents=True, exist_ok=True)

        for filename in files:
            src_path = root / filename
            ext = src_path.suffix.lower()
            if ext not in IMAGE_EXTENSIONS:
                continue

            dest_path = dest_dir / filename

            # Skip if an up-to-date thumbnail already exists.
            if dest_path.exists() and dest_path.stat().st_mtime >= src_path.stat().st_mtime:
                continue

            try:
                with Image.open(src_path) as img:
                    img.load()

                    # JPEG can't hold alpha; flatten to RGB when needed.
                    if ext in JPEG_EXTENSIONS and img.mode in ("RGBA", "P", "LA"):
                        img = img.convert("RGB")
                    elif img.mode == "P":
                        img = img.convert("RGBA")

                    # In-place, aspect-ratio-preserving resize.
                    img.thumbnail((max_dim, max_dim), Image.LANCZOS)
                    img.save(dest_path)
            except Exception as e:
                print(f"Skipping {src_path}: {e}")
                continue

            generated.append((str(src_path), str(dest_path)))

    return generated


if __name__ == "__main__":
    import sys

    if len(sys.argv) < 3:
        print("Usage: python thumbnails.py <input_dir> <cache_dir> [max_dim]")
        sys.exit(1)

    in_dir, out_dir = sys.argv[1], sys.argv[2]
    dim = int(sys.argv[3]) if len(sys.argv) > 3 else 128

    results = generate_thumbnails(in_dir, out_dir, dim)
    print(f"Generated/updated {len(results)} thumbnail(s).")
```
