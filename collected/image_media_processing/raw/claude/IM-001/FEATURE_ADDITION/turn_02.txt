```python
from pathlib import Path
from PIL import Image

def batch_resize(input_dir, output_dir, target_size, resample_filter=Image.LANCZOS):
    input_dir, output_dir = Path(input_dir), Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    count = 0
    for path in input_dir.iterdir():
        if path.suffix.lower() in (".png", ".jpg", ".jpeg"):
            with Image.open(path) as img:
                img = img.convert("RGBA" if path.suffix.lower() == ".png" else "RGB")
                resized = img.resize(target_size, resample_filter)
                resized.save(output_dir / path.name)
            count += 1

    return count
```
