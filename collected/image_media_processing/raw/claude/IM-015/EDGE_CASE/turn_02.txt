```python
from pathlib import Path

from PIL import Image

IMAGE_EXTENSIONS = {".png", ".jpg", ".jpeg", ".bmp", ".tif", ".tiff", ".webp"}


def split_channels(input_dir, output_dir):
    """Split every image in input_dir into R, G, B grayscale PNG files.

    For an input named "photo.jpg", writes photo_R.png, photo_G.png and
    photo_B.png to output_dir. PNG is used so the channel data is saved
    losslessly. Returns a list of the paths written.
    """
    input_dir, output_dir = Path(input_dir), Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    written = []
    for path in sorted(input_dir.iterdir()):
        if not path.is_file() or path.suffix.lower() not in IMAGE_EXTENSIONS:
            continue
        with Image.open(path) as img:
            rgb = img.convert("RGB")  # handles RGBA, palette, grayscale, etc.
            channels = rgb.split()    # each is a mode "L" (grayscale) image
        for name, channel in zip("RGB", channels):
            out_path = output_dir / f"{path.stem}_{name}.png"
            channel.save(out_path)
            written.append(out_path)
    return written


def merge_channels(r_path, g_path, b_path, output_path):
    """Combine three grayscale images (R, G, B) into one RGB image."""
    channels = []
    for p in (r_path, g_path, b_path):
        with Image.open(p) as img:
            channels.append(img.convert("L"))  # force single-channel

    if not (channels[0].size == channels[1].size == channels[2].size):
        sizes = [c.size for c in channels]
        raise ValueError(f"Channel images must be the same size, got {sizes}")

    merged = Image.merge("RGB", channels)
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    merged.save(output_path)
    return output_path
```
