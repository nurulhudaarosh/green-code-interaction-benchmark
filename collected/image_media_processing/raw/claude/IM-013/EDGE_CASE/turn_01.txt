"""Build a grid contact sheet from a directory of images."""

import math
import re
from pathlib import Path

from PIL import Image, ImageOps

IMAGE_EXTS = {".jpg", ".jpeg", ".png", ".gif", ".bmp", ".webp", ".tif", ".tiff"}


def _natural_key(path):
    """Sort 'img2' before 'img10'."""
    return [int(t) if t.isdigit() else t.lower() for t in re.split(r"(\d+)", path.name)]


def build_contact_sheet(input_dir, output_path, thumb_size, columns,
                        padding=8, background=(255, 255, 255)):
    """Tile all images in `input_dir` into a grid and save it to `output_path`.

    Args:
        input_dir:   Directory containing images (non-recursive).
        output_path: Where to save the sheet (format inferred from extension).
        thumb_size:  int (square cell) or (width, height) of each thumbnail cell.
        columns:     Fixed number of columns; rows are computed as needed.
        padding:     Pixels of spacing between cells and around the border.
        background:  Sheet background color.

    Returns:
        The number of images placed on the sheet.
    """
    if columns < 1:
        raise ValueError("columns must be >= 1")
    if isinstance(thumb_size, int):
        thumb_size = (thumb_size, thumb_size)
    tw, th = thumb_size
    if tw < 1 or th < 1:
        raise ValueError("thumb_size must be positive")

    input_dir = Path(input_dir)
    if not input_dir.is_dir():
        raise NotADirectoryError(f"{input_dir} is not a directory")

    files = sorted(
        (p for p in input_dir.iterdir()
         if p.is_file() and p.suffix.lower() in IMAGE_EXTS),
        key=_natural_key,
    )

    thumbs = []
    for path in files:
        try:
            with Image.open(path) as im:
                im = ImageOps.exif_transpose(im)  # respect camera orientation
                im = im.convert("RGBA")
                im.thumbnail((tw, th), Image.LANCZOS)  # keeps aspect ratio
                thumbs.append(im.copy())
        except Exception as exc:  # unreadable/corrupt file: skip it
            print(f"Skipping {path.name}: {exc}")

    if not thumbs:
        raise ValueError(f"No readable images found in {input_dir}")

    rows = math.ceil(len(thumbs) / columns)
    sheet_w = columns * tw + (columns + 1) * padding
    sheet_h = rows * th + (rows + 1) * padding
    sheet = Image.new("RGB", (sheet_w, sheet_h), background)

    for i, thumb in enumerate(thumbs):
        row, col = divmod(i, columns)
        # Center each thumbnail within its cell.
        x = padding + col * (tw + padding) + (tw - thumb.width) // 2
        y = padding + row * (th + padding) + (th - thumb.height) // 2
        sheet.paste(thumb, (x, y), mask=thumb.getchannel("A"))

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    sheet.save(output_path)
    return len(thumbs)


if __name__ == "__main__":
    import argparse

    ap = argparse.ArgumentParser(description="Build an image contact sheet.")
    ap.add_argument("input_dir")
    ap.add_argument("output_path")
    ap.add_argument("--thumb", type=int, default=200, help="square cell size in px")
    ap.add_argument("--columns", type=int, default=5)
    args = ap.parse_args()
    n = build_contact_sheet(args.input_dir, args.output_path, args.thumb, args.columns)
    print(f"Placed {n} images in {args.output_path}")
