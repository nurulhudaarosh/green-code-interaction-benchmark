import os
from PIL import Image


def extract_metadata(input_dir):
    """
    Scan input_dir for image files and return a sorted list of dicts,
    each describing one image's metadata.

    Each dict contains:
        - filename: str, the file's base name
        - format: str, image format as reported by Pillow (e.g. "JPEG", "PNG")
        - mode: str, Pillow image mode (e.g. "RGB", "RGBA", "L")
        - width: int, image width in pixels
        - height: int, image height in pixels
        - file_size_bytes: int, size of the file on disk in bytes

    The returned list is sorted by filename. Files that exist in input_dir
    but cannot be opened as images (e.g. non-image files, corrupted files)
    are silently skipped.

    Args:
        input_dir: path to the directory containing images.

    Returns:
        list[dict]: metadata dicts, sorted by filename.

    Raises:
        NotADirectoryError: if input_dir does not exist or is not a directory.
    """
    if not os.path.isdir(input_dir):
        raise NotADirectoryError(f"Not a directory: {input_dir}")

    results = []

    for entry in os.scandir(input_dir):
        if not entry.is_file():
            continue

        filepath = entry.path
        try:
            with Image.open(filepath) as img:
                width, height = img.size
                metadata = {
                    "filename": entry.name,
                    "format": img.format,
                    "mode": img.mode,
                    "width": width,
                    "height": height,
                    "file_size_bytes": os.path.getsize(filepath),
                }
            results.append(metadata)
        except Exception:
            # Not a valid image (or unreadable) - skip it.
            continue

    results.sort(key=lambda d: d["filename"])
    return results


if __name__ == "__main__":
    import sys
    import json

    if len(sys.argv) != 2:
        print(f"Usage: python {sys.argv[0]} <input_dir>")
        sys.exit(1)

    data = extract_metadata(sys.argv[1])
    print(json.dumps(data, indent=2))
