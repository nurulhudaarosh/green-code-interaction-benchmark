```python
import os
from PIL import Image

def extract_metadata(input_dir):
    """
    Extract metadata for every image file in input_dir.

    Returns a list of dicts, sorted by filename, each containing:
        - filename
        - format
        - mode
        - width
        - height
        - file_size_bytes
        - aspect_ratio (width / height, rounded to 3 decimal places)
    """
    results = []

    for entry in os.listdir(input_dir):
        full_path = os.path.join(input_dir, entry)

        if not os.path.isfile(full_path):
            continue

        try:
            with Image.open(full_path) as img:
                width, height = img.size
                aspect_ratio = round(width / height, 3) if height else None
                results.append({
                    "filename": entry,
                    "format": img.format,
                    "mode": img.mode,
                    "width": width,
                    "height": height,
                    "file_size_bytes": os.path.getsize(full_path),
                    "aspect_ratio": aspect_ratio,
                })
        except (IOError, OSError):
            # Not a valid/openable image file — skip it
            continue

    results.sort(key=lambda d: d["filename"])
    return results
```

**Notes:**
- `aspect_ratio` is `None` in the (essentially impossible for real images, but safe to guard) case of `height == 0`, to avoid a `ZeroDivisionError`.
- Rounding uses Python's built-in `round`, which follows banker's rounding on ties — fine for this use case since aspect ratios rarely land exactly on a rounding boundary.
- Everything else (skip logic, sort by filename, sizes from `os.path.getsize`) is unchanged.
