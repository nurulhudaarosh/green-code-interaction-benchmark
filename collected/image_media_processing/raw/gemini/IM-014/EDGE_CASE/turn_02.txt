### Key Adjustments for Edge Cases

1. **Zero-Byte File Guard**: Avoids division-by-zero errors (`ZeroDivisionError`) if `orig_size` is 0 bytes.
2. **Explicit 1x1 Pixel Handling**: Ensure palette (`P`) and alpha (`RGBA`/`LA`/`PA`) modes are handled properly even for 1x1 images, flattening transparency against a solid white background rather than dropping alpha channels unpredictably.
3. **Division Safeguard**: Handles cases where `orig_size == 0` gracefully by setting compression ratio to `0.00%`.

```python
import os
from pathlib import Path
from PIL import Image

def bulk_reencode(input_dir: str, output_dir: str, quality: int = 85) -> None:
    """
    Re-encodes all images in input_dir to JPEG format at the given quality setting,
    saving them in output_dir and printing size/compression stats per file.
    Handles 1x1 pixel images and zero-byte file edge cases.
    """
    input_path = Path(input_dir)
    output_path = Path(output_dir)
    
    # Ensure output directory exists
    output_path.mkdir(parents=True, exist_ok=True)
    
    # Supported input extensions
    valid_extensions = {".jpg", ".jpeg", ".png", ".webp", ".bmp", ".tiff", ".gif"}
    
    # Header for report
    print(f"{'Filename':<30} | {'Original (KB)':<15} | {'Re-encoded (KB)':<15} | {'Ratio':<10}")
    print("-" * 78)
    
    total_orig_size = 0
    total_new_size = 0
    processed_count = 0

    for file in input_path.iterdir():
        if file.is_file() and file.suffix.lower() in valid_extensions:
            out_filename = file.stem + ".jpg"
            out_file = output_path / out_filename
            
            try:
                orig_size = file.stat().st_size
                
                with Image.open(file) as img:
                    # Convert transparent or palette modes cleanly to RGB
                    if img.mode in ("RGBA", "LA") or (img.mode == "P" and "transparency" in img.info):
                        # Create a solid white background for transparent pixels (even 1x1)
                        background = Image.new("RGB", img.size, (255, 255, 255))
                        rgba_img = img.convert("RGBA")
                        background.paste(rgba_img, mask=rgba_img.split()[3])
                        img = background
                    elif img.mode != "RGB":
                        img = img.convert("RGB")
                    
                    # Save as JPEG with specified quality
                    img.save(out_file, "JPEG", quality=quality, optimize=True)
                
                new_size = out_file.stat().st_size
                
                # Zero-division guard for empty/corrupted files
                if orig_size > 0:
                    compression_ratio = (1 - (new_size / orig_size)) * 100
                else:
                    compression_ratio = 0.0
                
                # Accumulate totals
                total_orig_size += orig_size
                total_new_size += new_size
                processed_count += 1
                
                # Report per file
                print(
                    f"{file.name:<30} | "
                    f"{orig_size / 1024:<15.2f} | "
                    f"{new_size / 1024:<15.2f} | "
                    f"{compression_ratio:>8.2f}%"
                )
                
            except Exception as e:
                print(f"Error processing {file.name}: {e}")

    # Summary report
    print("-" * 78)
    if processed_count > 0:
        overall_ratio = (1 - (total_new_size / total_orig_size)) * 100 if total_orig_size > 0 else 0.0
        print(f"Processed {processed_count} files.")
        print(f"Total Original Size:   {total_orig_size / (1024 * 1024):.2f} MB")
        print(f"Total Re-encoded Size: {total_new_size / (1024 * 1024):.2f} MB")
        print(f"Overall Space Saved:   {overall_ratio:.2f}%")
    else:
        print("No valid images found in the input directory.")

```

### Why a 1x1 JPEG might show a negative compression ratio

When converting a 1x1 PNG or GIF (which can be as small as 60–100 bytes) to JPEG, the output JPEG file size will usually increase to ~500–1000 bytes due to required JPEG header structure (JFIF header, quantization tables, and Huffman tables). The ratio calculation will reflect this cleanly as a negative percentage (e.g. `-400.00%`) without throwing error exceptions.
