Here is the corrected implementation. Setting `expand=True` expands the canvas during rotation to fit the full rotated bounding box so no content is lost before the center-crop/pad step.

```python
import os
from PIL import Image

def rotate_and_crop(input_dir, output_dir, angle_degrees, crop_size):
    """
    Rotates all images in `input_dir` by `angle_degrees` with `expand=True`
    to preserve all rotated image content, then center-crops or pads each
    image with black to match `crop_size`.
    
    Parameters:
        input_dir (str): Path to directory containing source images.
        output_dir (str): Path to directory where processed images will be saved.
        angle_degrees (float): Angle to rotate images counter-clockwise.
        crop_size (tuple): Target dimensions as (width, height).
    """
    os.makedirs(output_dir, exist_ok=True)
    target_width, target_height = crop_size

    valid_extensions = ('.jpg', '.jpeg', '.png', '.bmp', '.tiff', '.webp')

    for filename in os.listdir(input_dir):
        if not filename.lower().endswith(valid_extensions):
            continue

        file_path = os.path.join(input_dir, filename)
        
        with Image.open(file_path) as img:
            mode = img.mode
            bg_color = (0, 0, 0, 0) if mode == 'RGBA' else (0, 0, 0)

            # 1. Expand canvas so no image content is clipped during rotation
            rotated = img.rotate(
                angle_degrees, 
                expand=True, 
                resample=Image.BICUBIC, 
                fillcolor=bg_color
            )

            # 2. Create fixed target canvas
            result = Image.new(mode, (target_width, target_height), bg_color)
            rot_w, rot_h = rotated.size

            # 3. Calculate source region to crop from expanded rotated image
            src_left = max(0, (rot_w - target_width) // 2)
            src_top = max(0, (rot_h - target_height) // 2)
            src_right = min(rot_w, src_left + target_width)
            src_bottom = min(rot_h, src_top + target_height)

            cropped = rotated.crop((src_left, src_top, src_right, src_bottom))

            # 4. Calculate destination paste offset if rotated image is smaller than target
            dst_left = max(0, (target_width - rot_w) // 2)
            dst_top = max(0, (target_height - rot_h) // 2)

            result.paste(cropped, (dst_left, dst_top))

            output_path = os.path.join(output_dir, filename)
            result.save(output_path)

```

### Key Fixes

* **`expand=True`**: Calculates the new bounding rectangle after rotation, preventing corner clipping.
* **`fillcolor=bg_color`**: Passes background color directly to `rotate()` so newly revealed canvas corners during expansion match the output padding color.
