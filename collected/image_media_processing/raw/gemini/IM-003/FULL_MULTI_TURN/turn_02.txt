from pathlib import Path
from PIL import Image

def convert_format(input_dir: str | Path, output_dir: str | Path, target_format: str) -> None:
    """
    Recursively converts image files from input_dir to target_format (JPEG or PNG).
    Safely flattens transparent channels onto a white background for JPEG output.
    """
    in_path = Path(input_dir)
    out_path = Path(output_dir)
    
    fmt = target_format.upper().strip()
    if fmt == "JPG":
        fmt = "JPEG"
    if fmt not in ("JPEG", "PNG"):
        raise ValueError(f"Unsupported target format: {target_format}. Choose 'JPEG' or 'PNG'.")

    ext = ".jpg" if fmt == "JPEG" else ".png"
    valid_extensions = {".png", ".jpg", ".jpeg", ".webp", ".bmp", ".tiff", ".tif", ".gif"}

    for file_path in in_path.rglob("*"):
        if file_path.is_file() and file_path.suffix.lower() in valid_extensions:
            relative_path = file_path.relative_to(in_path)
            destination = out_path / relative_path.with_suffix(ext)
            destination.parent.mkdir(parents=True, exist_ok=True)

            try:
                with Image.open(file_path) as img:
                    if fmt == "JPEG":
                        # Always convert to RGBA to ensure standard 4-channel representation
                        # for images with transparency (RGBA, LA, or Palette 'P' with alpha)
                        rgba_img = img.convert("RGBA")
                        
                        # Create a solid white background of the same size
                        background = Image.new("RGBA", rgba_img.size, (255, 255, 255, 255))
                        
                        # Alpha composite: overlays transparent pixels onto white background
                        composited = Image.alpha_composite(background, rgba_img)
                        
                        # Safely convert to 3-channel RGB for JPEG saving
                        final_img = composited.convert("RGB")
                        final_img.save(destination, format="JPEG", quality=95)

                    elif fmt == "PNG":
                        if img.mode == "CMYK":
                            final_img = img.convert("RGB")
                        else:
                            final_img = img
                        
                        final_img.save(destination, format="PNG")

                    print(f"Converted: {file_path.name} -> {destination}")

            except Exception as e:
                print(f"Error processing {file_path}: {e}")
