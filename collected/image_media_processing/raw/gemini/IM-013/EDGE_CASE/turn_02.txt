Here is the updated implementation. When `input_dir` contains zero valid images, it creates a minimal 1x1 pixel image (using the specified `bg_color`), saves it to `output_path`, and returns `(0, columns)`. For non-empty directories, it returns `(rows, columns)`.

```python
import math
import os
from PIL import Image, ImageOps

def build_contact_sheet(
    input_dir: str, 
    output_path: str, 
    thumb_size: tuple[int, int], 
    columns: int, 
    bg_color: tuple[int, int, int] = (255, 255, 255)
) -> tuple[int, int]:
    """
    Tiles all images in a directory into a grid contact sheet.

    :param input_dir: Path to directory containing images.
    :param output_path: Output file path for the contact sheet.
    :param thumb_size: Tuple (width, height) specifying thumbnail dimensions.
    :param columns: Number of columns in the grid.
    :param bg_color: RGB background color tuple (default is white).
    :return: A tuple of (rows, columns). Returns (0, columns) if no images are found.
    """
    valid_extensions = ('.jpg', '.jpeg', '.png', '.bmp', '.webp', '.tiff')
    
    # 1. Gather all valid image file paths
    image_paths = []
    if os.path.exists(input_dir) and os.path.isdir(input_dir):
        image_paths = [
            os.path.join(input_dir, f) for f in sorted(os.listdir(input_dir))
            if os.path.isfile(os.path.join(input_dir, f)) and f.lower().endswith(valid_extensions)
        ]
    
    # 2. Edge Case: Handle 0 valid images
    if not image_paths:
        # Create a minimal 1x1 image, save it, and return (0, columns)
        empty_sheet = Image.new('RGB', (1, 1), color=bg_color)
        empty_sheet.save(output_path)
        return (0, columns)

    # 3. Calculate grid geometry
    thumb_w, thumb_h = thumb_size
    total_images = len(image_paths)
    rows = math.ceil(total_images / columns)
    
    sheet_width = columns * thumb_w
    sheet_height = rows * thumb_h

    # 4. Create canvas image
    contact_sheet = Image.new('RGB', (sheet_width, sheet_height), color=bg_color)

    # 5. Process and paste each image
    for index, path in enumerate(image_paths):
        col = index % columns
        row = index // columns
        
        x = col * thumb_w
        y = row * thumb_h

        try:
            with Image.open(path) as img:
                if img.mode in ('RGBA', 'P', 'LA'):
                    img = img.convert('RGB')
                
                thumb = ImageOps.fit(img, thumb_size, Image.Resampling.LANCZOS)
                contact_sheet.paste(thumb, (x, y))
        except Exception as e:
            print(f"Warning: Skipping unreadable image '{path}': {e}")

    # 6. Save final contact sheet and return grid shape
    contact_sheet.save(output_path)
    return (rows, columns)

```

### Key Updates:

* **Zero-Image Edge Case Handling:** Checks `if not image_paths:`, creates a minimal `1x1` canvas of `bg_color`, saves it, and directly returns `(0, columns)`.
* **Directory Validation:** Handles missing/invalid directories gracefully without throwing unhandled exceptions before file checking.
* **Return Type:** Always returns `(rows, columns)` as a tuple.
