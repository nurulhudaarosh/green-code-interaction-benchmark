Here is the updated function with an `align` parameter supporting `'center'` (default) and `'top-left'`:

```python
import os
import cv2

def letterbox_resize(
    input_dir,
    output_dir,
    target_size=(640, 640),
    pad_color=(0, 0, 0),
    align='center'
):
    """
    Resizes images while preserving aspect ratio and pads to target_size.

    :param input_dir: Path to input images directory.
    :param output_dir: Path to save resized images.
    :param target_size: Tuple (target_w, target_h) specifying exact output dimensions.
    :param pad_color: Tuple (B, G, R) for padding color (default: black).
    :param align: Positioning of resized content within canvas ('center' or 'top-left').
    """
    if align not in ('center', 'top-left'):
        raise ValueError("align must be either 'center' or 'top-left'")

    os.makedirs(output_dir, exist_ok=True)
    target_w, target_h = target_size
    valid_extensions = ('.jpg', '.jpeg', '.png', '.bmp', '.webp', '.tiff')

    for filename in os.listdir(input_dir):
        if not filename.lower().endswith(valid_extensions):
            continue

        img_path = os.path.join(input_dir, filename)
        img = cv2.imread(img_path)

        if img is None:
            continue

        src_h, src_w = img.shape[:2]

        # Calculate scale factor and target resized dimensions
        scale = min(target_w / src_w, target_h / src_h)
        new_w = int(round(src_w * scale))
        new_h = int(round(src_h * scale))

        # Resize image
        interpolation = cv2.INTER_AREA if scale < 1.0 else cv2.INTER_LINEAR
        resized_img = cv2.resize(img, (new_w, new_h), interpolation=interpolation)

        # Total required padding
        pad_w = target_w - new_w
        pad_h = target_h - new_h

        # Calculate padding borders based on alignment
        if align == 'top-left':
            top = 0
            bottom = pad_h
            left = 0
            right = pad_w
        else:  # 'center'
            top = pad_h // 2
            bottom = pad_h - top
            left = pad_w // 2
            right = pad_w - left

        # Apply border padding
        letterboxed = cv2.copyMakeBorder(
            resized_img,
            top, bottom, left, right,
            borderType=cv2.BORDER_CONSTANT,
            value=pad_color
        )

        output_path = os.path.join(output_dir, filename)
        cv2.imwrite(output_path, letterboxed)

```

### Alignment Mechanics

* **`'center'`**: Distributes `pad_h` and `pad_w` equally across top/bottom and left/right.
* **`'top-left'`**: Sets `top = 0` and `left = 0`, directing all remaining padding to `bottom` and `right`. The image is anchored at coordinate `(0, 0)` of the output canvas.
