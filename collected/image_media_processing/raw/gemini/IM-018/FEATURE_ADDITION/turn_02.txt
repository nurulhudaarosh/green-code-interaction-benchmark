Here is the extended implementation adding the `distance_metric` parameter with support for both `'l1'` (Manhattan distance) and `'chi_squared'` (Chi-Squared distance).

```python
import os
import glob
import cv2
import numpy as np

class UnionFind:
    """Disjoint Set Union (DSU) to track connected components transitively."""
    def __init__(self, elements):
        self.parent = {el: el for el in elements}

    def find(self, item):
        if self.parent[item] != item:
            self.parent[item] = self.find(self.parent[item])  # Path compression
        return self.parent[item]

    def union(self, item1, item2):
        root1 = self.find(item1)
        root2 = self.find(item2)
        if root1 != root2:
            self.parent[root2] = root1

    def get_clusters(self):
        clusters = {}
        for item in self.parent:
            root = self.find(item)
            clusters.setdefault(root, []).append(item)
        return [group for group in clusters.values() if len(group) > 1]


def compute_normalized_histogram(image_path, bins=256):
    """Loads an image in grayscale, computes its 1D histogram, and normalizes it."""
    img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        return None
    
    hist = cv2.calcHist([img], [0], None, [bins], [0, 256]).ravel()
    norm = np.linalg.norm(hist, ord=1)
    
    if norm == 0:
        return None
    
    return hist / norm


def compute_distance(hist1, hist2, metric='l1', eps=1e-10):
    """Computes histogram distance given a metric string."""
    if metric == 'l1':
        # L1 (Manhattan) distance: sum(|h1 - h2|)
        return np.sum(np.abs(hist1 - hist2))
    elif metric == 'chi_squared':
        # Chi-Squared distance: 0.5 * sum((h1 - h2)^2 / (h1 + h2 + eps))
        return 0.5 * np.sum(((hist1 - hist2) ** 2) / (hist1 + hist2 + eps))
    else:
        raise ValueError(f"Unsupported distance metric '{metric}'. Expected 'l1' or 'chi_squared'.")


def find_near_duplicates(input_dir, threshold=0.1, distance_metric='l1', bins=256):
    """
    Finds clusters of near-duplicate images in input_dir using normalized 
    grayscale histogram distance and transitive Union-Find grouping.

    Parameters:
        input_dir (str): Path to directory containing images.
        threshold (float): Distance threshold below which images are merged. 
                           Typical ranges: 'l1' -> [0.05, 0.2], 'chi_squared' -> [0.01, 0.1].
        distance_metric (str): Distance calculation method ('l1' or 'chi_squared').
        bins (int): Number of bins for the grayscale histogram.

    Returns:
        list[list[str]]: A list of clusters containing file paths of near-duplicate images.
    """
    extensions = ('*.jpg', '*.jpeg', '*.png', '*.bmp', '*.webp', '*.tiff')
    image_paths = []
    for ext in extensions:
        image_paths.extend(glob.glob(os.path.join(input_dir, ext)))
        image_paths.extend(glob.glob(os.path.join(input_dir, ext.upper())))
    
    image_paths = sorted(list(set(image_paths)))
    
    # 1. Precompute normalized histograms
    valid_paths = []
    histograms = []
    for path in image_paths:
        hist = compute_normalized_histogram(path, bins=bins)
        if hist is not None:
            valid_paths.append(path)
            histograms.append(hist)
            
    if not valid_paths:
        return []

    hist_matrix = np.array(histograms)  # Shape: (N, bins)
    
    # 2. Transitive grouping using Union-Find
    uf = UnionFind(valid_paths)
    num_images = len(valid_paths)

    # 3. Pairwise distance comparison
    for i in range(num_images):
        for j in range(i + 1, num_images):
            dist = compute_distance(hist_matrix[i], hist_matrix[j], metric=distance_metric)
            if dist <= threshold:
                uf.union(valid_paths[i], valid_paths[j])

    # 4. Return grouped clusters (clusters with > 1 element)
    return uf.get_clusters()


# Example Usage:
if __name__ == "__main__":
    dir_path = "./sample_images"
    
    # Using default L1 distance
    l1_clusters = find_near_duplicates(dir_path, threshold=0.1, distance_metric='l1')
    
    # Using Chi-Squared distance
    chi2_clusters = find_near_duplicates(dir_path, threshold=0.05, distance_metric='chi_squared')

```

### Formula Mechanics

* **$L_1$ Distance:**

$$d_{L1}(h_1, h_2) = \sum_{k=1}^{256} \vert{}h_{1,k} - h_{2,k}\vert{}$$


* **Chi-Squared Distance:**

$$d_{\chi^2}(h_1, h_2) = \frac{1}{2} \sum_{k=1}^{256} \frac{(h_{1,k} - h_{2,k})^2}{h_{1,k} + h_{2,k} + \epsilon}$$



*Small $\epsilon = 10^{-10}$ prevents division by zero when both bins are zero.*
