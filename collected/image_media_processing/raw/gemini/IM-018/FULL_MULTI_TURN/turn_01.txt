import os
import glob
from PIL import Image
import numpy as np

class UnionFind:
    def __init__(self, items):
        self.parent = {item: item for item in items}

    def find(self, item):
        if self.parent[item] != item:
            self.parent[item] = self.find(self.parent[item])  # Path compression
        return self.parent[item]

    def union(self, item1, item2):
        root1 = self.find(item1)
        root2 = self.find(item2)
        if root1 != root2:
            self.parent[root2] = root1


def compute_grayscale_histogram(image_path, bins=256):
    """Loads an image, converts it to grayscale, and computes a normalized L1 histogram."""
    with Image.open(image_path) as img:
        gray = img.convert('L')
        hist, _ = np.histogram(gray, bins=bins, range=(0, 256))
        
    total_pixels = hist.sum()
    if total_pixels == 0:
        return np.zeros(bins, dtype=np.float64)
        
    return hist.astype(np.float64) / total_pixels


def find_near_duplicates(input_dir, threshold):
    """
    Finds and clusters near-duplicate images in a directory using normalized
    grayscale histogram L1 distance and transitive Union-Find grouping.

    Parameters:
        input_dir (str): Path to directory containing images.
        threshold (float): L1 distance threshold below which images are grouped.

    Returns:
        list[list[str]]: A list of clusters, where each cluster is a list of file paths.
    """
    # Supported image extensions
    extensions = ('*.jpg', '*.jpeg', '*.png', '*.bmp', '*.webp', '*.tiff')
    filepaths = []
    
    for ext in extensions:
        filepaths.extend(glob.glob(os.path.join(input_dir, ext)))
        filepaths.extend(glob.glob(os.path.join(input_dir, ext.upper())))

    filepaths = sorted(list(set(filepaths)))
    if not filepaths:
        return []

    # 1. Compute normalized grayscale histogram for each image
    histograms = {}
    valid_paths = []
    
    for path in filepaths:
        try:
            histograms[path] = compute_grayscale_histogram(path)
            valid_paths.append(path)
        except Exception:
            continue  # Skip unreadable or corrupt image files

    # 2. Compare pairwise L1 distances and merge clusters via Union-Find
    uf = UnionFind(valid_paths)
    num_images = len(valid_paths)

    for i in range(num_images):
        for j in range(i + 1, num_images):
            path_a = valid_paths[i]
            path_b = valid_paths[j]

            # L1 distance: sum of absolute differences between normalized histograms
            l1_dist = np.sum(np.abs(histograms[path_a] - histograms[path_b]))

            if l1_dist <= threshold:
                uf.union(path_a, path_b)

    # 3. Group images by their Union-Find root
    clusters = {}
    for path in valid_paths:
        root = uf.find(path)
        clusters.setdefault(root, []).append(path)

    return list(clusters.values())
