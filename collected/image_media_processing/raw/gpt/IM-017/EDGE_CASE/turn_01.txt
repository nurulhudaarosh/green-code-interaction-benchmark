```python
from PIL import Image, PngImagePlugin


def embed_tags(input_path, output_path, tags):
    """
    Store custom string key/value metadata in PNG text chunks.

    Args:
        input_path: Path to the input PNG.
        output_path: Path where the tagged PNG will be saved.
        tags: Dictionary of string keys and string values.
    """
    if not isinstance(tags, dict):
        raise TypeError("tags must be a dictionary")

    with Image.open(input_path) as img:
        metadata = PngImagePlugin.PngInfo()

        for key, value in tags.items():
            if not isinstance(key, str) or not isinstance(value, str):
                raise TypeError("PNG metadata keys and values must be strings")
            metadata.add_text(key, value)

        img.save(output_path, format="PNG", pnginfo=metadata)


def read_tags(image_path):
    """
    Retrieve custom string key/value metadata from PNG text chunks.

    Args:
        image_path: Path to the PNG image.

    Returns:
        Dictionary containing the embedded text metadata.
    """
    with Image.open(image_path) as img:
        # Pillow exposes PNG text chunks through img.text.
        return dict(getattr(img, "text", {}))
```


This uses Pillow's `PngInfo.add_text()` for standard PNG `tEXt` chunks and `img.text` to retrieve the embedded text chunks.
