```python
import os
import shutil
from PIL import Image


def organize_by_resolution(input_dir, output_dir):
    """
    Copy images into small/medium/large folders based on their
    maximum dimension.

    Categories:
        small:  max(width, height) < 640
        medium: 640 <= max(width, height) <= 1920
        large:  max(width, height) > 1920
    """

    # Create output subfolders
    small_dir = os.path.join(output_dir, "small")
    medium_dir = os.path.join(output_dir, "medium")
    large_dir = os.path.join(output_dir, "large")

    os.makedirs(small_dir, exist_ok=True)
    os.makedirs(medium_dir, exist_ok=True)
    os.makedirs(large_dir, exist_ok=True)

    # Supported image extensions
    image_extensions = {".jpg", ".jpeg", ".png", ".bmp", ".gif", ".tiff", ".webp"}

    for filename in os.listdir(input_dir):
        input_path = os.path.join(input_dir, filename)

        # Skip directories and unsupported files
        if not os.path.isfile(input_path):
            continue

        if os.path.splitext(filename)[1].lower() not in image_extensions:
            continue

        try:
            with Image.open(input_path) as img:
                width, height = img.size
                max_dim = max(width, height)

                # Inclusive boundaries
                if max_dim < 640:
                    destination_dir = small_dir
                elif max_dim <= 1920:
                    destination_dir = medium_dir
                else:
                    destination_dir = large_dir

            shutil.copy2(
                input_path,
                os.path.join(destination_dir, filename)
            )

        except (OSError, Image.UnidentifiedImageError) as e:
            print(f"Skipping {filename}: {e}")
```

### Classification

- `max_dim < 640` → **small**
- `640 <= max_dim <= 1920` → **medium**
- `max_dim > 1920` → **large**

So an image exactly **640×640** goes to `medium`, and one with a maximum dimension exactly **1920** also goes to `medium`.
