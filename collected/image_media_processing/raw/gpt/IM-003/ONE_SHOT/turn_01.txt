from pathlib import Path
from PIL import Image


def convert_format(input_dir, output_dir, target_format):
    """
    Convert all supported images in input_dir recursively to PNG or JPEG.

    Args:
        input_dir: Source directory.
        output_dir: Destination directory.
        target_format: "PNG" or "JPEG".

    Returns:
        dict with processed_count and skipped files.
    """
    input_dir = Path(input_dir)
    output_dir = Path(output_dir)

    target_format = target_format.upper()
    if target_format not in {"PNG", "JPEG"}:
        raise ValueError("target_format must be 'PNG' or 'JPEG'")

    output_dir.mkdir(parents=True, exist_ok=True)

    supported_extensions = {".png", ".jpg", ".jpeg", ".bmp", ".gif", ".tiff", ".webp"}
    skipped = []
    processed_count = 0

    for src_path in input_dir.rglob("*"):
        if not src_path.is_file() or src_path.suffix.lower() not in supported_extensions:
            continue

        relative_path = src_path.relative_to(input_dir)
        dest_dir = output_dir / relative_path.parent
        dest_dir.mkdir(parents=True, exist_ok=True)

        dest_path = dest_dir / f"{src_path.stem}.{target_format.lower()}"

        try:
            with Image.open(src_path) as img:
                if target_format == "JPEG":
                    # JPEG does not support transparency.
                    if img.mode in ("RGBA", "LA") or (
                        img.mode == "P" and "transparency" in img.info
                    ):
                        rgba = img.convert("RGBA")
                        background = Image.new("RGB", rgba.size, "white")
                        background.paste(rgba, mask=rgba.getchannel("A"))
                        img = background
                    else:
                        img = img.convert("RGB")

                    img.save(dest_path, "JPEG", quality=95)

                else:  # PNG
                    img.save(dest_path, "PNG")

                processed_count += 1

        except (OSError, ValueError) as exc:
            skipped.append({
                "file": str(src_path),
                "reason": str(exc)
            })

    return {
        "processed_count": processed_count,
        "skipped": skipped
    }
