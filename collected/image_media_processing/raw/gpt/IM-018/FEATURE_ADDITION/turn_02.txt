Here’s the extended version. `distance_metric` defaults to `"l1"` and supports `"chi_squared"` with a safe zero-denominator handling.

```python
import os
from PIL import Image
import numpy as np


def find_near_duplicates(input_dir, threshold, distance_metric="l1"):
    """
    Cluster images by normalized grayscale histogram distance.

    Args:
        input_dir (str): Directory containing images.
        threshold (float): Maximum allowed histogram distance.
        distance_metric (str): Distance metric to use:
            - "l1" (default): L1 histogram distance.
            - "chi_squared": Chi-squared histogram distance.

    Returns:
        list[list[str]]: Transitive clusters of image paths.

    Raises:
        ValueError: If threshold is negative or distance_metric is unsupported.
    """
    if threshold < 0:
        raise ValueError("threshold must be non-negative")

    if distance_metric not in {"l1", "chi_squared"}:
        raise ValueError(
            "distance_metric must be 'l1' or 'chi_squared'"
        )

    extensions = {
        ".jpg", ".jpeg", ".png", ".bmp",
        ".gif", ".tiff", ".webp"
    }

    image_paths = sorted(
        os.path.join(input_dir, name)
        for name in os.listdir(input_dir)
        if os.path.splitext(name)[1].lower() in extensions
    )

    n = len(image_paths)

    # ---------- Union-Find ----------
    parent = list(range(n))
    rank = [0] * n

    def find(x):
        # Path compression
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(a, b):
        root_a = find(a)
        root_b = find(b)

        if root_a == root_b:
            return

        # Union by rank
        if rank[root_a] < rank[root_b]:
            parent[root_a] = root_b
        elif rank[root_a] > rank[root_b]:
            parent[root_b] = root_a
        else:
            parent[root_b] = root_a
            rank[root_a] += 1

    # ---------- Histogram ----------
    def histogram(path):
        with Image.open(path) as img:
            gray = img.convert("L")
            pixels = np.asarray(gray, dtype=np.uint8)

        hist = np.bincount(
            pixels.ravel(),
            minlength=256
        ).astype(np.float64)

        total = hist.sum()
        if total > 0:
            hist /= total

        return hist

    # Compute each histogram only once.
    histograms = [histogram(path) for path in image_paths]

    # ---------- Distance functions ----------
    def l1_distance(h1, h2):
        return np.sum(np.abs(h1 - h2))

    def chi_squared_distance(h1, h2):
        # Standard symmetric chi-squared distance:
        #
        #   0.5 * sum((h1-h2)^2 / (h1+h2))
        #
        # Terms where both histograms are zero contribute zero.
        denominator = h1 + h2
        mask = denominator > 0

        return 0.5 * np.sum(
            ((h1[mask] - h2[mask]) ** 2) /
            denominator[mask]
        )

    if distance_metric == "l1":
        distance_fn = l1_distance
    else:
        distance_fn = chi_squared_distance

    # ---------- Pairwise comparison ----------
    for i in range(n):
        for j in range(i + 1, n):
            distance = distance_fn(histograms[i], histograms[j])

            if distance <= threshold:
                union(i, j)

    # ---------- Build transitive clusters ----------
    clusters = {}

    for i, path in enumerate(image_paths):
        root = find(i)
        clusters.setdefault(root, []).append(path)

    return list(clusters.values())
```



The union-find behavior remains unchanged, so near-duplicate relationships are **transitive** regardless of which distance metric is selected.
