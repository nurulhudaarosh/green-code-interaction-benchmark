Sure. The function can accept `custom_thresholds=(dark_max, bright_min)` while retaining `(85, 170)` as the default.

```python id="58321"
from pathlib import Path
from PIL import Image
import numpy as np


def analyze_brightness(input_dir, custom_thresholds=(85, 170)):
    """
    Analyze brightness statistics for every valid image.

    Parameters:
        input_dir: Directory containing images.
        custom_thresholds: Tuple (dark_max, bright_min) used for
                           brightness classification.

    Returns:
        List of dictionaries containing:
            filename
            r_mean
            g_mean
            b_mean
            brightness
            classification
    """
    input_dir = Path(input_dir)

    dark_max, bright_min = custom_thresholds

    if dark_max < 0 or bright_min > 255 or dark_max >= bright_min:
        raise ValueError(
            "custom_thresholds must satisfy "
            "0 <= dark_max < bright_min <= 255"
        )

    valid_extensions = {
        ".jpg", ".jpeg", ".png", ".bmp",
        ".tif", ".tiff", ".webp"
    }

    results = []

    for path in sorted(input_dir.iterdir()):
        if not path.is_file() or path.suffix.lower() not in valid_extensions:
            continue

        try:
            with Image.open(path) as img:
                # Per-channel RGB means
                rgb = img.convert("RGB")
                rgb_array = np.asarray(rgb, dtype=np.float64)

                r_mean = float(rgb_array[:, :, 0].mean())
                g_mean = float(rgb_array[:, :, 1].mean())
                b_mean = float(rgb_array[:, :, 2].mean())

                # True grayscale brightness using perceptual luminance
                grayscale = img.convert("L")
                grayscale_array = np.asarray(grayscale, dtype=np.float64)
                brightness = float(grayscale_array.mean())

                # Classification using configurable thresholds
                if brightness < dark_max:
                    classification = "dark"
                elif brightness >= bright_min:
                    classification = "bright"
                else:
                    classification = "normal"

                results.append({
                    "filename": path.name,
                    "r_mean": r_mean,
                    "g_mean": g_mean,
                    "b_mean": b_mean,
                    "brightness": brightness,
                    "classification": classification,
                })

        except (OSError, ValueError):
            continue

    return results
```


The thresholds affect **only the classification**; `brightness` is still calculated from the true grayscale (`L`) image.
