Yes. The important part is to calculate the mean over the actual grayscale pixels without integer rounding the overall result. A 50/50 black-and-white image therefore produces approximately `(0 + 255) / 2 = 127.5`, which falls into the default **normal** range.

```python id="74126"
from pathlib import Path
from PIL import Image
import numpy as np


def analyze_brightness(input_dir, custom_thresholds=(85, 170)):
    """
    Analyze brightness statistics for every valid image.

    Parameters:
        input_dir: Directory containing images.
        custom_thresholds: Tuple (dark_max, bright_min) used for
                           brightness classification.

    Returns:
        List of dictionaries containing:
            filename
            r_mean
            g_mean
            b_mean
            brightness
            classification
    """
    input_dir = Path(input_dir)

    # Validate thresholds
    if (
        not isinstance(custom_thresholds, (tuple, list))
        or len(custom_thresholds) != 2
    ):
        raise ValueError(
            "custom_thresholds must be a tuple "
            "(dark_max, bright_min)"
        )

    dark_max, bright_min = custom_thresholds

    if not (0 <= dark_max < bright_min <= 255):
        raise ValueError(
            "custom_thresholds must satisfy "
            "0 <= dark_max < bright_min <= 255"
        )

    valid_extensions = {
        ".jpg", ".jpeg", ".png", ".bmp",
        ".tif", ".tiff", ".webp"
    }

    results = []

    for path in sorted(input_dir.iterdir()):
        if not path.is_file() or path.suffix.lower() not in valid_extensions:
            continue

        try:
            with Image.open(path) as img:
                # RGB channel means
                rgb = img.convert("RGB")
                rgb_array = np.asarray(rgb, dtype=np.float64)

                r_mean = float(rgb_array[:, :, 0].mean())
                g_mean = float(rgb_array[:, :, 1].mean())
                b_mean = float(rgb_array[:, :, 2].mean())

                # True grayscale conversion using PIL's perceptual
                # luminance weighting.
                grayscale = img.convert("L")

                # Use float64 so the overall mean retains fractional
                # values such as 127.5.
                grayscale_array = np.asarray(
                    grayscale, dtype=np.float64
                )

                brightness = float(grayscale_array.mean())

                # Classify using the configurable boundaries.
                if brightness < dark_max:
                    classification = "dark"
                elif brightness >= bright_min:
                    classification = "bright"
                else:
                    classification = "normal"

                results.append({
                    "filename": path.name,
                    "r_mean": r_mean,
                    "g_mean": g_mean,
                    "b_mean": b_mean,
                    "brightness": brightness,
                    "classification": classification,
                })

        except (OSError, ValueError):
            continue

    return results
```
