from pathlib import Path
from PIL import Image
import numpy as np


def analyze_brightness(input_dir):
    """
    Analyze brightness for every valid image in input_dir.

    Returns:
        dict: {
            filename: {
                "channel_means": {
                    "red": float,
                    "green": float,
                    "blue": float
                },
                "grayscale_brightness": float,
                "classification": "dark" | "normal" | "bright"
            }
        }

    Classification thresholds:
        grayscale brightness < 85   -> dark
        85 <= brightness < 170       -> normal
        brightness >= 170            -> bright
    """
    input_dir = Path(input_dir)

    if not input_dir.is_dir():
        raise ValueError(f"Input directory does not exist: {input_dir}")

    results = {}

    valid_extensions = {".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff", ".webp"}

    for image_path in sorted(input_dir.iterdir()):
        if not image_path.is_file() or image_path.suffix.lower() not in valid_extensions:
            continue

        try:
            with Image.open(image_path) as img:
                # Convert to RGB so every image has three channels.
                rgb = np.asarray(img.convert("RGB"), dtype=np.float32)

                # Per-channel means.
                channel_means = {
                    "red": float(rgb[:, :, 0].mean()),
                    "green": float(rgb[:, :, 1].mean()),
                    "blue": float(rgb[:, :, 2].mean()),
                }

                # Convert RGB to grayscale using standard luminance weights.
                grayscale = (
                    0.299 * rgb[:, :, 0]
                    + 0.587 * rgb[:, :, 1]
                    + 0.114 * rgb[:, :, 2]
                )

                grayscale_brightness = float(grayscale.mean())

                if grayscale_brightness < 85:
                    classification = "dark"
                elif grayscale_brightness < 170:
                    classification = "normal"
                else:
                    classification = "bright"

                results[image_path.name] = {
                    "channel_means": channel_means,
                    "grayscale_brightness": grayscale_brightness,
                    "classification": classification,
                }

        except (OSError, ValueError):
            # Skip files that cannot be opened as valid images.
            continue

    return results
