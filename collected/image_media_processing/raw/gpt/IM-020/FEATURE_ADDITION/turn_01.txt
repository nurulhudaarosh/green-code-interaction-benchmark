from pathlib import Path
from PIL import Image


def analyze_brightness(input_dir):
    """
    Analyze brightness statistics for every valid image in input_dir.

    Returns:
        list of dicts, one per image, containing:
        - filename
        - r_mean
        - g_mean
        - b_mean
        - brightness
        - classification
    """
    input_dir = Path(input_dir)
    results = []

    valid_extensions = {".jpg", ".jpeg", ".png", ".bmp", ".tiff", ".webp"}

    for path in sorted(input_dir.iterdir()):
        if not path.is_file() or path.suffix.lower() not in valid_extensions:
            continue

        try:
            with Image.open(path) as img:
                rgb = img.convert("RGB")

                # Per-channel means
                channels = rgb.split()
                r_mean = sum(channels[0].getdata()) / channels[0].size
                g_mean = sum(channels[1].getdata()) / channels[1].size
                b_mean = sum(channels[2].getdata()) / channels[2].size

                # True perceptual grayscale brightness
                grayscale = rgb.convert("L")
                brightness = sum(grayscale.getdata()) / grayscale.size

                # Classification based on grayscale brightness
                if brightness < 85:
                    classification = "dark"
                elif brightness < 170:
                    classification = "normal"
                else:
                    classification = "bright"

                results.append({
                    "filename": path.name,
                    "r_mean": r_mean,
                    "g_mean": g_mean,
                    "b_mean": b_mean,
                    "brightness": brightness,
                    "classification": classification,
                })

        except (OSError, ValueError):
            # Skip files that cannot be opened as valid images
            continue

    return results
