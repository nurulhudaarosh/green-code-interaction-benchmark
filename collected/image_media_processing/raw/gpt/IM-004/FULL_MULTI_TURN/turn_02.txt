```python
from pathlib import Path
from PIL import Image
import numpy as np


def preprocess_images(input_dir, output_dir):
    input_dir = Path(input_dir)
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    stats = {}

    for file in input_dir.iterdir():
        if not file.is_file():
            continue

        try:
            img = Image.open(file).convert("L")
            arr = np.array(img, dtype=np.float32)

            lo = float(arr.min())
            hi = float(arr.max())

            stats[file.name] = {"min": lo, "max": hi}

            # Handle uniformly-colored (flat) images
            if hi == lo:
                normalized = np.full_like(
                    arr, 255 if hi == 255 else 0, dtype=np.uint8
                )
            else:
                # Normal min-max normalization to 0-255
                normalized = ((arr - lo) / (hi - lo) * 255).astype(np.uint8)

            output_path = output_dir / f"{file.stem}.png"
            Image.fromarray(normalized).save(output_path, "PNG")

        except Exception as e:
            print(f"Skipping {file.name}: {e}")

    return stats
```
