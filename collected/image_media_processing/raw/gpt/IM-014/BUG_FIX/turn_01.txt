```python
from pathlib import Path
from PIL import Image


def bulk_reencode(input_dir, output_dir, quality):
    """
    Re-encode every valid image in input_dir as JPEG.

    Args:
        input_dir: Directory containing input images.
        output_dir: Directory where JPEG files will be saved.
        quality: JPEG quality (1-95).

    Returns:
        A list of dictionaries containing:
        filename, original_size, reencoded_size, compression_ratio
    """
    input_dir = Path(input_dir)
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    if not 1 <= quality <= 95:
        raise ValueError("quality must be between 1 and 95")

    results = []

    for path in input_dir.iterdir():
        if not path.is_file():
            continue

        try:
            with Image.open(path) as img:
                # Convert modes such as RGBA/P to JPEG-compatible RGB
                if img.mode != "RGB":
                    img = img.convert("RGB")

                output_path = output_dir / f"{path.stem}.jpg"

                img.save(output_path, "JPEG", quality=quality)

            original_size = path.stat().st_size
            reencoded_size = output_path.stat().st_size

            compression_ratio = (
                original_size / reencoded_size
                if reencoded_size > 0
                else float("inf")
            )

            result = {
                "filename": path.name,
                "original_size": original_size,
                "reencoded_size": reencoded_size,
                "compression_ratio": compression_ratio,
            }

            results.append(result)

            print(
                f"{path.name}: "
                f"{original_size} -> {reencoded_size} bytes, "
                f"ratio={compression_ratio:.2f}:1"
            )

        except (OSError, ValueError):
            # Skip files that are not valid images
            continue

    return results
```
