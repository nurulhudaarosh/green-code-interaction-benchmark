The implementation should explicitly return before entering the pairwise-comparison loop when fewer than two valid images remain.

```python
from pathlib import Path
from PIL import Image


def _average_hash(image_path, hash_size=8):
    """Return the average hash as an integer."""
    with Image.open(image_path) as img:
        gray = img.convert("L").resize(
            (hash_size, hash_size),
            Image.Resampling.LANCZOS
        )

        pixels = list(gray.getdata())
        avg = sum(pixels) / len(pixels)

        hash_value = 0
        for pixel in pixels:
            hash_value = (hash_value << 1) | (pixel >= avg)

        return hash_value


def _hamming_distance(hash1, hash2):
    """Return the Hamming distance between two integer hashes."""
    return (hash1 ^ hash2).bit_count()


def find_fuzzy_duplicates(input_dir, hash_size=8, max_hamming=5):
    """
    Cluster images by average-hash Hamming distance using transitive
    union-find grouping.

    Returns:
        List of clusters, with each cluster containing at least 2
        filenames.
    """
    if not isinstance(hash_size, int) or hash_size <= 0:
        raise ValueError("hash_size must be a positive integer")

    max_distance = hash_size * hash_size
    if not isinstance(max_hamming, int) or not 0 <= max_hamming <= max_distance:
        raise ValueError(
            f"max_hamming must be between 0 and {max_distance}"
        )

    input_dir = Path(input_dir)
    if not input_dir.is_dir():
        raise ValueError(f"Not a directory: {input_dir}")

    image_extensions = {
        ".jpg", ".jpeg", ".png", ".bmp",
        ".gif", ".tif", ".tiff", ".webp"
    }

    image_paths = sorted(
        path for path in input_dir.iterdir()
        if path.is_file() and path.suffix.lower() in image_extensions
    )

    # Compute hashes only for valid/readable images.
    hashed_images = []

    for path in image_paths:
        try:
            hash_value = _average_hash(path, hash_size)
            hashed_images.append((path.name, hash_value))
        except (OSError, ValueError):
            continue

    # Important edge case:
    # With 0 or 1 valid image there cannot be a duplicate cluster,
    # so return before any pairwise comparison.
    if len(hashed_images) < 2:
        return []

    n = len(hashed_images)

    # Union-Find / Disjoint Set Union.
    parent = list(range(n))
    rank = [0] * n

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(a, b):
        root_a = find(a)
        root_b = find(b)

        if root_a == root_b:
            return

        if rank[root_a] < rank[root_b]:
            root_a, root_b = root_b, root_a

        parent[root_b] = root_a

        if rank[root_a] == rank[root_b]:
            rank[root_a] += 1

    # Pairwise comparison is reached only when there are >= 2 images.
    for i in range(n):
        for j in range(i + 1, n):
            if _hamming_distance(
                hashed_images[i][1],
                hashed_images[j][1]
            ) <= max_hamming:
                union(i, j)

    # Build transitive clusters.
    clusters = {}

    for i, (filename, _) in enumerate(hashed_images):
        root = find(i)
        clusters.setdefault(root, []).append(filename)

    return [
        cluster
        for cluster in clusters.values()
        if len(cluster) >= 2
    ]
```


This handles both **zero valid images** and **exactly one valid image**, with no pairwise comparison attempted.
