The fix is to calculate the scale using **both dimensions** with `min()`. This guarantees the resized image fits completely inside the target canvas before padding.

```python
from pathlib import Path
from PIL import Image


def letterbox_resize(input_dir, output_dir, target_size, pad_color=(0, 0, 0)):
    """
    Resize images while preserving aspect ratio and pad them to exactly
    target_size, with the resized content centered.

    Args:
        input_dir: Directory containing input images.
        output_dir: Directory for processed images.
        target_size: Tuple (target_width, target_height).
        pad_color: RGB tuple used for padding.
    """
    input_dir = Path(input_dir)
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    target_w, target_h = target_size

    for file_path in input_dir.iterdir():
        if not file_path.is_file():
            continue

        try:
            with Image.open(file_path) as img:
                img = img.convert("RGB")

                src_w, src_h = img.size

                # Scale using BOTH dimensions so the entire
                # image fits within target_size.
                scale = min(
                    target_w / src_w,
                    target_h / src_h
                )

                new_w = max(1, round(src_w * scale))
                new_h = max(1, round(src_h * scale))

                resized = img.resize(
                    (new_w, new_h),
                    Image.Resampling.LANCZOS
                )

                # Create the final canvas.
                canvas = Image.new(
                    "RGB",
                    (target_w, target_h),
                    pad_color
                )

                # Center the resized image.
                x = (target_w - new_w) // 2
                y = (target_h - new_h) // 2

                canvas.paste(resized, (x, y))

                canvas.save(output_dir / file_path.name)

        except (OSError, ValueError):
            # Skip invalid or unreadable image files.
            continue
```
