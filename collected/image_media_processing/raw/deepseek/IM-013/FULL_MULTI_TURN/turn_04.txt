Here's the updated function. When `input_dir` contains zero valid images, it no longer raises — it writes a minimal valid output image and returns `(0, columns)` as `(num_images, columns)`.

```python
import os
from PIL import Image, ImageDraw, ImageFont
from pathlib import Path

def build_contact_sheet(input_dir, output_path, thumb_size=(200, 200),
                        columns=4, label=True, label_font_size=12,
                        label_max_chars=None):
    """
    Create a contact sheet by tiling all images from input_dir into a grid.

    Args:
        input_dir (str): Directory containing images.
        output_path (str): Path where the contact sheet will be saved.
        thumb_size (tuple): (width, height) of each thumbnail's image area.
        columns (int): Number of columns in the grid.
        label (bool): If True, draw each image's filename as a caption
            beneath its thumbnail. A caption strip is reserved in each cell.
        label_font_size (int): Point size for the caption font.
        label_max_chars (int|None): Truncate labels longer than this.
            If None, derive a limit from the thumbnail width.

    Returns:
        tuple: (num_images, columns). When the input directory contains no
            valid images, returns (0, columns) and still writes a minimal
            valid output image (no exception raised).
    """
    if columns < 1:
        raise ValueError("columns must be >= 1")

    image_extensions = {'.jpg', '.jpeg', '.png', '.gif', '.bmp', '.tiff', '.webp'}

    input_path = Path(input_dir)
    if not input_path.exists():
        raise FileNotFoundError(f"Input directory '{input_dir}' does not exist")

    image_files = [
        f for f in sorted(input_path.iterdir())
        if f.is_file() and f.suffix.lower() in image_extensions
    ]

    num_images = len(image_files)
    padding = 10
    thumb_width, thumb_height = thumb_size

    # --- Font / caption strip setup (skip entirely if no images) ---
    font = None
    caption_height = 0

    if num_images > 0 and label:
        try:
            font = ImageFont.truetype("arial.ttf", label_font_size)
        except (OSError, IOError):
            font = ImageFont.load_default()

        try:
            ascent, descent = font.getmetrics()
            text_h = ascent + descent
        except AttributeError:
            bbox = font.getbbox("Ag")
            text_h = bbox[3] - bbox[1] if bbox else label_font_size

        caption_height = text_h + 6

    cell_height = thumb_height + caption_height

    # --- Compute canvas size ---
    if num_images == 0:
        # Minimal valid output: a 1-pixel-tall (or 1x1) RGB image.
        # Still honor `columns` in the return value; the canvas itself is
        # kept tiny so we don't fabricate a huge empty grid.
        contact_sheet = Image.new('RGB', (1, 1), 'white')
        contact_sheet.save(output_path)
        print(f"Contact sheet saved to '{output_path}' "
              f"(0 images, no grid drawn)")
        return (0, columns)

    # Ceiling division: include any trailing partial row.
    rows = (num_images + columns - 1) // columns

    canvas_width = (columns * thumb_width) + ((columns + 1) * padding)
    canvas_height = (rows * cell_height) + ((rows + 1) * padding)

    contact_sheet = Image.new('RGB', (canvas_width, canvas_height), 'white')
    draw = ImageDraw.Draw(contact_sheet)

    for idx, image_path in enumerate(image_files):
        row = idx // columns
        col = idx % columns

        cell_x = padding + col * (thumb_width + padding)
        cell_y = padding + row * (cell_height + padding)
        thumb_area_x = cell_x
        thumb_area_y = cell_y

        try:
            with Image.open(image_path) as img:
                if img.mode not in ('RGB', 'L'):
                    img = img.convert('RGB')

                img.thumbnail(thumb_size, Image.Resampling.LANCZOS)

                thumb_w, thumb_h = img.size
                offset_x = thumb_area_x + (thumb_width - thumb_w) // 2
                offset_y = thumb_area_y + (thumb_height - thumb_h) // 2

                if img.mode == 'L':
                    img = img.convert('RGB')
                contact_sheet.paste(img, (offset_x, offset_y))

        except Exception as e:
            print(f"Warning: Could not process '{image_path}': {e}")
            draw.rectangle(
                [thumb_area_x, thumb_area_y,
                 thumb_area_x + thumb_width, thumb_area_y + thumb_height],
                outline='red', width=2
            )
            draw.text(
                (thumb_area_x + 10, thumb_area_y + thumb_height // 2),
                "Error", fill='red', font=font
            )

        if label and caption_height > 0:
            text = image_path.name

            limit = label_max_chars
            if limit is None:
                limit = max(1, thumb_width // max(1, label_font_size // 2))
            if limit is not None and len(text) > limit:
                text = text[: max(1, limit - 3)] + "..."

            while text and _text_width(draw, text, font) > thumb_width:
                core = text[:-3] if text.endswith("...") else text
                if len(core) <= 1:
                    text = core
                    break
                text = core[:-1] + "..."

            text_w = _text_width(draw, text, font)
            text_x = cell_x + (thumb_width - text_w) // 2
            text_y = cell_y + thumb_height + 3

            draw.text((text_x, text_y), text, fill='black', font=font)

    contact_sheet.save(output_path, quality=95)
    print(f"Contact sheet saved to '{output_path}' "
          f"({num_images} images, {columns}x{rows} grid, "
          f"labels={'on' if label else 'off'})")
    return (num_images, columns)


def _text_width(draw, text, font):
    """Return the pixel width of `text` rendered with `font`."""
    if not text:
        return 0
    try:
        bbox = draw.textbbox((0, 0), text, font=font)
        return bbox[2] - bbox[0]
    except AttributeError:
        return draw.textsize(text, font=font)[0]
```

## What changed for the empty-directory edge case

**1. No more exception on empty input.**  
The old `raise ValueError(f"No image files found ...")` is removed. Instead, an empty directory is treated as a valid — if degenerate — input.

**2. Early-return path writes a minimal valid image.**  
Before any grid math or font loading runs, the function checks `num_images == 0` and:


This gives you a 1×1 white PNG/JPEG that any image tool can open — no zero-size canvas, no `PIL` errors from `save()` on a 0×0 image (which some formats reject).

**3. Return value is now a tuple for all paths.**  
The function now always returns `(num_images, columns)`. For non-empty input that's `(n, columns)`; for empty input it's `(0, columns)`, matching your spec exactly.

**4. Font setup is skipped when there are no images.**  
`if num_images > 0 and label:` guards the font-loading block, so an empty directory never touches `arial.ttf` or the font fallback — avoiding a pointless filesystem hit and any platform-specific font errors.

**5. The missing-directory case still raises.**  
`FileNotFoundError` is preserved for a *nonexistent* `input_dir`. That's a different condition from "exists but has no images," and silently succeeding there would mask real path bugs. If you'd prefer that case to also produce a minimal image, say so and I'll fold it into the same early-return.

## Why 1×1 rather than a full empty grid

You offered two options ("1-pixel-row or empty"). A 1×1 image is the safest:

- Some encoders (notably certain PNG/JPEG paths) error on 0-width or 0-height buffers.
- A full `columns × 1` grid of blank cells would work but invents a large white canvas that conveys no information and is misleading as "output."
- 1×1 is trivially detectable by callers if they want to distinguish "empty sheet" from "one tiny thumbnail."
