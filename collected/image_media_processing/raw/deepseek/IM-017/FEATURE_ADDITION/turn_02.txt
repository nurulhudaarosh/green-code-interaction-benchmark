# Extended Implementation with `compress_level`

The tricky part: the existing `embed_tags` works at the **byte level** (rewriting chunks manually) and never re-encodes the image data. PIL's `compress_level` only applies when PIL *encodes* the PNG.

To honor `compress_level` **without touching tag values**, we take a hybrid approach:

- **Fast path (`compress_level=None`)**: byte-level chunk surgery exactly as before — preserves the original IDAT bytes, zero image re-encoding.
- **Re-encode path (`compress_level` is 0–9)**: decode the source with PIL, save with the requested `compress_level`, then run the *same* byte-level tag-embedding pass on the re-encoded bytes. Tags go in as raw `iTXt` chunks, so PIL's encoder never sees them and cannot mangle them.

This keeps the two concerns fully separated: `compress_level` controls image data compression, the tag-embedding code is unchanged.

```python
import io
import struct
import zlib
from typing import Optional

from PIL import Image

PNG_SIGNATURE = b"\x89PNG\r\n\x1a\n"


# ---------------------------------------------------------------------------
# Low-level chunk helpers  (unchanged)
# ---------------------------------------------------------------------------

def _make_chunk(chunk_type: bytes, data: bytes) -> bytes:
    length = struct.pack(">I", len(data))
    crc = struct.pack(">I", zlib.crc32(chunk_type + data) & 0xFFFFFFFF)
    return length + chunk_type + data + crc


def _iter_chunks(data: bytes):
    if not data.startswith(PNG_SIGNATURE):
        raise ValueError("Not a valid PNG file (bad signature)")
    pos = len(PNG_SIGNATURE)
    while pos < len(data):
        if pos + 8 > len(data):
            raise ValueError("Truncated PNG chunk header")
        length = struct.unpack(">I", data[pos:pos + 4])[0]
        chunk_type = data[pos + 4:pos + 8]
        chunk_data = data[pos + 8:pos + 8 + length]
        raw = data[pos:pos + 8 + length + 4]
        yield chunk_type, chunk_data, raw
        pos += 8 + length + 4


def _build_itxt(keyword: str, value: str) -> bytes:
    kw = keyword.encode("latin-1")
    if not 1 <= len(kw) <= 79:
        raise ValueError(f"iTXt keyword must be 1-79 Latin-1 bytes: {keyword!r}")
    if b"\x00" in kw:
        raise ValueError("Keyword cannot contain a null byte")
    return (
        kw + b"\x00"
        + b"\x00"          # compression flag: uncompressed
        + b"\x00"          # compression method: 0
        + b"" + b"\x00"    # language tag
        + b"" + b"\x00"    # translated keyword
        + value.encode("utf-8")
    )


def _parse_itxt(data: bytes):
    kw_end = data.index(b"\x00")
    keyword = data[:kw_end].decode("latin-1")
    rest = data[kw_end + 1:]
    comp_flag, comp_method = rest[0], rest[1]
    rest = rest[2:]
    lang_end = rest.index(b"\x00")
    rest = rest[lang_end + 1:]
    trans_end = rest.index(b"\x00")
    text_bytes = rest[trans_end + 1:]
    if comp_flag:
        if comp_method != 0:
            raise ValueError(f"Unsupported iTXt compression method: {comp_method}")
        text_bytes = zlib.decompress(text_bytes)
    return keyword, text_bytes.decode("utf-8")


def _parse_text(data: bytes):
    kw_end = data.index(b"\x00")
    return data[:kw_end].decode("latin-1"), data[kw_end + 1:].decode("latin-1")


# ---------------------------------------------------------------------------
# Tag embedding pass (works on raw PNG bytes; no re-encoding)
# ---------------------------------------------------------------------------

def _inject_tags(png_bytes: bytes, tags: dict) -> bytes:
    """Insert/replace iTXt chunks in `png_bytes` and return new PNG bytes."""
    tags = {str(k): str(v) for k, v in tags.items()}
    new_chunks = {k: _make_chunk(b"iTXt", _build_itxt(k, v)) for k, v in tags.items()}

    out = bytearray(PNG_SIGNATURE)
    inserted = False

    for chunk_type, chunk_data, raw in _iter_chunks(png_bytes):
        if chunk_type in (b"tEXt", b"iTXt", b"zTXt"):
            try:
                if chunk_type == b"iTXt":
                    kw, _ = _parse_itxt(chunk_data)
                elif chunk_type == b"tEXt":
                    kw, _ = _parse_text(chunk_data)
                else:  # zTXt
                    kw = chunk_data[:chunk_data.index(b"\x00")].decode("latin-1")
            except (ValueError, IndexError):
                kw = None
            if kw in new_chunks:
                continue

        out.extend(raw)

        if chunk_type == b"IHDR" and not inserted:
            for c in new_chunks.values():
                out.extend(c)
            inserted = True

    if not inserted:
        raise ValueError("PNG has no IHDR chunk; file appears corrupt")
    return bytes(out)


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def embed_tags(
    input_path: str,
    output_path: str,
    tags: dict,
    compress_level: Optional[int] = None,
) -> None:
    """
    Copy the PNG at `input_path` to `output_path`, embedding each key/value
    pair in `tags` as an iTXt text chunk (UTF-8).

    Parameters
    ----------
    input_path, output_path : str
        Source and destination PNG paths.
    tags : dict
        Mapping of str -> str metadata to embed.
    compress_level : int or None, optional
        zlib compression level (0-9) forwarded to PIL's PNG encoder when
        re-encoding image data. 0 = store, 9 = max compression.

        If None (default), the original IDAT bytes are preserved byte-for-byte
        and no re-encoding occurs — the function behaves exactly as before.
        If an int in [0, 9], the image is decoded and re-saved with this
        compression level; tag values are unaffected either way.

    Notes
    -----
    Tag values are written as raw iTXt chunks *after* the PIL save step, so
    they are always UTF-8 encoded regardless of `compress_level`.
    """
    if compress_level is not None:
        if not isinstance(compress_level, int) or isinstance(compress_level, bool):
            raise TypeError("compress_level must be an int in [0, 9] or None")
        if not 0 <= compress_level <= 9:
            raise ValueError("compress_level must be between 0 and 9")

    # ---- Path A: byte-for-byte, no re-encoding (original behavior) ----
    if compress_level is None:
        with open(input_path, "rb") as f:
            source_bytes = f.read()
        result = _inject_tags(source_bytes, tags)
        with open(output_path, "wb") as f:
            f.write(result)
        return

    # ---- Path B: re-encode with the requested compression level ----
    # We open the source, decode pixels, and re-save via PIL with the
    # requested compress_level. PIL's own text handling is bypassed entirely
    # (we pass no `pnginfo`), so tag values are untouched by this step.
    with Image.open(input_path) as im:
        im.load()  # force full decode before the file handle closes

        buf = io.BytesIO()
        # Preserve essential PNG format flags that PIL would otherwise reset.
        save_kwargs = {
            "format": "PNG",
            "compress_level": compress_level,
        }
        # Preserve alpha / palette / bit depth where sensible.
        if im.mode == "P":
            save_kwargs["optimize"] = False  # let palette survive
        im.save(buf, **save_kwargs)

    reencoded = buf.getvalue()

    # Now inject our tags into the freshly-encoded PNG.
    result = _inject_tags(reencoded, tags)

    with open(output_path, "wb") as f:
        f.write(result)


def read_tags(image_path: str) -> dict:
    """Read all tEXt and iTXt metadata chunks from a PNG into a dict."""
    with open(image_path, "rb") as f:
        data = f.read()

    tags = {}
    for chunk_type, chunk_data, _ in _iter_chunks(data):
        if chunk_type == b"iTXt":
            try:
                kw, value = _parse_itxt(chunk_data)
                tags[kw] = value
            except (ValueError, IndexError):
                pass
        elif chunk_type == b"tEXt":
            try:
                kw, value = _parse_text(chunk_data)
                tags[kw] = value
            except (ValueError, IndexError):
                pass
    return tags


# ---------------------------------------------------------------------------
# Demo
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    import os

    tags = {
        "Author": "Ada Lovelace",
        "Copyright": "© 2025 — All rights reserved",
        "Description": "Unicode: 日本語 ✓",
        "Software": "embed_tags.py v1.1",
    }

    # No re-encoding: identical IDAT bytes, tags injected in place.
    embed_tags("in.png", "out_none.png", tags)
    print("none   :", read_tags("out_none.png"))

    # Force maximum compression on the image data.
    embed_tags("in.png", "out_9.png", tags, compress_level=9)
    print("level 9:", read_tags("out_9.png"))

    # Force no compression on the image data (largest file).
    embed_tags("in.png", "out_0.png", tags, compress_level=0)
    print("level 0:", read_tags("out_0.png"))

    for name in ("in.png", "out_none.png", "out_0.png", "out_9.png"):
        print(f"{name:14s} {os.path.getsize(name):>8d} bytes")
```
