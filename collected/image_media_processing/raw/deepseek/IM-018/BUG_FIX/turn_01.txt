Here's a complete implementation of `find_near_duplicates` that clusters images by normalized grayscale histogram L1 distance using transitive union-find grouping:

```python
import os
import numpy as np
from PIL import Image
from pathlib import Path
from collections import defaultdict

# Supported image extensions
IMAGE_EXTS = {'.jpg', '.jpeg', '.png', '.bmp', '.gif', '.tif', '.tiff', '.webp'}


class UnionFind:
    """Disjoint-set/union-find with path compression and union by rank."""

    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, x):
        # Path compression (iterative)
        root = x
        while self.parent[root] != root:
            root = self.parent[root]
        while self.parent[x] != root:
            self.parent[x], x = root, self.parent[x]
        return root

    def union(self, x, y):
        rx, ry = self.find(x), self.find(y)
        if rx == ry:
            return False
        # Union by rank
        if self.rank[rx] < self.rank[ry]:
            rx, ry = ry, rx
        self.parent[ry] = rx
        if self.rank[rx] == self.rank[ry]:
            self.rank[rx] += 1
        return True


def compute_histogram(path, bins=256):
    """
    Load an image, convert to grayscale, and return its normalized histogram.
    Returns a float32 numpy array of length `bins` summing to 1.0, or None on failure.
    """
    try:
        with Image.open(path) as img:
            gray = img.convert('L')
            arr = np.asarray(gray, dtype=np.uint8)
    except Exception as e:
        print(f"  [skip] {path}: {e}")
        return None

    hist, _ = np.histogram(arr, bins=bins, range=(0, 256))
    total = hist.sum()
    if total == 0:
        return None
    return (hist / total).astype(np.float32)


def find_near_duplicates(input_dir, threshold, bins=256):
    """
    Cluster images in `input_dir` by normalized grayscale histogram L1 distance.

    Two images are considered "near duplicates" if the L1 distance between their
    normalized histograms is <= `threshold`. Grouping is transitive: if A~B and
    B~C, then A, B, and C end up in the same cluster (union-find).

    Parameters
    ----------
    input_dir : str | os.PathLike
        Directory to scan recursively for image files.
    threshold : float
        Maximum L1 distance (in [0, 2]) to consider two histograms matching.
        Typical useful range: 0.02 (very strict) to 0.20 (loose).
    bins : int, default 256
        Number of histogram bins. Must match between compared images.

    Returns
    -------
    clusters : list[list[str]]
        Each inner list contains absolute paths of images in one cluster,
        sorted by path. Only clusters with >= 2 members are returned, sorted
        by descending size (then by first path).
    histograms : dict[str, np.ndarray]
        Map from path -> normalized histogram (for inspection/debugging).
    """
    input_dir = Path(input_dir).resolve()
    if not input_dir.is_dir():
        raise NotADirectoryError(f"Not a directory: {input_dir}")

    # --- 1. Collect image paths ---
    paths = []
    for root, _, files in os.walk(input_dir):
        for fname in files:
            if Path(fname).suffix.lower() in IMAGE_EXTS:
                paths.append(str(Path(root) / fname))
    paths.sort()

    if len(paths) < 2:
        return [], {}

    # --- 2. Compute histograms ---
    print(f"Found {len(paths)} candidate images. Computing histograms...")
    histograms = {}
    for p in paths:
        h = compute_histogram(p, bins=bins)
        if h is not None:
            histograms[p] = h

    keys = list(histograms.keys())
    n = len(keys)
    if n < 2:
        return [], histograms

    # --- 3. Pairwise comparison + union-find ---
    # Vectorize: stack histograms into an (n, bins) matrix, then use
    # broadcasting to compute an (n, n) L1 distance matrix in one shot.
    # For very large n, swap this for a chunked loop (see note below).
    H = np.stack([histograms[k] for k in keys])  # (n, bins)
    # L1 distance: sum |h_i - h_j| over bins
    # (n, 1, bins) - (1, n, bins) -> (n, n, bins) then abs+sum over axis 2
    dist = np.abs(H[:, None, :] - H[None, :, :]).sum(axis=2)

    uf = UnionFind(n)
    for i in range(n):
        # Only compare j > i (symmetric matrix); diagonal is 0
        close = np.where(dist[i, i + 1:] <= threshold)[0]
        for offset in close:
            uf.union(i, i + 1 + int(offset))

    # --- 4. Collect clusters ---
    groups = defaultdict(list)
    for i, k in enumerate(keys):
        groups[uf.find(i)].append(k)

    clusters = [sorted(members) for members in groups.values() if len(members) >= 2]
    clusters.sort(key=lambda c: (-len(c), c[0]))

    return clusters, histograms


# ---------- Demo / CLI ----------
if __name__ == "__main__":
    import sys
    import json

    if len(sys.argv) < 3:
        print("Usage: python find_near_duplicates.py <input_dir> <threshold> [bins]")
        sys.exit(1)

    directory = sys.argv[1]
    thr = float(sys.argv[2])
    nbins = int(sys.argv[3]) if len(sys.argv) > 3 else 256

    clusters, _ = find_near_duplicates(directory, thr, bins=nbins)

    print(f"\nFound {len(clusters)} cluster(s) of near-duplicate images "
          f"(threshold={thr}, bins={nbins}):\n")
    for idx, cluster in enumerate(clusters, 1):
        print(f"Cluster #{idx}  ({len(cluster)} images)")
        for p in cluster:
            print(f"    {p}")
        print()
```

### How it works

1. **Discovery** — Walks `input_dir` recursively, filtering by common image extensions.
2. **Feature extraction** — Each image is converted to grayscale and reduced to a `bins`-bin intensity histogram, then normalized to sum to 1 (so brightness/contrast changes shift the shape but not the scale).
3. **Distance** — L1 (`sum |h_i − h_j|`) between normalized histograms. Range is `[0, 2]`; `0` means identical distributions. Computed as a full `(n, n)` matrix via NumPy broadcasting.
4. **Clustering** — **Union-find** performs the transitive grouping: any pair with distance `≤ threshold` triggers a union, so a chain `A–B–C` collapses into one cluster even if `A` and `C` are individually farther than `threshold`. Path compression + union-by-rank keep this near-linear.
5. **Output** — Returns only clusters with ≥ 2 members, sorted by descending size.

### Notes & tuning

- **Threshold guidance**: for 256-bin normalized histograms on typical photos:
  - `0.02–0.05`: near-identical (recompressed copies, minor crops)
  - `0.05–0.15`: visually similar (same scene, different exposure)
  - `> 0.20`: mostly unrelated images start merging
- **Memory**: the `(n, n)` distance matrix is `8·n²` bytes (float64). For >10k images, replace the matrix with a chunked loop:

- **Grayscale-only**: histograms ignore color and spatial layout, so two very different images with similar tonal distributions can match. If you need more discrimination, extend `compute_histogram` to concatenate per-channel or spatial-pyramid histograms.
