import sys

class FenwickTree:
    """Fenwick tree for point updates and prefix/range sum queries."""
    
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)
    
    def build(self, arr):
        """Build tree from 0-based array in O(n)."""
        for i in range(1, self.n + 1):
            self.tree[i] += arr[i - 1]
            parent = i + (i & -i)
            if parent <= self.n:
                self.tree[parent] += self.tree[i]
    
    def update(self, idx, delta):
        """Add delta to position idx (1-based)."""
        while idx <= self.n:
            self.tree[idx] += delta
            idx += idx & -idx
    
    def prefix_sum(self, idx):
        """Sum of [1..idx] (1-based)."""
        result = 0
        while idx > 0:
            result += self.tree[idx]
            idx -= idx & -idx
        return result
    
    def range_sum(self, left, right):
        """Sum of [left..right] inclusive (1-based)."""
        if left > right:
            left, right = right, left
        return self.prefix_sum(right) - self.prefix_sum(left - 1)


def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    
    it = iter(data)
    n = int(next(it))
    q = int(next(it))
    
    arr = [int(next(it)) for _ in range(n)]
    
    ft = FenwickTree(n)
    ft.build(arr)
    
    output = []
    
    for _ in range(q):
        typ = int(next(it))
        if typ == 1:
            pos = int(next(it))
            val = int(next(it))
            ft.update(pos, val)
        else:
            left = int(next(it))
            right = int(next(it))
            output.append(str(ft.range_sum(left, right)))
    
    sys.stdout.write("\n".join(output))


if __name__ == "__main__":
    solve()
