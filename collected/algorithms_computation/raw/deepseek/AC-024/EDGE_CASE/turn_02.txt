import heapq


def assign_rooms(intervals):
    """
    Minimum Room Assignment.

    Given intervals [start, end) (half-open), assign each interval to a room so
    that overlapping intervals never share a room. A room is reusable when its
    previous interval ends at or before the next interval's start
    (prev_end <= next_start). Minimize the number of rooms and return the room
    ID for every original interval (input order preserved).

    Determinism / tie-breaking:
      - Sort by (start, original_index).
      - Reuse the smallest available (freed) room ID first.
      - New room IDs allocated in order rooms are first opened (0-indexed).

    Complexity:
      - Time:  O(n log n)
      - Space: O(n)
    """
    n = len(intervals)
    if n == 0:
        return []

    order = sorted(range(n), key=lambda i: (intervals[i][0], i))
    room_ids = [None] * n

    active = []       # min-heap of (end_time, room_id) for occupied rooms
    available = []    # min-heap of freed, reusable room IDs
    next_room_id = 0

    for idx in order:
        start, end = intervals[idx]

        # Free rooms whose intervals ended at or before this start.
        while active and active[0][0] <= start:
            _, freed_room = heapq.heappop(active)
            heapq.heappush(available, freed_room)

        # Reuse smallest available room, else open a new one.
        if available:
            room = heapq.heappop(available)
        else:
            room = next_room_id
            next_room_id += 1

        room_ids[idx] = room
        heapq.heappush(active, (end, room))

    return room_ids


# ---------------------------------------------------------------------------
# Tests: original examples + difficult cases
# ---------------------------------------------------------------------------

def _assert(condition, msg):
    if not condition:
        raise AssertionError(msg)


def run_tests():
    # Original examples (preserved outputs)
    _assert(assign_rooms([[0, 30], [5, 10], [15, 20]]) == [0, 1, 2],
            "Example 1")
    _assert(assign_rooms([[1, 2], [2, 3], [3, 4]]) == [0, 0, 0],
            "Example 3: touching reuse")
    _assert(assign_rooms([]) == [], "Example 4: empty")
    _assert(assign_rooms([[3, 7]]) == [0], "Single interval")

    # Case A: all mutually overlap -> n distinct rooms
    n = 200
    intervals = [[i, i + 100] for i in range(n)]
    rooms = assign_rooms(intervals)
    _assert(len(set(rooms)) == n, "Case A: expected n distinct rooms")
    _assert(sorted(rooms) == list(range(n)), "Case A: rooms must be 0..n-1")

    # Case B: chained touching -> all room 0
    n = 500
    intervals = [[i, i + 1] for i in range(n)]
    _assert(assign_rooms(intervals) == [0] * n, "Case B: all reuse room 0")

    # Case C: reverse-sorted touching -> all room 0
    n = 500
    intervals = [[n - 1 - i, n - i] for i in range(n)]
    _assert(assign_rooms(intervals) == [0] * n,
            "Case C: reverse-sorted chain reuse")

    # Case D: identical starts -> rooms in input order
    n = 300
    intervals = [[5, 5 + i + 1] for i in range(n)]
    _assert(assign_rooms(intervals) == list(range(n)),
            "Case D: identical starts -> 0..n-1")

    # Case E: interleaved long/short -> valid, no overlap shares a room
    longs = [[i, i + 2] for i in range(0, 200, 2)]
    shorts = [[i, i + 1] for i in range(0, 200, 2)]
    intervals = longs + shorts
    rooms = assign_rooms(intervals)
    _assert(len(rooms) == len(intervals), "Case E: length mismatch")
    for i in range(len(intervals)):
        for j in range(i + 1, len(intervals)):
            si, ei = intervals[i]
            sj, ej = intervals[j]
            if si < ej and sj < ei:  # half-open overlap
                _assert(rooms[i] != rooms[j],
                        f"Case E: overlap {i},{j} shares room {rooms[i]}")

    # Case F: zero-length intervals -> all room 0
    _assert(assign_rooms([[0, 0], [0, 0], [1, 1], [2, 2]]) == [0, 0, 0, 0],
            "Case F: zero-length intervals reuse room 0")

    print("All tests passed.")


if __name__ == "__main__":
    print(assign_rooms([[0, 30], [5, 10], [15, 20]]))         # [0, 1, 2]
    print(assign_rooms([[7, 10], [2, 4], [1, 3], [5, 8]]))    # deterministic
    print(assign_rooms([[1, 2], [2, 3], [3, 4]]))             # [0, 0, 0]
    print(assign_rooms([]))                                   # []
    run_tests()
