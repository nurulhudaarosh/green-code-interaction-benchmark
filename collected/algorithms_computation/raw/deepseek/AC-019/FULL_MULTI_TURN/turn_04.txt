"""
Minimum-Cost Grid Path
======================
Given a non-negative m x n cost matrix, move only right or down from (0,0)
to (m-1,n-1). Return the minimum cost and one deterministic optimal path.

Original required outputs:
    1. min_cost : int
    2. path     : list[(row, col)] from start to end
Optional (when requested):
    3. operation_summary : dict with deterministic counts of the algorithm's
       major computational decisions.

Preserved requirements:
    - Right (i, j+1) and down (i+1, j) moves only.
    - DP with predecessor coordinates.
    - Deterministic tie handling: on a tie (dp[i-1][j] == dp[i][j-1]),
      prefer predecessor (i-1, j) -- i.e. the move INTO (i, j) is a DOWN move.
    - Standard library only; no network, APIs, external services,
      randomness, or human interaction.
    - Default call returns exactly (min_cost, path); summary is opt-in.
    - Non-negative costs, rectangular matrix, non-empty matrix.

Boundary cases explicitly handled and tested:
    - 1x1 matrix.
    - Single row (1 x n): only right moves.
    - Single column (m x 1): only down moves.
    - All-zero matrix: every interior cell is a tie.
    - All-equal matrix: deterministic path = right along top row, then down.
    - Very large integer costs (Python ints are unbounded).
    - Large dimensions (practical max).
"""

from typing import List, Tuple, Optional, Dict, Any, Union


def min_cost_path(
    cost: List[List[int]],
    return_operation_summary: bool = False,
) -> Union[
    Tuple[int, List[Tuple[int, int]]],
    Tuple[int, List[Tuple[int, int]], Dict[str, Any]],
]:
    """
    Minimum-cost path from top-left to bottom-right, moving only right or down.

    Tie rule (deterministic): when dp[i-1][j] == dp[i][j-1], prefer the
    predecessor (i-1, j), i.e. the move *into* (i, j) is a DOWN move.

    Returns:
        (min_cost, path)                                   by default
        (min_cost, path, operation_summary)                if requested

    operation_summary keys (deterministic):
        decisions      : interior cells where two predecessors were compared
        tie_decisions  : those comparisons that were ties
        forced_cells   : cells filled from a single predecessor
                         (base cell + first row + first column)
    """
    # ---- Validation -------------------------------------------------------
    if not cost or not cost[0]:
        raise ValueError("Cost matrix must be non-empty")
    m, n = len(cost), len(cost[0])
    if any(len(row) != n for row in cost):
        raise ValueError("Cost matrix must be rectangular")
    if any(c < 0 for row in cost for c in row):
        raise ValueError("Costs must be non-negative")

    # ---- DP tables --------------------------------------------------------
    dp = [[0] * n for _ in range(m)]
    pred: List[List[Optional[Tuple[int, int]]]] = [[None] * n for _ in range(m)]

    decisions = 0
    tie_decisions = 0
    forced_cells = 0

    # Base cell: forced, no decision.
    dp[0][0] = cost[0][0]
    forced_cells += 1

    # First row: forced (only left predecessor).
    for j in range(1, n):
        dp[0][j] = dp[0][j - 1] + cost[0][j]
        pred[0][j] = (0, j - 1)
        forced_cells += 1

    # First column: forced (only top predecessor).
    for i in range(1, m):
        dp[i][0] = dp[i - 1][0] + cost[i][0]
        pred[i][0] = (i - 1, 0)
        forced_cells += 1

    # Interior cells: each is one major decision between two predecessors.
    for i in range(1, m):
        for j in range(1, n):
            from_top = dp[i - 1][j]
            from_left = dp[i][j - 1]
            decisions += 1
            if from_top <= from_left:            # tie -> prefer DOWN into (i, j)
                if from_top == from_left:
                    tie_decisions += 1
                dp[i][j] = from_top + cost[i][j]
                pred[i][j] = (i - 1, j)
            else:
                dp[i][j] = from_left + cost[i][j]
                pred[i][j] = (i, j - 1)

    # ---- Path reconstruction (backwards via predecessors) -----------------
    path: List[Tuple[int, int]] = []
    cur: Optional[Tuple[int, int]] = (m - 1, n - 1)
    while cur is not None:
        path.append(cur)
        cur = pred[cur[0]][cur[1]]
    path.reverse()

    # ---- Output -----------------------------------------------------------
    if not return_operation_summary:
        return dp[m - 1][n - 1], path

    return dp[m - 1][n - 1], path, {
        "decisions": decisions,
        "tie_decisions": tie_decisions,
        "forced_cells": forced_cells,
    }


# =========================================================================
# Tests
# =========================================================================

def _assert_path_valid(path, m, n):
    assert path[0] == (0, 0), path
    assert path[-1] == (m - 1, n - 1), path
    for (r1, c1), (r2, c2) in zip(path, path[1:]):
        assert (r2 - r1, c2 - c1) in {(0, 1), (1, 0)}, (r1, c1, r2, c2)
    assert len(path) == m + n - 1, (len(path), m + n - 1)


def test_original_behavior_preserved():
    # Default call: exactly a 2-tuple (min_cost, path).
    result = min_cost_path([[0, 1], [1, 0]])
    assert isinstance(result, tuple) and len(result) == 2
    c, p = result
    # Tie rule -> prefer DOWN into (1,1) -> path through (0,1).
    assert c == 1 and p == [(0, 0), (0, 1), (1, 1)], (c, p)
    _assert_path_valid(p, 2, 2)

    c, p = min_cost_path([[1, 3, 1], [1, 5, 1], [4, 2, 1]])
    assert c == 7, c
    _assert_path_valid(p, 3, 3)
    print("original behavior preserved: OK")


def test_boundary_cases():
    # 1) Minimum size: single cell.
    c, p, s = min_cost_path([[0]], return_operation_summary=True)
    assert c == 0 and p == [(0, 0)]
    assert s == {"decisions": 0, "tie_decisions": 0, "forced_cells": 1}
    _assert_path_valid(p, 1, 1)

    c, p, s = min_cost_path([[7]], return_operation_summary=True)
    assert c == 7 and p == [(0, 0)]

    # 2) Single row: only right moves.
    c, p, s = min_cost_path([[1, 2, 3, 4]], return_operation_summary=True)
    assert c == 10 and p == [(0, 0), (0, 1), (0, 2), (0, 3)]
    assert s == {"decisions": 0, "tie_decisions": 0, "forced_cells": 4}
    _assert_path_valid(p, 1, 4)

    # 3) Single column: only down moves.
    c, p, s = min_cost_path([[1], [2], [3], [4]], return_operation_summary=True)
    assert c == 10 and p == [(0, 0), (1, 0), (2, 0), (3, 0)]
    assert s == {"decisions": 0, "tie_decisions": 0, "forced_cells": 4}
    _assert_path_valid(p, 4, 1)

    # 4) Minimum cost value: all zeros (maximum ties).
    m, n = 3, 4
    z = [[0] * n for _ in range(m)]
    c, p, s = min_cost_path(z, return_operation_summary=True)
    assert c == 0
    expected = [(0, j) for j in range(n)] + [(i, n - 1) for i in range(1, m)]
    assert p == expected, (p, expected)
    assert s["decisions"] == (m - 1) * (n - 1)
    assert s["tie_decisions"] == (m - 1) * (n - 1)
    assert s["forced_cells"] == m + n - 1
    _assert_path_valid(p, m, n)

    # 5) Maximum cost value: very large integers (no overflow).
    BIG = 10 ** 100
    c, p, s = min_cost_path([[BIG, BIG], [BIG, BIG]], return_operation_summary=True)
    assert c == 3 * BIG, c
    assert p == [(0, 0), (0, 1), (1, 1)], p
    assert s == {"decisions": 1, "tie_decisions": 1, "forced_cells": 3}
    _assert_path_valid(p, 2, 2)

    # 6) Large dimensions (practical max): correctness + path length.
    m, n = 500, 400
    big = [[1] * n for _ in range(m)]
    c, p, s = min_cost_path(big, return_operation_summary=True)
    assert c == (m + n - 1) * 1, c
    assert len(p) == m + n - 1
    assert s["decisions"] == (m - 1) * (n - 1)
    assert s["tie_decisions"] == (m - 1) * (n - 1)
    _assert_path_valid(p, m, n)

    # 7) All-equal non-square matrix -> right along top, then down.
    m, n = 4, 6
    eq = [[5] * n for _ in range(m)]
    c, p, _ = min_cost_path(eq)
    assert c == (m + n - 1) * 5
    expected = [(0, j) for j in range(n)] + [(i, n - 1) for i in range(1, m)]
    assert p == expected, (p, expected)
    _assert_path_valid(p, m, n)

    # 8) Non-tie case where the rule must NOT force the all-equal path.
    #    Grid favors the bottom row for a cheaper route.
    g = [
        [0, 9, 9, 9],
        [0, 1, 1, 1],
        [9, 9, 9, 0],
    ]
    c, p, s = min_cost_path(g, return_operation_summary=True)
    # Optimal: down, right,right,right, down -> cost 0+0+1+1+1+0 = 3
    assert c == 3, c
    assert p[0] == (0, 0) and p[-1] == (2, 3)
    _assert_path_valid(p, 3, 4)

    # 9) Validation errors.
    for bad in ([], [[]], [[1, 2], [3]], [[-1, 0], [0, 0]]):
        try:
            min_cost_path(bad)
        except ValueError:
            pass
        else:
            raise AssertionError(f"expected ValueError for {bad!r}")

    print("all boundary tests passed: OK")


if __name__ == "__main__":
    test_original_behavior_preserved()
    test_boundary_cases()
    print("ALL TESTS PASSED")
