```python
#!/usr/bin/env python3
"""
Maximum Simultaneous Meetings

Given half-open meeting intervals [start, end), find:
- The maximum number of simultaneously active meetings.
- The sorted list of start times at which that maximum begins.

Key constraints:
- Intervals are half-open: [start, end)
- End events at the same timestamp occur before start events.
- Times can be any comparable values (integers, floats, strings, etc.)
- Empty list of meetings: maximum = 0, times = []

Algorithm: Sweep-line / event-based approach.
- For each meeting [s, e), create:
    * Start event: (s, +1)
    * End event: (e, -1)   (processed before starts at same time)
- Sort events by time. For equal times, end events (-1) come before start events (+1).
- Sweep through events:
    * Apply delta to current count.
    * After processing all events at a given time, check if current count > max.
    * If greater, update max and record the time.
    * If equal, append the time (if it's a new time).
- Return (max_count, sorted_times)
"""

from typing import List, Tuple, Union


def max_simultaneous_meetings(
    meetings: List[Tuple[Union[int, float], Union[int, float]]]
) -> Tuple[int, List[Union[int, float]]]:
    """
    Find maximum number of simultaneous meetings and the start times
    when that maximum begins.

    Args:
        meetings: List of (start, end) half-open intervals.

    Returns:
        (max_count, sorted_times) where times are sorted ascending.
    """
    if not meetings:
        return 0, []

    # Build events: (time, delta)
    # We use delta = +1 for start, -1 for end.
    # For sorting, we want end events (-1) before start events (+1) at same time.
    events = []
    for start, end in meetings:
        events.append((start, 1))   # start
        events.append((end, -1))    # end

    # Sort by (time, -delta) so that -1 (end) comes before +1 (start)
    # Since we want end before start, we sort by delta descending at same time.
    events.sort(key=lambda x: (x[0], -x[1]))

    max_count = 0
    current = 0
    max_start_times = []
    last_recorded_time = None

    i = 0
    n = len(events)

    while i < n:
        time = events[i][0]
        # Process all events at this time
        while i < n and events[i][0] == time:
            current += events[i][1]  # delta is +1 or -1
            i += 1

        # After processing all events at this time, check current count
        if current > max_count:
            max_count = current
            max_start_times = [time]
            last_recorded_time = time
        elif current == max_count:
            # Only add if this time is different from the last recorded time
            # (safety: ensure we don't duplicate if same time appears twice)
            if last_recorded_time != time:
                max_start_times.append(time)
                last_recorded_time = time

    return max_count, max_start_times


def main():
    """Example usage and test cases."""
    # Test case 1: Basic overlapping
    meetings1 = [(1, 5), (2, 6), (3, 7)]
    count1, times1 = max_simultaneous_meetings(meetings1)
    print(f"Meetings: {meetings1}")
    print(f"Max simultaneous: {count1}, at times: {times1}")
    # Expected: 3, [3] (since at time 3 all three overlap)

    # Test case 2: End before start at same time
    meetings2 = [(1, 3), (3, 5)]
    count2, times2 = max_simultaneous_meetings(meetings2)
    print(f"\nMeetings: {meetings2}")
    print(f"Max simultaneous: {count2}, at times: {times2}")
    # Expected: 1, [1, 3] (meeting1 ends at 3, meeting2 starts at 3, never overlap)

    # Test case 3: Equal max at multiple times
    meetings3 = [(1, 4), (2, 5), (3, 6)]
    count3, times3 = max_simultaneous_meetings(meetings3)
    print(f"\nMeetings: {meetings3}")
    print(f"Max simultaneous: {count3}, at times: {times3}")
    # Expected: 3, [3] (only at time 3 all three overlap)

    # Test case 4: Multiple peaks
    meetings4 = [(1, 3), (2, 4), (3, 5), (4, 6)]
    count4, times4 = max_simultaneous_meetings(meetings4)
    print(f"\nMeetings: {meetings4}")
    print(f"Max simultaneous: {count4}, at times: {times4}")
    # Expected: 2, [2, 3, 4] (at times 2,3,4 two meetings are active)

    # Test case 5: Empty list
    count5, times5 = max_simultaneous_meetings([])
    print(f"\nMeetings: []")
    print(f"Max simultaneous: {count5}, at times: {times5}")
    # Expected: 0, []

    # Test case 6: Single meeting
    meetings6 = [(5, 10)]
    count6, times6 = max_simultaneous_meetings(meetings6)
    print(f"\nMeetings: {meetings6}")
    print(f"Max simultaneous: {count6}, at times: {times6}")
    # Expected: 1, [5]

    # Test case 7: Floating point times
    meetings7 = [(1.5, 3.5), (2.0, 4.0), (3.5, 5.0)]
    count7, times7 = max_simultaneous_meetings(meetings7)
    print(f"\nMeetings: {meetings7}")
    print(f"Max simultaneous: {count7}, at times: {times7}")
    # Expected: 2, [2.0, 3.5] (2.0-3.5: meetings1&2; 3.5: meetings1 ends, meeting2&3 active)


if __name__ == "__main__":
    main()
```
