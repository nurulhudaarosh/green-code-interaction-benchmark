```python
"""
Shortest Path from Single Source in Directed Graph with Non-negative Weights
---------------------------------------------------------------------------
Problem:
- Given a directed graph with n vertices (0 to n-1) and m edges.
- Each edge has a non-negative weight.
- Source vertex: s.
- Compute shortest distance from s to every vertex.
- If a vertex is unreachable, distance = -1.

Constraints (typical):
- 1 <= n <= 10^5
- 0 <= m <= 2 * 10^5
- 0 <= weight <= 10^9
- Graph may be disconnected.
- No negative cycles (weights are non-negative, so none).

Output:
- List/array of length n where dist[i] is shortest distance from s to i, or -1 if unreachable.

Algorithm:
- Use Dijkstra's algorithm with adjacency list representation.
- Priority queue (binary heap) implemented via heapq.
- Time complexity: O((n + m) log n).
- Space complexity: O(n + m).

Deterministic, no randomness, no external dependencies.
"""

import sys
import heapq
from typing import List, Tuple

def dijkstra(
    n: int,
    adj: List[List[Tuple[int, int]]],
    source: int
) -> List[int]:
    """
    Dijkstra's algorithm using binary heap (priority queue).

    Args:
        n: Number of vertices (0..n-1).
        adj: Adjacency list where adj[u] = [(v, weight), ...].
        source: Source vertex index.

    Returns:
        List of shortest distances from source to each vertex.
        Unreachable vertices have distance -1.
    """
    # Initialize distances with infinity
    INF = 10**18  # large enough (weights up to 1e9, n up to 1e5 => max path 1e14)
    dist = [INF] * n
    dist[source] = 0

    # Priority queue: (current_distance, vertex)
    pq = [(0, source)]

    while pq:
        d, u = heapq.heappop(pq)

        # If this distance is outdated, skip it
        if d != dist[u]:
            continue

        # Relax edges from u
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))

    # Convert INF to -1 for unreachable vertices
    return [-1 if d == INF else d for d in dist]


def solve() -> None:
    """
    Reads input from stdin, builds graph, runs Dijkstra, prints results.
    Input format (exact for deterministic processing):
    First line: two integers n m
    Next m lines: u v w (directed edge from u to v with weight w)
    Last line: source vertex s

    Example:
    5 6
    0 1 2
    0 2 4
    1 2 1
    1 3 7
    2 4 3
    3 4 2
    0

    Output:
    0 2 3 9 6
    """
    data = sys.stdin.read().strip().split()
    if not data:
        return

    it = iter(data)
    try:
        n = int(next(it))
        m = int(next(it))
    except StopIteration:
        return

    # Build adjacency list
    adj: List[List[Tuple[int, int]]] = [[] for _ in range(n)]

    for _ in range(m):
        try:
            u = int(next(it))
            v = int(next(it))
            w = int(next(it))
        except StopIteration:
            return
        # Input validation: weights are non-negative
        if w < 0:
            raise ValueError("Edge weights must be non-negative.")
        adj[u].append((v, w))

    # Read source vertex
    try:
        source = int(next(it))
    except StopIteration:
        return

    # Run Dijkstra
    dist = dijkstra(n, adj, source)

    # Output: space-separated distances
    print(" ".join(map(str, dist)))


if __name__ == "__main__":
    solve()
```
