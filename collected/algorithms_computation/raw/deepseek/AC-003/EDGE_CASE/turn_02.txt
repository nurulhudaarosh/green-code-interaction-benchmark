"""
Single-Source Route Analyzer
----------------------------
Problem Restatement:
- Input: A directed graph with n vertices (0 to n-1) and m edges.
  Each edge (u, v, w) is directed from u to v with non-negative weight w.
- Source vertex: s.
- Objective: Compute the shortest path distance from s to every vertex.
- If a vertex is unreachable, its distance is -1.
- All edge weights are non-negative (w >= 0).
- There are no negative cycles (guaranteed by non-negative weights).

Constraints (boundary values handled):
- n: 1 <= n <= 10^5
- m: 0 <= m <= 2 * 10^5
- weights: 0 <= w <= 10^9
- source: 0 <= s < n
- Graph may be disconnected, have self-loops, parallel edges, zero-weight edges.

Algorithm:
- Dijkstra's algorithm with adjacency list and binary heap (heapq).
- Time: O((n + m) log n)
- Space: O(n + m)

Tie-breaking:
- When multiple vertices have the same distance in the priority queue,
  heapq naturally orders by vertex index (since tuples are (distance, vertex)).
  This is deterministic and preserves the original tie-breaking rule.
- If multiple paths yield the same shortest distance, any shortest path is acceptable.
  Our algorithm does not reconstruct paths, only distances.

Output:
- Space-separated distances in vertex order: dist[0] dist[1] ... dist[n-1]
- Unreachable: -1
"""

import sys
import heapq
from typing import List, Tuple


def dijkstra(
    n: int,
    adj: List[List[Tuple[int, int]]],
    source: int
) -> List[int]:
    """
    Dijkstra's algorithm using binary heap.

    Args:
        n: Number of vertices (0..n-1).
        adj: Adjacency list where adj[u] = [(v, weight), ...].
        source: Source vertex index.

    Returns:
        List of shortest distances from source to each vertex.
        Unreachable vertices have distance -1.
    """
    INF = 10**18  # Larger than max possible path (1e5 * 1e9 = 1e14)
    dist = [INF] * n
    dist[source] = 0

    # Priority queue: (distance, vertex)
    # Tie-breaking: if distances equal, heapq compares vertex index (deterministic)
    pq = [(0, source)]

    while pq:
        d, u = heapq.heappop(pq)

        # Skip outdated entries
        if d != dist[u]:
            continue

        # Relax outgoing edges
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))

    # Convert INF to -1 for unreachable vertices
    return [-1 if d == INF else d for d in dist]


def solve() -> None:
    """
    Reads input from stdin, builds graph, runs Dijkstra, prints results.

    Input format:
    First line: n m
    Next m lines: u v w (directed edge from u to v with weight w)
    Last line: source vertex s

    Handles boundary cases:
    - n = 1, m = 0 (single vertex, no edges)
    - m = 0 (empty graph)
    - Self-loops (u == v)
    - Parallel edges (multiple edges between same u,v)
    - Zero-weight edges
    - Source unreachable to many vertices
    - Maximum values: n=100000, m=200000, w=1e9
    """
    data = sys.stdin.read().strip().split()
    if not data:
        return

    it = iter(data)
    try:
        n = int(next(it))
        m = int(next(it))
    except StopIteration:
        return

    # Validate n
    if n < 1:
        raise ValueError("n must be >= 1")

    # Build adjacency list
    adj: List[List[Tuple[int, int]]] = [[] for _ in range(n)]

    for _ in range(m):
        try:
            u = int(next(it))
            v = int(next(it))
            w = int(next(it))
        except StopIteration:
            return

        # Validate vertices
        if u < 0 or u >= n or v < 0 or v >= n:
            raise ValueError(f"Vertex index out of range: u={u}, v={v}, n={n}")

        # Validate weight (non-negative)
        if w < 0:
            raise ValueError(f"Edge weight must be non-negative: {w}")

        adj[u].append((v, w))

    # Read source vertex
    try:
        source = int(next(it))
    except StopIteration:
        return

    if source < 0 or source >= n:
        raise ValueError(f"Source vertex {source} out of range [0, {n-1}]")

    # Run Dijkstra
    dist = dijkstra(n, adj, source)

    # Output: space-separated distances
    print(" ".join(map(str, dist)))

def run_tests():
    """
    Run test cases covering boundary values and difficult cases.
    Each test compares output against expected result.
    """
    import io
    from contextlib import redirect_stdout

    test_cases = [
        # Test 1: Single vertex, no edges
        {
            "input": "1 0\n0\n",
            "expected": "0\n"
        },
        # Test 2: Single vertex with self-loop (should be ignored for distance 0)
        {
            "input": "1 1\n0 0 5\n0\n",
            "expected": "0\n"
        },
        # Test 3: Two vertices, no path from source to vertex 1
        {
            "input": "2 0\n0\n",
            "expected": "0 -1\n"
        },
        # Test 4: Simple graph with multiple paths
        {
            "input": "3 3\n0 1 2\n0 2 5\n1 2 1\n0\n",
            "expected": "0 2 3\n"
        },
        # Test 5: Zero-weight edges
        {
            "input": "3 3\n0 1 0\n1 2 0\n0 2 5\n0\n",
            "expected": "0 0 0\n"
        },
        # Test 6: Parallel edges (should pick minimum weight)
        {
            "input": "3 4\n0 1 5\n0 1 2\n0 2 3\n1 2 1\n0\n",
            "expected": "0 2 3\n"
        },
        # Test 7: Self-loop and parallel edges with unreachable
        {
            "input": "4 5\n0 0 1\n0 1 4\n1 2 3\n2 3 2\n1 3 10\n0\n",
            "expected": "0 4 7 9\n"
        },
        # Test 8: Source is not 0
        {
            "input": "4 3\n0 1 1\n1 2 1\n2 3 1\n2\n",
            "expected": "-1 -1 0 1\n"
        },
        # Test 9: Large weights (boundary value 1e9)
        {
            "input": "3 2\n0 1 1000000000\n1 2 1000000000\n0\n",
            "expected": "0 1000000000 2000000000\n"
        },
        # Test 10: Dense graph with many parallel edges (minimal weights)
        {
            "input": "3 6\n0 1 10\n0 1 5\n0 1 7\n1 2 3\n1 2 2\n0 2 100\n0\n",
            "expected": "0 5 7\n"
        },
        # Test 11: Disconnected graph with component
        {
            "input": "5 3\n0 1 1\n1 2 2\n3 4 3\n0\n",
            "expected": "0 1 3 -1 -1\n"
        },
        # Test 12: Cycle with zero weight (should still terminate)
        {
            "input": "3 3\n0 1 0\n1 2 0\n2 0 0\n0\n",
            "expected": "0 0 0\n"
        },
        # Test 13: Maximum n with minimal edges (n=1 already tested, test n=5)
        {
            "input": "5 0\n0\n",
            "expected": "0 -1 -1 -1 -1\n"
        },
        # Test 14: Multiple shortest paths, tie-breaking doesn't affect distances
        {
            "input": "4 5\n0 1 2\n0 2 2\n1 3 3\n2 3 3\n0 3 5\n0\n",
            "expected": "0 2 2 5\n"
        },
    ]

    print("=" * 60)
    print("RUNNING TESTS FOR SINGLE-SOURCE ROUTE ANALYZER")
    print("=" * 60)

    passed = 0
    for i, test in enumerate(test_cases, 1):
        # Redirect stdin to test input
        sys.stdin = io.StringIO(test["input"])

        # Capture stdout
        f = io.StringIO()
        with redirect_stdout(f):
            try:
                solve()
            except Exception as e:
                print(f"Test {i}: FAILED with exception: {e}")
                continue

        output = f.getvalue()
        if output == test["expected"]:
            print(f"Test {i}: PASSED")
            passed += 1
        else:
            print(f"Test {i}: FAILED")
            print(f"  Expected: {repr(test['expected'])}")
            print(f"  Got:      {repr(output)}")

    print("=" * 60)
    print(f"Passed {passed}/{len(test_cases)} tests")
    print("=" * 60)

    # Reset stdin
    sys.stdin = sys.__stdin__


if __name__ == "__main__":
    # If no command-line arguments, run tests
    if len(sys.argv) == 1:
        run_tests()
    else:
        # Otherwise, run solve() for actual input
        solve()
