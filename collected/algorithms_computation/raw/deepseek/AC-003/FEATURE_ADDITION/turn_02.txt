"""
Shortest Path from Single Source in Directed Graph with Non-negative Weights
===========================================================================
Problem:
    Given a directed graph with n vertices (0 to n-1) and m edges, each with
    non-negative integer weight, and a source vertex s, compute the shortest
    distance from s to every other vertex. If a vertex is unreachable from s,
    its distance should be -1.

    ADDITIONAL FEATURE: Return an optional `operation_summary` field containing
    a deterministic summary of the computation. When the feature is not requested,
    the original fields remain unchanged.

Constraints:
    - 1 ≤ n ≤ 10^5
    - 0 ≤ m ≤ 2*10^5
    - 0 ≤ weight ≤ 10^9
    - Graph may contain multiple edges and self-loops.
    - All weights are non-negative.

Required Original Output:
    A list `dist` of length n where dist[i] is the shortest distance from s to i,
    or -1 if i is unreachable.

New Optional Output:
    An `operation_summary` dictionary containing:
        - 'vertices_processed': number of vertices popped from priority queue
        - 'edges_relaxed': number of edges examined/relaxed attempts
        - 'distance_updates': number of times a distance was improved
        - 'heap_operations': total push + pop operations on heap

Algorithm:
    Dijkstra's algorithm using adjacency lists and a binary heap (via heapq).
    Time complexity: O((n + m) log n)
    Space complexity: O(n + m)

Implementation:
    - `shortest_path(n, edges, s, return_summary=False)` expects:
        n: int
        edges: List[Tuple[int, int, int]]  # (u, v, w) meaning edge u -> v with weight w
        s: int
        return_summary: bool (default False) - if True, returns (dist, summary)
                      if False, returns dist only (backward compatible)
    - Returns:
        If return_summary is False: List[int] of length n
        If return_summary is True: Tuple[List[int], Dict[str, int]]
"""

import heapq
from typing import List, Tuple, Dict, Union

def shortest_path(
    n: int, 
    edges: List[Tuple[int, int, int]], 
    s: int,
    return_summary: bool = False
) -> Union[List[int], Tuple[List[int], Dict[str, int]]]:
    """
    Computes shortest distances from source s to all vertices in a directed graph.
    Optionally returns an operation summary.

    Args:
        n: number of vertices (0..n-1)
        edges: list of (u, v, w) where edge u -> v has weight w (w >= 0)
        s: source vertex
        return_summary: if True, return (dist, summary) tuple; else return dist only

    Returns:
        If return_summary is False:
            dist: list of length n where dist[i] is shortest distance from s to i,
                  or -1 if unreachable.
        If return_summary is True:
            (dist, summary) where summary is a dict with operation counts.
    """
    # Initialize counters
    vertices_processed = 0
    edges_relaxed = 0
    distance_updates = 0
    heap_operations = 0  # pushes + pops

    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        # Only add edge if weight is non-negative (already guaranteed but safe)
        if w >= 0:
            adj[u].append((v, w))
        # If negative weights were allowed, we'd need Bellman-Ford,
        # but problem states non-negative.

    # Initialize distances
    INF = 10**30  # larger than any possible path (max weight 1e9 * 1e5 edges = 1e14)
    dist = [INF] * n
    dist[s] = 0

    # Priority queue: (distance, vertex)
    pq = [(0, s)]
    heap_operations += 1  # initial push

    while pq:
        d, u = heapq.heappop(pq)
        heap_operations += 1
        vertices_processed += 1

        # If this is an outdated entry, skip it
        if d != dist[u]:
            continue

        # Relax edges from u
        for v, w in adj[u]:
            edges_relaxed += 1
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                distance_updates += 1
                heapq.heappush(pq, (nd, v))
                heap_operations += 1

    # Convert INF to -1 for unreachable vertices
    result_dist = [-1 if d == INF else d for d in dist]

    # Return based on flag
    if return_summary:
        summary = {
            'vertices_processed': vertices_processed,
            'edges_relaxed': edges_relaxed,
            'distance_updates': distance_updates,
            'heap_operations': heap_operations
        }
        return result_dist, summary
    
    return result_dist

    # Test 1: Simple graph
