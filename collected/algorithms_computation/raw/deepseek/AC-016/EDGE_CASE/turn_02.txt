import heapq
from dataclasses import dataclass
from typing import List, Tuple


# ============================================================
# Priority Queue Simulator
# ------------------------------------------------------------
# Non-preemptive single-server queue. Larger priority runs first.
# Ties: earlier arrival_time, then smaller job_id.
# Returns (job_id, completion_time, waiting_time) per job and
# the average waiting time.
# ============================================================


@dataclass(frozen=True)
class Job:
    job_id: int
    arrival_time: int
    burst_time: int
    priority: int


def simulate_nonpreemptive_priority_queue(
    jobs: List[Job],
) -> Tuple[List[Tuple[int, int, int]], float]:
    """
    Non-preemptive single-server priority queue.

    Selection when the server is free:
      1. Larger `priority` first.
      2. Tie -> earlier `arrival_time`.
      3. Tie -> smaller `job_id`.

    Returns
    -------
    results : list of (job_id, completion_time, waiting_time), sorted by job_id
    avg_wait : float, average waiting time
    """
    if not jobs:
        return [], 0.0

    # Deterministic arrival order for admission.
    arrivals = sorted(jobs, key=lambda j: (j.arrival_time, j.job_id))

    # Min-heap keyed by (-priority, arrival_time, job_id).
    # heapq is a min-heap, so negating priority pops the largest priority first.
    # Tuple ordering handles the secondary and tertiary tie-breakers automatically.
    ready: List[Tuple[int, int, int, int]] = []  # (neg_prio, arrival, id, burst)

    current_time = 0
    idx = 0
    n = len(arrivals)
    completion: dict[int, int] = {}
    waiting: dict[int, int] = {}

    while idx < n or ready:
        # Server idle: jump to next arrival (no busy-waiting).
        if not ready and idx < n and arrivals[idx].arrival_time > current_time:
            current_time = arrivals[idx].arrival_time

        # Admit every job that has arrived by current_time.
        # Monotonic idx pointer -> each job pushed at most once -> O(n log n) total,
        # even in the worst case (all same priority + same arrival).
        while idx < n and arrivals[idx].arrival_time <= current_time:
            j = arrivals[idx]
            heapq.heappush(ready, (-j.priority, j.arrival_time, j.job_id, j.burst_time))
            idx += 1

        # Non-preemptive: run chosen job to completion.
        _, arr, jid, burst = heapq.heappop(ready)
        start_time = current_time
        current_time += burst
        completion[jid] = current_time
        waiting[jid] = start_time - arr

    results = [
        (j.job_id, completion[j.job_id], waiting[j.job_id])
        for j in sorted(jobs, key=lambda x: x.job_id)
    ]
    avg_wait = sum(waiting.values()) / n
    return results, avg_wait


# ============================================================
# Tests
# ============================================================

def test_empty():
    res, avg = simulate_nonpreemptive_priority_queue([])
    assert res == [] and avg == 0.0, "empty input"


def test_single_late_job():
    jobs = [Job(1, 10, 5, 3)]
    res, avg = simulate_nonpreemptive_priority_queue(jobs)
    assert res == [(1, 15, 0)], res
    assert avg == 0.0


def test_idle_gap_between_jobs():
    # Job 1 runs 0..2; next arrival is 7, so server idles 2..7.
    jobs = [Job(1, 0, 2, 1), Job(2, 7, 3, 1)]
    res, avg = simulate_nonpreemptive_priority_queue(jobs)
    assert res == [(1, 2, 0), (2, 10, 0)], res
    assert avg == 0.0


def test_priority_beats_earlier_arrival():
    # Job 1 arrives first but Job 2 has higher priority and runs right after Job 1.
    jobs = [Job(1, 0, 4, 1), Job(2, 1, 2, 9)]
    res, avg = simulate_nonpreemptive_priority_queue(jobs)
    assert res == [(1, 4, 0), (2, 6, 3)], res
    assert abs(avg - 1.5) < 1e-9


def test_tie_break_priority_then_arrival_then_id():
    jobs = [
        Job(3, 0, 1, 5),
        Job(1, 0, 1, 5),
        Job(2, 1, 1, 5),
    ]
    res, avg = simulate_nonpreemptive_priority_queue(jobs)
    assert res == [(1, 1, 0), (2, 3, 1), (3, 2, 1)], res
    assert abs(avg - (0 + 1 + 1) / 3) < 1e-9


def test_worst_case_same_priority_same_arrival():
    """
    Worst case for the heap key: all arrivals = 0, all priorities equal.
    Heap must fully resolve via job_id. Stresses admission loop too.
    """
    n = 200
    # Reverse id order in input to prove sorting/heap fixes ordering.
    jobs = [Job(job_id=i, arrival_time=0, burst_time=1, priority=1)
            for i in range(n, 0, -1)]

    res, avg = simulate_nonpreemptive_priority_queue(jobs)

    expected = [(i, i, i - 1) for i in range(1, n + 1)]
    assert res == expected, (res[:5], expected[:5])

    expected_avg = sum(i - 1 for i in range(1, n + 1)) / n
    assert abs(avg - expected_avg) < 1e-9


def test_worst_case_staggered_arrivals_equal_priority():
    """
    Staggered arrivals (0,1,2,...), equal priority, burst=1.
    Each job's arrival equals previous completion -> zero waiting.
    """
    n = 200
    jobs = [Job(job_id=i, arrival_time=i - 1, burst_time=1, priority=1)
            for i in range(1, n + 1)]

    res, avg = simulate_nonpreemptive_priority_queue(jobs)

    expected = [(i, i, 0) for i in range(1, n + 1)]
    assert res == expected, (res[:5], expected[:5])
    assert avg == 0.0


def test_worst_case_all_same_priority_high_arrivals():
    """
    All jobs arrive at the same late time with equal priority -> pure id order.
    Server idles until that arrival.
    """
    n = 50
    arrive = 100
    jobs = [Job(job_id=i, arrival_time=arrive, burst_time=2, priority=7)
            for i in range(1, n + 1)]

    res, avg = simulate_nonpreemptive_priority_queue(jobs)

    expected = []
    t = arrive
    for i in range(1, n + 1):
        t += 2
        expected.append((i, t, (t - 2) - arrive))

    assert res == expected, (res[:3], expected[:3])
    expected_avg = sum(w for _, _, w in expected) / n
    assert abs(avg - expected_avg) < 1e-9


# ============================================================
# Demo + Test Runner
# ============================================================

def _run_demo():
    jobs = [
        Job(job_id=1, arrival_time=0, burst_time=4, priority=2),
        Job(job_id=2, arrival_time=1, burst_time=3, priority=5),
        Job(job_id=3, arrival_time=2, burst_time=1, priority=5),
        Job(job_id=4, arrival_time=3, burst_time=2, priority=1),
        Job(job_id=5, arrival_time=4, burst_time=5, priority=5),
    ]
    results, avg = simulate_nonpreemptive_priority_queue(jobs)

    print("=== Demo ===")
    print(f"{'JobID':>5} {'Completion':>11} {'Waiting':>8}")
    for jid, comp, wait in results:
        print(f"{jid:>5} {comp:>11} {wait:>8}")
    print(f"\nAverage waiting time: {avg:.3f}\n")


if __name__ == "__main__":
    _run_demo()

    tests = [
        test_empty,
        test_single_late_job,
        test_idle_gap_between_jobs,
        test_priority_beats_earlier_arrival,
        test_tie_break_priority_then_arrival_then_id,
        test_worst_case_same_priority_same_arrival,
        test_worst_case_staggered_arrivals_equal_priority,
        test_worst_case_all_same_priority_high_arrivals,
    ]
    for t in tests:
        t()
        print(f"PASS  {t.__name__}")
    print("\nAll tests passed.")
