```python
#!/usr/bin/env python3
"""
Problem: n stages, choose exactly one option per stage. Each option has
(cost, memory, score). Total cost <= B and total memory <= M.
Maximize total score; ties -> lexicographically smallest choice vector.

Input (stdin, whitespace separated integers):
    n B M
    then for each stage: k, followed by k triples "cost memory score"
Output:
    best total score on line 1, 0-based choice vector on line 2.
    Prints -1 if no feasible selection exists.

Algorithm: layered suffix DP over (stage, budget, memory).
    dp[i][b][m] = max score from stages i..n-1 using at most b budget and
                  at most m memory (-inf if infeasible); dp[n][*][*] = 0.
Reconstruction goes forward from (B, M), at each stage taking the smallest
option index that still attains dp[i][b][m]. Since dp[i+1] is the exact
optimum for the remaining capacity, this yields the lexicographically
smallest optimal vector.
Time O(sum(k_i) * B * M), memory O(n * B * M).
"""
import sys

NEG = float("-inf")


def solve(B, M, stages):
    n = len(stages)
    layers = [None] * (n + 1)
    layers[n] = [[0] * (M + 1) for _ in range(B + 1)]

    for i in range(n - 1, -1, -1):
        nxt = layers[i + 1]
        cur = [[NEG] * (M + 1) for _ in range(B + 1)]
        for c, w, s in stages[i]:
            if c > B or w > M:
                continue
            for b in range(c, B + 1):
                src = nxt[b - c]
                row = cur[b]
                for m in range(w, M + 1):
                    v = src[m - w]
                    if v != NEG:
                        v += s
                        if v > row[m]:
                            row[m] = v
        layers[i] = cur

    best = layers[0][B][M]
    if best == NEG:
        return None

    b, m, choice = B, M, []
    for i in range(n):
        target = layers[i][b][m]
        for j, (c, w, s) in enumerate(stages[i]):
            if c <= b and w <= m:
                v = layers[i + 1][b - c][m - w]
                if v != NEG and v + s == target:
                    choice.append(j)
                    b -= c
                    m -= w
                    break
    return best, choice


def main():
    data = sys.stdin.read().split()
    n, B, M = int(data[0]), int(data[1]), int(data[2])
    pos = 3
    stages = []
    for _ in range(n):
        k = int(data[pos]); pos += 1
        opts = []
        for _ in range(k):
            opts.append((int(data[pos]), int(data[pos + 1]), int(data[pos + 2])))
            pos += 3
        stages.append(opts)

    res = solve(B, M, stages)
    if res is None:
        print(-1)
    else:
        print(res[0])
        print(" ".join(map(str, res[1])))


if __name__ == "__main__":
    main()
```
