```python
"""
PROBLEM (restated)
-------------------
Given two strings s1 and s2, compute:
  1. The length of their Longest Common Subsequence (LCS) — the length of
     the longest sequence of characters that appears, in order (not
     necessarily contiguously), in both s1 and s2.
  2. ONE concrete LCS string achieving that length.

Multiple distinct LCS strings can share the same maximum length, so a
DETERMINISTIC tie-breaking rule is required so the same inputs always
produce the same output. Rule used here:

    While reconstructing backwards through the DP table, whenever a
    non-matching cell has two predecessor cells with EQUAL DP scores
    (dp[i-1][j] == dp[i][j-1]), we prefer the move that advances the
    pointer in the FIRST string (step i -> i-1) rather than stepping in s2.

CONSTRAINTS
------------
- Inputs: two strings s1, s2 (may be empty, may contain any characters,
  including repeated characters).
- Must use classic O(n*m) time / O(n*m) space quadratic DP — including for
  worst-case-like inputs (e.g. long strings, strings with many repeated
  characters, or strings with no common characters at all), which stress
  the size of the DP table and the number of tie-breaking decisions made
  during reconstruction.
- No randomness, no network/API calls, no external libraries, no user
  interaction beyond an optional __main__ demo — deterministic, pure
  standard-library Python only.
- Output: the integer LCS length, and one valid LCS string chosen by the
  deterministic rule above.

ALGORITHM
---------
1. Build dp table of size (n+1) x (m+1), where
       dp[i][j] = length of LCS of s1[:i] and s2[:j]
   Recurrence:
       dp[0][j] = dp[i][0] = 0
       if s1[i-1] == s2[j-1]: dp[i][j] = dp[i-1][j-1] + 1
       else:                  dp[i][j] = max(dp[i-1][j], dp[i][j-1])
2. dp[n][m] is the LCS length.
3. Reconstruct by walking from (n, m) back to (0, 0):
       - if s1[i-1] == s2[j-1]: this char is part of the LCS;
         prepend it, move to (i-1, j-1)
       - else if dp[i-1][j] >= dp[i][j-1]: move to (i-1, j)   <-- tie -> prefer s1 advance
       - else: move to (i, j-1)
   This produces a single deterministic LCS string.
"""

from typing import Tuple


def lcs(s1: str, s2: str) -> Tuple[int, str]:
    """Compute LCS length and one deterministic LCS string of s1, s2."""
    n, m = len(s1), len(s2)

    # dp[i][j] = LCS length of s1[:i] and s2[:j]
    dp = [[0] * (m + 1) for _ in range(n + 1)]

    for i in range(1, n + 1):
        ci = s1[i - 1]
        row, prev_row = dp[i], dp[i - 1]
        for j in range(1, m + 1):
            if ci == s2[j - 1]:
                row[j] = prev_row[j - 1] + 1
            else:
                a = prev_row[j]
                b = row[j - 1]
                row[j] = a if a >= b else b

    length = dp[n][m]

    # Reconstruction with deterministic tie-breaking.
    chars = []
    i, j = n, m
    while i > 0 and j > 0:
        if s1[i - 1] == s2[j - 1]:
            chars.append(s1[i - 1])
            i -= 1
            j -= 1
        else:
            up = dp[i - 1][j]     # advancing in s1
            left = dp[i][j - 1]   # advancing in s2
            if up >= left:        # tie or better -> prefer advancing in s1
                i -= 1
            else:
                j -= 1

    chars.reverse()
    return length, "".join(chars)


if __name__ == "__main__":
    # Original baseline tests (kept exactly, same output format / rules).
    baseline_tests = [
        ("ABCBDAB", "BDCABA"),
        ("", ""),
        ("abc", ""),
        ("", "xyz"),
        ("AGGTAB", "GXTXAYB"),
        ("aaaa", "aa"),
    ]

    # Additional worst-case-like structural tests for the stated O(n*m)
    # complexity: these stress the DP table size and the frequency of
    # tie-breaking decisions during reconstruction, without changing the
    # algorithm, its output format, or its tie-breaking rule.
    worst_case_tests = [
        # All characters identical in both strings -> every cell is a
        # match; maximal number of diagonal steps, dp table fully "dense".
        ("a" * 50, "a" * 40),

        # No characters in common at all -> dp stays all zeros, forcing
        # the tie-break rule (up >= left) to fire on every single step
        # of the reconstruction walk.
        ("abcdefghij", "klmnopqrst"),

        # One string entirely repeats a single character while the other
        # is a different single repeated character interleaved with a
        # rare match -> tests sparse matches inside a large dense table.
        ("b" * 30 + "x" + "b" * 30, "b" * 60),

        # Strongly alternating two-character strings -> repeated ties at
        # nearly every cell since both characters recur constantly,
        # heavily exercising the "prefer advancing in s1" rule.
        ("ab" * 25, "ba" * 25),

        # One string is a subsequence embedded in a much longer string
        # with many repeats of the same characters around it -> classic
        # worst-case-like shape for LCS DP (long, repetitive, O(n*m) cells).
        ("x" * 20 + "abcde" + "x" * 20, "y" * 15 + "abcde" + "y" * 15),

        # Long strings built from a small repeating alphabet -> maximizes
        # the number of equal-length ties encountered during traversal.
        ("abc" * 20, "cab" * 20),

        # Reverse of one another over a repeated alphabet -> many equal
        # dp scores throughout the table, stressing tie-breaking density.
        ("abcabcabcabc", "cbacbacbacba"),
    ]

    print("=== Baseline tests ===")
    for a, b in baseline_tests:
        length, subseq = lcs(a, b)
        print(f"s1={a!r} s2={b!r} -> length={length}, lcs={subseq!r}")

    print("\n=== Worst-case-like structural tests ===")
    for a, b in worst_case_tests:
        length, subseq = lcs(a, b)
        # For long strings, avoid flooding output: show lengths plus a
        # truncated preview of inputs/result while still printing the
        # full, exact length and LCS string (output content unchanged).
        a_preview = a if len(a) <= 40 else a[:37] + "..."
        b_preview = b if len(b) <= 40 else b[:37] + "..."
        subseq_preview = subseq if len(subseq) <= 40 else subseq[:37] + "..."
        print(
            f"s1={a_preview!r} (len={len(a)}) "
            f"s2={b_preview!r} (len={len(b)}) -> "
            f"length={length}, lcs={subseq_preview!r} (lcs_len={len(subseq)})"
        )
```
