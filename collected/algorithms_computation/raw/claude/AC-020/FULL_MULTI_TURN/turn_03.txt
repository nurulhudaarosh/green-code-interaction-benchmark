I'll write the feature and run it before showing it to you.## Original problem and required outputs (unchanged)

Given a complete symmetric distance matrix with at most 14 cities, find the minimum Hamiltonian cycle starting and ending at city 0.

- **Algorithm:** Held-Karp subset dynamic programming.
- **Tie rule:** the smaller predecessor wins. This applies at every DP transition and at the closing edge back to city 0, so the smaller last city wins.
- **Output:** the minimum total cost and the tour as a list that starts and ends at 0.
- **Environment:** standard library only, deterministic, with no network, randomness, or interaction.
- **Sample result:** cost 80, tour `[0, 2, 3, 1, 0]`.

## New feature

`held_karp(dist, with_summary=False)` now takes an optional flag.

- **Flag off (default):** it returns the same `(cost, tour)` tuple as before, so existing callers are unaffected.
- **Flag on:** it returns a dict with `min_cost`, `tour`, and the new `operation_summary`.

The summary counts:

- Base states initialized.
- DP states computed.
- Transition candidates examined.
- Strict improvements (a new best candidate replaced the old one).
- Ties kept for the smaller predecessor (an equal candidate was rejected, so the earlier, smaller predecessor stayed).
- Closing candidates examined.
- Reconstruction steps.
- `total_operations`, the sum of the above.

The counters only record what the algorithm does and never affect its decisions. They are deterministic, so the same input always gives the same summary. For the sample, `total_operations` is 44.

```python
from itertools import permutations
from typing import Any, Dict, List, Tuple, Union

INF = float("inf")


def held_karp(
    dist: List[List[float]], with_summary: bool = False
) -> Union[Tuple[float, List[int]], Dict[str, Any]]:
    """Minimum Hamiltonian cycle from/to city 0 via Held-Karp.

    Tie rule: the smaller predecessor wins at every step, including the
    closing edge back to city 0 (smaller last city wins).

    with_summary=False -> returns (min_cost, tour)            [original behavior]
    with_summary=True  -> returns {"min_cost", "tour", "operation_summary"}
    """
    n = len(dist)
    ops = {
        "base_states_initialized": 0,
        "dp_states_computed": 0,
        "transition_candidates_examined": 0,
        "strict_improvements": 0,
        "ties_kept_smaller_predecessor": 0,
        "closing_candidates_examined": 0,
        "reconstruction_steps": 0,
    }

    def finish(cost, tour):
        if not with_summary:
            return cost, tour
        summary = dict(ops)
        summary["total_operations"] = sum(ops.values())
        return {"min_cost": cost, "tour": tour, "operation_summary": summary}

    if n == 0:
        return finish(0, [])
    if n == 1:
        return finish(0, [0, 0])

    m = n - 1                                   # cities 1..n-1 -> bits 0..m-1
    full = (1 << m) - 1
    dp = [[INF] * m for _ in range(1 << m)]
    parent = [[-1] * m for _ in range(1 << m)]  # -1 => predecessor is city 0

    for j in range(m):
        dp[1 << j][j] = dist[0][j + 1]
        ops["base_states_initialized"] += 1

    for mask in range(1, 1 << m):
        for j in range(m):
            if not (mask >> j) & 1:
                continue
            prev = mask ^ (1 << j)
            if prev == 0:
                continue
            ops["dp_states_computed"] += 1
            best, best_i = INF, -1
            for i in range(m):                  # ascending, strict '<'
                if (prev >> i) & 1:
                    ops["transition_candidates_examined"] += 1
                    cand = dp[prev][i] + dist[i + 1][j + 1]
                    if cand < best:
                        best, best_i = cand, i
                        ops["strict_improvements"] += 1
                    elif cand == best:
                        ops["ties_kept_smaller_predecessor"] += 1
            dp[mask][j] = best
            parent[mask][j] = best_i

    best_cost, last = INF, -1
    for j in range(m):                          # smaller last city wins ties
        ops["closing_candidates_examined"] += 1
        cand = dp[full][j] + dist[j + 1][0]
        if cand < best_cost:
            best_cost, last = cand, j
            ops["strict_improvements"] += 1
        elif cand == best_cost:
            ops["ties_kept_smaller_predecessor"] += 1

    path, mask, j = [], full, last
    while j != -1:
        ops["reconstruction_steps"] += 1
        path.append(j + 1)
        p = parent[mask][j]
        mask ^= 1 << j
        j = p
    path.reverse()
    return finish(best_cost, [0] + path + [0])


def brute_force(dist):
    """Reference: min cost, ties broken by smallest last city, then earlier ones."""
    n = len(dist)
    best = None
    for p in permutations(range(1, n)):
        t = [0, *p, 0]
        c = sum(dist[a][b] for a, b in zip(t, t[1:]))
        key = (c, t[::-1])
        if best is None or key < best:
            best = key
    return best[0], best[1][::-1]


if __name__ == "__main__":
    sample = [
        [0, 10, 15, 20],
        [10, 0, 35, 25],
        [15, 35, 0, 30],
        [20, 25, 30, 0],
    ]

    plain = held_karp(sample)
    print("Default:", plain)                    # (80, [0, 2, 3, 1, 0])
    assert plain == (80, [0, 2, 3, 1, 0])

    rich = held_karp(sample, with_summary=True)
    print("With summary:", rich)
    assert (rich["min_cost"], rich["tour"]) == plain
    assert rich == held_karp(sample, with_summary=True)   # deterministic

    for n in range(2, 8):
        eq = [[0 if i == j else 1 for j in range(n)] for i in range(n)]
        mx = [[0 if i == j else 1 + (i + j) % 3 for j in range(n)] for i in range(n)]
        for d in (eq, mx):
            assert held_karp(d) == brute_force(d)
            r = held_karp(d, with_summary=True)
            assert (r["min_cost"], r["tour"]) == held_karp(d)
    print("All checks passed.")
```
