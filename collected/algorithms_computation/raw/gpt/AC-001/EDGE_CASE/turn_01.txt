```python
Interval Conflict Analyzer

Problem:
Given half-open meeting intervals [start, end), find:
1. The maximum number of simultaneously active meetings.
2. The sorted times at which that maximum begins.

Key constraints:
- Intervals are half-open: [start, end).
- A meeting is active at time t when start <= t < end.
- Meetings with end == start at the same timestamp do not overlap.
- End events at the same timestamp are processed before start events.
- Results must be deterministic.
- Only the Python standard library is used.

Algorithm:
Use a sweep-line over start/end events.
For each interval:
    - Add a start event at start.
    - Add an end event at end.

Events are sorted by:
    1. timestamp
    2. event type, with end events before start events

While sweeping:
- Process all events at each timestamp.
- Apply end events first, then start events.
- If the active count becomes larger than the previous maximum,
  record that timestamp as the beginning of a new maximum.
- If the active count returns to the same maximum later, record
  that timestamp only when the maximum begins again.

Output:
{
    "max_concurrent": int,
    "peak_times": list
}

For empty input, the result is:
{
    "max_concurrent": 0,
    "peak_times": []
}
```

from typing import Iterable, Tuple, Any


def interval_conflict_analyzer(
    intervals: Iterable[Tuple[Any, Any]]
) -> dict:
    """
    Analyze half-open meeting intervals [start, end).

    Args:
        intervals: An iterable of (start, end) pairs.

    Returns:
        A dictionary containing:
            max_concurrent: maximum number of simultaneously active meetings
            peak_times: sorted times at which that maximum begins

    Raises:
        ValueError: If an interval has start > end.
    """

    events = []

    for start, end in intervals:
        if start > end:
            raise ValueError(
                f"Invalid interval [{start}, {end}): start must not exceed end."
            )

        # Ignore zero-length intervals because [t, t) contains no time.
        if start == end:
            continue

        # Event type:
        # 0 = end
        # 1 = start
        #
        # Therefore, at the same timestamp, end events are processed
        # before start events.
        events.append((start, 1))
        events.append((end, 0))

    if not events:
        return {
            "max_concurrent": 0,
            "peak_times": []
        }

    # Python tuple sorting gives deterministic ordering:
    # timestamp first, then event type (end before start).
    events.sort()

    active = 0
    max_concurrent = 0
    peak_times = []

    i = 0

    while i < len(events):
        current_time = events[i][0]

        # Process every event at this timestamp.
        # End events are already ordered before start events.
        while i < len(events) and events[i][0] == current_time:
            event_type = events[i][1]

            if event_type == 0:
                active -= 1
            else:
                active += 1

            i += 1

        # The active count after processing the timestamp represents
        # the number of meetings active immediately after current_time.
        if active > max_concurrent:
            max_concurrent = active
            peak_times = [current_time]

        elif active == max_concurrent and max_concurrent > 0:
            # This timestamp begins another period having the maximum.
            #
            # Avoid duplicate timestamps, although normally each
            # timestamp is processed only once.
            if not peak_times or peak_times[-1] != current_time:
                peak_times.append(current_time)

    return {
        "max_concurrent": max_concurrent,
        "peak_times": peak_times
    }


if __name__ == "__main__":
    # Example:
    #
    # [1, 4) and [2, 5) overlap from 2 to 4.
    # [4, 6) starts exactly when [1, 4) ends, so they do not overlap.
    example_intervals = [
        (1, 4),
        (2, 5),
        (4, 6),
        (5, 7),
    ]

    result = interval_conflict_analyzer(example_intervals)
    print(result)
