```python
Interval Conflict Analyzer

Problem:
Given half-open meeting intervals [start, end), find:
1. The maximum number of simultaneously active meetings.
2. The sorted times at which that maximum begins.

Half-open interval rule:
A meeting [start, end) is active at time t when:
    start <= t < end

Therefore, if one meeting ends at time t and another starts at
the same time t, they do not overlap.

Algorithm:
Use a sweep-line over start/end events.
End events at the same timestamp occur before start events.

Key constraints:
- Intervals are represented as (start, end).
- start must not be greater than end.
- Zero-length intervals [t, t) are valid but contain no active time.
- Empty input is allowed and produces zero concurrency.
- Disconnected intervals are allowed.
- Results are deterministic.
- peak_times are sorted.
- Only standard-library functionality is used.
- No network access, APIs, external services, randomness, or
  human interaction are required.

Output:
{
    "max_concurrent": int,
    "peak_times": list
}

For empty input:
{
    "max_concurrent": 0,
    "peak_times": []
}

Tie-breaking:
At the same timestamp, end events are processed before start events.
A timestamp is reported when the active meeting count begins a period
at the maximum concurrency. Each timestamp appears at most once.
"""


from typing import Any, Iterable, Tuple


def interval_conflict_analyzer(
    intervals: Iterable[Tuple[Any, Any]]
) -> dict:
    """
    Return the maximum simultaneous meetings and the sorted times
    at which that maximum begins.
    ```

    events = []

    for start, end in intervals:
        if start > end:
            raise ValueError(
                f"Invalid interval [{start}, {end}): "
                "start must not exceed end."
            )

        # [t, t) contains no active time.
        if start == end:
            continue

        # 0 = end event
        # 1 = start event
        #
        # Sorting by (time, event_type) ensures that end events
        # occur before start events at the same timestamp.
        events.append((start, 1))
        events.append((end, 0))

    # Empty input, or input containing only zero-length intervals.
    if not events:
        return {
            "max_concurrent": 0,
            "peak_times": []
        }

    events.sort()

    active = 0
    max_concurrent = 0
    peak_times = []

    i = 0

    while i < len(events):
        current_time = events[i][0]

        # Process all events at this timestamp.
        # End events are processed first because event_type 0 < 1.
        while i < len(events) and events[i][0] == current_time:
            event_type = events[i][1]

            if event_type == 0:
                active -= 1
            else:
                active += 1

            i += 1

        # The count after processing current_time is the number of
        # meetings active on the interval immediately after current_time.
        if active > max_concurrent:
            max_concurrent = active
            peak_times = [current_time]

        elif active == max_concurrent and max_concurrent > 0:
            if not peak_times or peak_times[-1] != current_time:
                peak_times.append(current_time)

    return {
        "max_concurrent": max_concurrent,
        "peak_times": peak_times
    }


def run_tests() -> None:
    """Test difficult valid cases and the original behavior."""

    # ------------------------------------------------------------
    # 1. Empty input
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([]) == {
        "max_concurrent": 0,
        "peak_times": []
    }

    # ------------------------------------------------------------
    # 2. Smallest permitted non-empty input
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([(5, 10)]) == {
        "max_concurrent": 1,
        "peak_times": [5]
    }

    # ------------------------------------------------------------
    # 3. Smallest input with a zero-length interval
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([(5, 5)]) == {
        "max_concurrent": 0,
        "peak_times": []
    }

    # ------------------------------------------------------------
    # 4. Multiple zero-length intervals
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 1),
        (5, 5),
        (10, 10)
    ]) == {
        "max_concurrent": 0,
        "peak_times": []
    }

    # ------------------------------------------------------------
    # 5. Disconnected intervals
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 2),
        (5, 6),
        (10, 11)
    ]) == {
        "max_concurrent": 1,
        "peak_times": [1, 5, 10]
    }

    # ------------------------------------------------------------
    # 6. Adjacent intervals must not overlap
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 3),
        (3, 5)
    ]) == {
        "max_concurrent": 1,
        "peak_times": [1, 3]
    }

    # ------------------------------------------------------------
    # 7. Exact overlap at the same start time
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 5),
        (1, 5)
    ]) == {
        "max_concurrent": 2,
        "peak_times": [1]
    }

    # ------------------------------------------------------------
    # 8. Maximum begins again after a lower-concurrency gap
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 4),
        (2, 3),
        (5, 8),
        (6, 7)
    ]) == {
        "max_concurrent": 2,
        "peak_times": [2, 6]
    }

    # ------------------------------------------------------------
    # 9. End and start at the same timestamp.
    #    The old meeting must end before the new one starts.
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 4),
        (4, 7)
    ]) == {
        "max_concurrent": 1,
        "peak_times": [1, 4]
    }

    # ------------------------------------------------------------
    # 10. Three meetings with the same boundary.
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 5),
        (2, 5),
        (5, 8)
    ]) == {
        "max_concurrent": 2,
        "peak_times": [2]
    }

    # ------------------------------------------------------------
    # 11. Nested intervals
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 10),
        (2, 9),
        (3, 8),
        (4, 7)
    ]) == {
        "max_concurrent": 4,
        "peak_times": [4]
    }

    # ------------------------------------------------------------
    # 12. Mixed disconnected and overlapping structures
    # ------------------------------------------------------------
    assert interval_conflict_analyzer([
        (1, 2),
        (10, 20),
        (12, 15),
        (30, 31),
        (30, 35),
        (35, 40)
    ]) == {
        "max_concurrent": 2,
        "peak_times": [12, 30]
    }

    # ------------------------------------------------------------
    # 13. Invalid interval remains rejected
    # ------------------------------------------------------------
    try:
        interval_conflict_analyzer([(10, 5)])
        assert False, "Expected ValueError for start > end"
    except ValueError:
        pass


if __name__ == "__main__":
    run_tests()

    example_intervals = [
        (1, 4),
        (2, 5),
        (4, 6),
        (5, 7),
    ]

    print(interval_conflict_analyzer(example_intervals))
