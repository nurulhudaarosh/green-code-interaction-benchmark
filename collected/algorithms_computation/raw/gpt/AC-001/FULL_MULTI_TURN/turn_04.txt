```python
Interval Conflict Analyzer

Problem:
Given half-open meeting intervals [start, end), determine:
1. The maximum number of simultaneously active meetings.
2. The sorted timestamps at which a maximum-concurrency period begins.

Optional feature:
When include_operation_summary=True, also return:
    "operation_summary": {
        "major_operations": <number of sweep-line events processed>
    }

Rules:
- Intervals are half-open: [start, end).
- A meeting is active at t when start <= t < end.
- End events at the same timestamp occur before start events.
- A maximum start time is recorded only when a new maximum begins.
- Results are deterministic.
- Invalid intervals where start >= end raise ValueError.
- Empty input is valid.
- Standard library only.
```


def analyze_intervals(intervals, include_operation_summary=False):
    """
    Analyze half-open meeting intervals.

    Returns the original two fields unless operation summaries are
    explicitly requested.
    """

    events = []

    for start, end in intervals:
        if start >= end:
            raise ValueError(
                f"Invalid interval [{start}, {end}): "
                "start must be less than end."
            )

        # 0 = end event
        # 1 = start event
        #
        # Sorting by (timestamp, event_type) guarantees that end
        # events occur before start events at the same timestamp.
        events.append((start, 1))
        events.append((end, 0))

    # Explicitly handle the empty structure.
    if not events:
        result = {
            "max_concurrent": 0,
            "max_start_times": []
        }

        if include_operation_summary:
            result["operation_summary"] = {
                "major_operations": 0
            }

        return result

    events.sort()

    active = 0
    max_concurrent = 0
    max_start_times = []
    major_operations = 0

    for timestamp, event_type in events:
        major_operations += 1

        if event_type == 0:
            # End events are processed before starts at this timestamp.
            active -= 1

        else:
            active += 1

            # Record only the timestamp where a genuinely new maximum
            # begins. This preserves deterministic tie handling.
            if active > max_concurrent:
                max_concurrent = active
                max_start_times = [timestamp]

    result = {
        "max_concurrent": max_concurrent,
        "max_start_times": max_start_times
    }

    if include_operation_summary:
        result["operation_summary"] = {
            "major_operations": major_operations
        }

    return result


def run_tests():
    """Tests for normal, smallest, empty, disconnected, and difficult cases."""

    # 1. Smallest permitted non-empty input:
    # One valid interval.
    assert analyze_intervals([(5, 10)]) == {
        "max_concurrent": 1,
        "max_start_times": [5]
    }

    # 2. Smallest input with operation summary:
    # One interval creates exactly two sweep-line events.
    assert analyze_intervals(
        [(5, 10)],
        include_operation_summary=True
    ) == {
        "max_concurrent": 1,
        "max_start_times": [5],
        "operation_summary": {
            "major_operations": 2
        }
    }

    # 3. Empty input.
    assert analyze_intervals([]) == {
        "max_concurrent": 0,
        "max_start_times": []
    }

    # 4. Empty input with operation summary.
    assert analyze_intervals(
        [],
        include_operation_summary=True
    ) == {
        "max_concurrent": 0,
        "max_start_times": [],
        "operation_summary": {
            "major_operations": 0
        }
    }

    # 5. Disconnected intervals:
    # No intervals overlap, so the maximum is 1.
    # Every interval starts a separate maximum-concurrency period.
    assert analyze_intervals([
        (1, 2),
        (5, 6),
        (10, 11)
    ]) == {
        "max_concurrent": 1,
        "max_start_times": [1, 5, 10]
    }

    # 6. Touching intervals:
    # [1, 2) and [2, 3) do NOT overlap.
    assert analyze_intervals([
        (1, 2),
        (2, 3)
    ]) == {
        "max_concurrent": 1,
        "max_start_times": [1, 2]
    }

    # 7. Maximum continues through a replacement.
    # At t=3, [1,3) ends before [3,5) starts.
    # The concurrency returns to 2, but a new maximum does not begin.
    assert analyze_intervals([
        (1, 3),
        (2, 4),
        (3, 5)
    ]) == {
        "max_concurrent": 2,
        "max_start_times": [2]
    }

    # 8. Multiple disconnected maximum periods.
    assert analyze_intervals([
        (1, 4),
        (2, 3),
        (10, 13),
        (11, 12)
    ]) == {
        "max_concurrent": 2,
        "max_start_times": [2, 11]
    }

    # 9. Several intervals beginning at the same timestamp.
    assert analyze_intervals([
        (5, 10),
        (5, 8),
        (5, 7)
    ]) == {
        "max_concurrent": 3,
        "max_start_times": [5]
    }

    # 10. End-before-start rule.
    # [1, 2) ends exactly when [2, 4) starts, so maximum concurrency is 1.
    assert analyze_intervals([
        (1, 2),
        (2, 4)
    ]) == {
        "max_concurrent": 1,
        "max_start_times": [1, 2]
    }

    # 11. Invalid interval remains invalid.
    try:
        analyze_intervals([(4, 4)])
        assert False, "Expected ValueError for zero-length interval."
    except ValueError:
        pass

    try:
        analyze_intervals([(5, 2)])
        assert False, "Expected ValueError for reversed interval."
    except ValueError:
        pass

    # 12. Deterministic operation count for disconnected intervals.
    # Three intervals => six events.
    assert analyze_intervals(
        [(1, 2), (5, 6), (10, 11)],
        include_operation_summary=True
    ) == {
        "max_concurrent": 1,
        "max_start_times": [1, 5, 10],
        "operation_summary": {
            "major_operations": 6
        }
    }

    print("All tests passed.")


if __name__ == "__main__":
    run_tests()
