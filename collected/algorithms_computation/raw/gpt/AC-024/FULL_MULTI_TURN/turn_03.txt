```python
"""
Minimum Room Assignment

Original problem:
Assign every interval to a room so that overlapping intervals never share
the same room. A room can be reused when its previous interval ends at or
before the next interval starts. Minimize the number of rooms and return the
room ID for every original interval.

Original required outputs:
- room_count: minimum number of rooms required.
- room_ids: list where room_ids[i] is the room assigned to the original
  interval i.
- Room IDs start at 0.
- Output assignments remain in the original input order.

Existing requirements:
- Intervals use half-open semantics [start, end).
- An interval ending at time t can share a room with one starting at t.
- Overlapping intervals must never share a room.
- The number of rooms must be minimal.
- Sort intervals by (start, end, original_index).
- Use a min-heap keyed by (availability_time, room_id).
- When multiple rooms are available at the same earliest time, use the
  smallest room ID for deterministic tie handling.
- If no room is available, create a new room.
- Empty input returns (0, []) when the optional feature is not requested.
- Invalid intervals with start > end raise ValueError.
- Standard library only; no network, APIs, external services, randomness,
  or human interaction.

New optional feature:
If operation_summary=True, return an additional deterministic
`operation_summary` field.

The summary reports the number of major computational decisions/operations:
- `intervals_processed`: number of intervals processed.
- `rooms_created`: number of new rooms created.
- `rooms_reused`: number of times an existing room was reused.
- `heap_pushes`: number of heap insertions.
- `heap_pops`: number of heap removals caused by room reuse.
- `sort_operations`: 1 when sorting is performed, otherwise 0.

When operation_summary is disabled, the original output remains exactly:
    (room_count, room_ids)

When operation_summary is enabled, the output is:
    (room_count, room_ids, operation_summary)

This preserves every original field and requirement while making the new
feature explicitly optional.
"""

from heapq import heappop, heappush
from typing import Dict, List, Sequence, Tuple, Union


def minimum_room_assignment(
    intervals: Sequence[Tuple[int, int]],
    operation_summary: bool = False,
) -> Union[
    Tuple[int, List[int]],
    Tuple[int, List[int], Dict[str, int]],
]:
    """
    Assign intervals to the minimum number of rooms.

    Parameters:
        intervals:
            Sequence of (start, end) pairs using [start, end) semantics.

        operation_summary:
            If False, return the original two-field result:
                (room_count, room_ids)

            If True, return:
                (room_count, room_ids, operation_summary)

    Returns:
        Original result when operation_summary=False:
            (room_count, room_ids)

        Extended result when operation_summary=True:
            (room_count, room_ids, summary)

    Raises:
        ValueError:
            If start > end for any interval.
    """

    n = len(intervals)

    # Preserve original empty-input behavior.
    if n == 0:
        if operation_summary:
            summary = {
                "intervals_processed": 0,
                "rooms_created": 0,
                "rooms_reused": 0,
                "heap_pushes": 0,
                "heap_pops": 0,
                "sort_operations": 0,
            }
            return 0, [], summary

        return 0, []

    # Validate intervals.
    for start, end in intervals:
        if start > end:
            raise ValueError(
                "Interval start cannot be greater than its end."
            )

    # Deterministic sorting:
    # 1. start time
    # 2. end time
    # 3. original index
    ordered = sorted(
        (start, end, index)
        for index, (start, end) in enumerate(intervals)
    )

    # Heap entries:
    # (availability_time, room_id)
    room_heap: List[Tuple[int, int]] = []

    room_ids = [-1] * n
    next_room_id = 0

    # Operation counters.
    intervals_processed = 0
    rooms_created = 0
    rooms_reused = 0
    heap_pushes = 0
    heap_pops = 0

    for start, end, original_index in ordered:
        intervals_processed += 1

        # The heap's first element is always the earliest available room.
        if room_heap and room_heap[0][0] <= start:
            _, room_id = heappop(room_heap)
            heap_pops += 1
            rooms_reused += 1
        else:
            # No existing room is available, so a new room is necessary.
            room_id = next_room_id
            next_room_id += 1
            rooms_created += 1

        room_ids[original_index] = room_id

        # The selected room becomes available at `end`.
        heappush(room_heap, (end, room_id))
        heap_pushes += 1

    room_count = next_room_id

    if not operation_summary:
        return room_count, room_ids

    summary = {
        "intervals_processed": intervals_processed,
        "rooms_created": rooms_created,
        "rooms_reused": rooms_reused,
        "heap_pushes": heap_pushes,
        "heap_pops": heap_pops,
        "sort_operations": 1,
    }

    return room_count, room_ids, summary


# ------------------------------------------------------------
# Tests
# ------------------------------------------------------------

def run_tests() -> None:

    # --------------------------------------------------------
    # 1. Original behavior remains unchanged by default.
    # --------------------------------------------------------
    intervals = [
        (0, 5),
        (0, 10),
        (5, 6),
    ]

    result = minimum_room_assignment(intervals)

    assert result == (2, [0, 1, 0])
    assert len(result) == 2

    # --------------------------------------------------------
    # 2. New operation_summary feature.
    # --------------------------------------------------------
    rooms, assignments, summary = minimum_room_assignment(
        intervals,
        operation_summary=True,
    )

    assert rooms == 2
    assert assignments == [0, 1, 0]

    assert summary == {
        "intervals_processed": 3,
        "rooms_created": 2,
        "rooms_reused": 1,
        "heap_pushes": 3,
        "heap_pops": 1,
        "sort_operations": 1,
    }

    # --------------------------------------------------------
    # 3. Non-overlapping intervals: one room reused.
    # --------------------------------------------------------
    rooms, assignments, summary = minimum_room_assignment(
        [(1, 3), (3, 5), (5, 8)],
        operation_summary=True,
    )

    assert rooms == 1
    assert assignments == [0, 0, 0]

    assert summary["intervals_processed"] == 3
    assert summary["rooms_created"] == 1
    assert summary["rooms_reused"] == 2
    assert summary["heap_pushes"] == 3
    assert summary["heap_pops"] == 2

    # --------------------------------------------------------
    # 4. Fully overlapping intervals: no reuse.
    # --------------------------------------------------------
    rooms, assignments, summary = minimum_room_assignment(
        [(1, 5), (2, 6), (3, 7)],
        operation_summary=True,
    )

    assert rooms == 3
    assert assignments == [0, 1, 2]

    assert summary["intervals_processed"] == 3
    assert summary["rooms_created"] == 3
    assert summary["rooms_reused"] == 0
    assert summary["heap_pushes"] == 3
    assert summary["heap_pops"] == 0

    # --------------------------------------------------------
    # 5. Equal availability times use the smallest room ID.
    # --------------------------------------------------------
    rooms, assignments, summary = minimum_room_assignment(
        [
            (0, 5),
            (0, 5),
            (5, 6),
        ],
        operation_summary=True,
    )

    assert rooms == 2
    assert assignments == [0, 1, 0]

    assert summary["rooms_created"] == 2
    assert summary["rooms_reused"] == 1

    # --------------------------------------------------------
    # 6. Empty input.
    # --------------------------------------------------------
    assert minimum_room_assignment([]) == (0, [])

    rooms, assignments, summary = minimum_room_assignment(
        [],
        operation_summary=True,
    )

    assert rooms == 0
    assert assignments == []
    assert summary == {
        "intervals_processed": 0,
        "rooms_created": 0,
        "rooms_reused": 0,
        "heap_pushes": 0,
        "heap_pops": 0,
        "sort_operations": 0,
    }

    # --------------------------------------------------------
    # 7. Original order is preserved.
    # --------------------------------------------------------
    intervals = [
        (5, 6),
        (1, 5),
        (0, 3),
        (3, 7),
    ]

    rooms, assignments, summary = minimum_room_assignment(
        intervals,
        operation_summary=True,
    )

    assert rooms == 2
    assert len(assignments) == len(intervals)
    assert summary["intervals_processed"] == 4

    # Verify that overlapping intervals never share a room.
    for i in range(len(intervals)):
        for j in range(i + 1, len(intervals)):
            s1, e1 = intervals[i]
            s2, e2 = intervals[j]

            overlap = max(s1, s2) < min(e1, e2)

            if overlap:
                assert assignments[i] != assignments[j]

    # --------------------------------------------------------
    # 8. Invalid interval remains invalid.
    # --------------------------------------------------------
    try:
        minimum_room_assignment([(5, 2)])
        assert False, "Expected ValueError"
    except ValueError:
        pass

    try:
        minimum_room_assignment([(5, 2)], operation_summary=True)
        assert False, "Expected ValueError"
    except ValueError:
        pass

    # --------------------------------------------------------
    # 9. Determinism: repeated calls give exactly the same result.
    # --------------------------------------------------------
    intervals = [
        (2, 8),
        (1, 4),
        (4, 7),
        (1, 3),
        (7, 9),
    ]

    result1 = minimum_room_assignment(
        intervals,
        operation_summary=True,
    )
    result2 = minimum_room_assignment(
        intervals,
        operation_summary=True,
    )

    assert result1 == result2

    print("All tests passed.")


if __name__ == "__main__":
    run_tests()
```
