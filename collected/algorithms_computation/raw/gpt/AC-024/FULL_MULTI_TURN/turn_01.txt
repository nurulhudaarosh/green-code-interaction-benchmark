```python
"""
Minimum Room Assignment

Problem:
Assign every interval to a room so that overlapping intervals never share
the same room. A room can be reused when its previous interval ends at or
before the next interval starts. The goal is to minimize the number of rooms
and return the assigned room ID for every original interval.

Key constraints:
- Each interval is represented as (start, end).
- Intervals use half-open semantics [start, end).
- Therefore, an interval ending at time t can share a room with an interval
  starting at time t.
- Every interval must receive exactly one room.
- Overlapping intervals must never use the same room.
- The number of rooms must be minimal.
- Room IDs are assigned deterministically.
- No randomness, network access, APIs, external services, or user interaction.
- Standard library only.

Required output:
- A list `room_ids` where `room_ids[i]` is the room assigned to the
  original interval `i`.
- The minimum number of rooms used.
- Room IDs are deterministic and start from 0.

Algorithm:
1. Sort intervals by `(start, end, original_index)` so processing is
   deterministic.
2. Maintain a min-heap of currently occupied rooms:
      (availability_time, room_id)
   The room with the earliest availability is considered first.
3. Before assigning an interval [start, end], repeatedly reuse available
   rooms whose availability time is <= start.
4. If the earliest available room can be reused, assign that room.
5. Otherwise, create a new room with the next room ID.
6. Push the selected room back with its new availability time `end`.
7. Store the assignment using the original interval index.

Why this is optimal:
At each interval, if no existing room is available, every existing room
overlaps the new interval, so a new room is unavoidable. Thus the number of
rooms created equals the maximum number of simultaneously overlapping
intervals.

Complexity:
- Sorting: O(n log n)
- Each interval performs heap operations: O(log n)
- Total: O(n log n)
- Space: O(n)
"""

from heapq import heappop, heappush
from typing import List, Sequence, Tuple


def minimum_room_assignment(
    intervals: Sequence[Tuple[int, int]]
) -> Tuple[int, List[int]]:
    """
    Assign every interval to a minimum number of rooms.

    Parameters:
        intervals:
            A sequence of (start, end) pairs using half-open intervals [start, end).

    Returns:
        (room_count, room_ids)
            room_count: minimum number of rooms required.
            room_ids[i]: room assigned to intervals[i].

    Raises:
        ValueError: if an interval has start > end.
    """
    n = len(intervals)

    if n == 0:
        return 0, []

    # Validate intervals.
    for start, end in intervals:
        if start > end:
            raise ValueError("Interval start cannot be greater than its end.")

    # (start, end, original_index)
    ordered = sorted(
        (start, end, index)
        for index, (start, end) in enumerate(intervals)
    )

    # Heap entries:
    # (availability_time, room_id)
    occupied = []

    room_ids = [-1] * n
    next_room_id = 0

    for start, end, original_index in ordered:
        # Reuse every room that is already free.
        # Because occupied is a min-heap, the first entry has the earliest
        # availability time.
        if occupied and occupied[0][0] <= start:
            availability, room_id = heappop(occupied)

            # Keep deterministic behavior if multiple rooms are available:
            # the heap orders them by availability time and then room ID.
            room_ids[original_index] = room_id
        else:
            room_id = next_room_id
            next_room_id += 1
            room_ids[original_index] = room_id

        # The selected room becomes available after this interval ends.
        heappush(occupied, (end, room_id))

    return next_room_id, room_ids


# -------------------------
# Tests
# -------------------------

def run_tests() -> None:
    # No intervals.
    rooms, assignments = minimum_room_assignment([])
    assert rooms == 0
    assert assignments == []

    # One interval.
    rooms, assignments = minimum_room_assignment([(1, 5)])
    assert rooms == 1
    assert assignments == [0]

    # Non-overlapping intervals reuse the same room.
    rooms, assignments = minimum_room_assignment(
        [(1, 3), (3, 5), (5, 8)]
    )
    assert rooms == 1
    assert assignments == [0, 0, 0]

    # Fully overlapping intervals require three rooms.
    rooms, assignments = minimum_room_assignment(
        [(1, 5), (2, 6), (3, 7)]
    )
    assert rooms == 3
    assert assignments == [0, 1, 2]

    # Original order must be preserved in the returned assignment list.
    rooms, assignments = minimum_room_assignment(
        [(5, 7), (1, 3), (3, 5), (2, 6)]
    )
    assert rooms == 2
    assert len(assignments) == 4

    # Verify no overlapping intervals share a room.
    intervals = [(5, 7), (1, 3), (3, 5), (2, 6)]
    for i in range(len(intervals)):
        for j in range(i + 1, len(intervals)):
            s1, e1 = intervals[i]
            s2, e2 = intervals[j]

            overlap = max(s1, s2) < min(e1, e2)
            if overlap:
                assert assignments[i] != assignments[j]

    # Equal start times require separate rooms.
    rooms, assignments = minimum_room_assignment(
        [(1, 4), (1, 4), (1, 4)]
    )
    assert rooms == 3
    assert assignments == [0, 1, 2]

    # End-at-start is reusable because intervals are half-open.
    rooms, assignments = minimum_room_assignment(
        [(1, 2), (2, 3), (3, 4)]
    )
    assert rooms == 1
    assert assignments == [0, 0, 0]

    # Zero-length intervals can be processed deterministically.
    rooms, assignments = minimum_room_assignment(
        [(2, 2), (2, 2), (2, 3)]
    )
    assert rooms == 1
    assert assignments == [0, 0, 0]

    # Negative times are valid as long as start <= end.
    rooms, assignments = minimum_room_assignment(
        [(-5, -2), (-4, 0), (-2, 3)]
    )
    assert rooms == 2
    assert len(assignments) == 3

    # Invalid interval.
    try:
        minimum_room_assignment([(5, 2)])
        assert False, "Expected ValueError"
    except ValueError:
        pass


if __name__ == "__main__":
    run_tests()

    example = [
        (1, 4),
        (2, 5),
        (4, 7),
        (6, 8),
    ]

    room_count, room_ids = minimum_room_assignment(example)

    print("Room count:", room_count)
    print("Room IDs:", room_ids)
```
