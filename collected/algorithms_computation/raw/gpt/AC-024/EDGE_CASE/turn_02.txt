```python
"""
Minimum Room Assignment

Problem:
Assign every interval to a room so that overlapping intervals never share
the same room. A room can be reused when its previous interval ends at or
before the next interval's start.

The goal is to minimize the number of rooms and return the room ID assigned
to every interval in the original input order.

Key constraints:
- Input intervals are (start, end).
- A room is reusable when previous_end <= next_start.
- Intervals may be provided in arbitrary order.
- Room IDs start at 0.
- Results must be deterministic.
- For equal start times, processing is deterministic by:
      (start, end, original_index)
- When multiple rooms are available, the smallest room ID is reused.
- Only the Python standard library is used.
- No network access, APIs, external services, randomness, or human
  interaction.

Required output:
{
    "room_count": <minimum number of rooms>,
    "room_ids": [<room ID for every original interval>]
}

Algorithm:
1. Sort all intervals by (start, end, original_index).
2. Maintain a min-heap of busy rooms:
       (available_time, room_id)
3. Before assigning an interval, move every room with
       available_time <= start
   from the busy heap into a free-room min-heap.
4. If a free room exists, reuse the smallest room ID.
5. Otherwise, create a new room.
6. Store the selected room ID at the interval's original index.

Why it is optimal:
A new room is needed only when every existing room overlaps the current
interval. Thus, the number of rooms created equals the maximum number of
simultaneously overlapping intervals.

Complexity:
- Sorting: O(n log n)
- Heap operations: O(n log n)
- Total: O(n log n)
- Space: O(n)

The tests below explicitly include worst-case-like structures for this
complexity: many intervals with distinct start times and heavy overlap,
forcing substantial sorting and heap activity while remaining valid input.
"""

from heapq import heappop, heappush
from typing import Any, Dict, Sequence, Tuple


def minimum_room_assignment(
    intervals: Sequence[Tuple[int, int]]
) -> Dict[str, Any]:
    """
    Assign every interval to a minimum number of rooms.

    Returns:
        {
            "room_count": minimum number of rooms,
            "room_ids": room ID for each interval in original order
        }
    """

    n = len(intervals)

    if n == 0:
        return {
            "room_count": 0,
            "room_ids": []
        }

    jobs = []

    for index, (start, end) in enumerate(intervals):
        if end < start:
            raise ValueError(
                f"Invalid interval at index {index}: end must be >= start."
            )

        jobs.append((start, end, index))

    # Deterministic chronological ordering.
    jobs.sort(key=lambda item: (item[0], item[1], item[2]))

    # (available_time, room_id)
    busy_rooms = []

    # Available room IDs.
    free_rooms = []

    room_ids = [-1] * n
    next_room_id = 0

    for start, end, original_index in jobs:
        # A room ending exactly when this interval starts is reusable.
        while busy_rooms and busy_rooms[0][0] <= start:
            available_time, room_id = heappop(busy_rooms)
            heappush(free_rooms, room_id)

        # Always reuse the smallest available room ID.
        if free_rooms:
            room_id = heappop(free_rooms)
        else:
            room_id = next_room_id
            next_room_id += 1

        room_ids[original_index] = room_id

        # This room remains occupied until 'end'.
        heappush(busy_rooms, (end, room_id))

    return {
        "room_count": next_room_id,
        "room_ids": room_ids
    }


# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------

def run_tests() -> None:
    # ---------------------------------------------------------------
    # Basic empty input
    # ---------------------------------------------------------------
    assert minimum_room_assignment([]) == {
        "room_count": 0,
        "room_ids": []
    }

    # ---------------------------------------------------------------
    # No overlap: one room is enough.
    # End == next start means reuse is allowed.
    # ---------------------------------------------------------------
    intervals = [(1, 3), (3, 5), (5, 8)]

    assert minimum_room_assignment(intervals) == {
        "room_count": 1,
        "room_ids": [0, 0, 0]
    }

    # ---------------------------------------------------------------
    # Complete overlap: every interval needs a different room.
    # ---------------------------------------------------------------
    intervals = [(1, 5), (2, 6), (3, 7)]

    assert minimum_room_assignment(intervals) == {
        "room_count": 3,
        "room_ids": [0, 1, 2]
    }

    # ---------------------------------------------------------------
    # Arbitrary input order.
    # Room IDs must still be returned in original input order.
    # ---------------------------------------------------------------
    intervals = [(5, 7), (1, 4), (3, 6)]

    assert minimum_room_assignment(intervals) == {
        "room_count": 2,
        "room_ids": [1, 0, 0]
    }

    # ---------------------------------------------------------------
    # Exact boundary reuse.
    # ---------------------------------------------------------------
    intervals = [(1, 2), (2, 3), (2, 4)]

    assert minimum_room_assignment(intervals) == {
        "room_count": 2,
        "room_ids": [0, 0, 1]
    }

    # ---------------------------------------------------------------
    # Same start time.
    # Sorting must be deterministic.
    # ---------------------------------------------------------------
    intervals = [(1, 4), (1, 3), (1, 2)]

    assert minimum_room_assignment(intervals) == {
        "room_count": 3,
        "room_ids": [2, 1, 0]
    }

    # ---------------------------------------------------------------
    # Zero-length intervals.
    # They can immediately release their room.
    # ---------------------------------------------------------------
    intervals = [(2, 2), (2, 2), (2, 3)]

    assert minimum_room_assignment(intervals) == {
        "room_count": 1,
        "room_ids": [0, 0, 0]
    }

    # ---------------------------------------------------------------
    # Deterministic smallest-room reuse.
    #
    # At start=5, rooms 0 and 1 are both available.
    # The smallest room ID (0) must be selected.
    # ---------------------------------------------------------------
    intervals = [
        (1, 5),   # room 0
        (2, 5),   # room 1
        (5, 6),   # must reuse room 0
    ]

    assert minimum_room_assignment(intervals) == {
        "room_count": 2,
        "room_ids": [0, 1, 0]
    }

    # ---------------------------------------------------------------
    # Worst-case-like structure for O(n log n):
    #
    # Many intervals have different start times but all remain active
    # for a long time. This forces:
    # - sorting of many intervals,
    # - many heap insertions,
    # - many heap comparisons,
    # - and a large active-room heap.
    #
    # Since every interval overlaps all later intervals, n rooms
    # are required.
    # ---------------------------------------------------------------
    n = 1000

    worst_case = [
        (i, n + i + 1)
        for i in range(n)
    ]

    result = minimum_room_assignment(worst_case)

    assert result["room_count"] == n
    assert result["room_ids"] == list(range(n))

    # ---------------------------------------------------------------
    # Worst-case-like input in reverse order.
    #
    # This additionally stresses the sorting step because the input
    # starts in descending order while all intervals heavily overlap.
    # ---------------------------------------------------------------
    reverse_worst_case = [
        (i, 2 * n)
        for i in range(n - 1, -1, -1)
    ]

    result = minimum_room_assignment(reverse_worst_case)

    assert result["room_count"] == n

    # Original intervals are reversed, so their assigned room IDs are
    # also reversed relative to chronological processing.
    assert result["room_ids"] == list(range(n - 1, -1, -1))

    # ---------------------------------------------------------------
    # Large number of reusable rooms.
    #
    # First create many simultaneously active rooms. Then end all of
    # them at the same time. The next interval must deterministically
    # reuse the smallest room ID.
    # ---------------------------------------------------------------
    room_count = 200

    intervals = []

    for i in range(room_count):
        intervals.append((i, room_count + 1))

    # All rooms become available before this interval.
    intervals.append((room_count + 1, room_count + 2))

    result = minimum_room_assignment(intervals)

    assert result["room_count"] == room_count
    assert result["room_ids"][-1] == 0

    # ---------------------------------------------------------------
    # Many repeated starts with deterministic tie handling.
    # ---------------------------------------------------------------
    repeated_start = [
        (10, 30),
        (10, 20),
        (10, 10),
        (10, 25),
        (10, 15),
    ]

    result = minimum_room_assignment(repeated_start)

    assert result["room_count"] == 4
    assert result["room_ids"] == [3, 2, 0, 1, 0]

    # ---------------------------------------------------------------
    # Invalid interval must still be rejected.
    # ---------------------------------------------------------------
    try:
        minimum_room_assignment([(5, 2)])
        assert False, "Expected ValueError"
    except ValueError:
        pass


if __name__ == "__main__":
    run_tests()
    print("All tests passed.")
```
