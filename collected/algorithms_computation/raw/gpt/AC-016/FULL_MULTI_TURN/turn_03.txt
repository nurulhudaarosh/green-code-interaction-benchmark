# Priority Queue Simulator
#
# Original problem:
# Simulate a non-preemptive single-server queue.
#
# Each job is represented as:
#     (job_id, arrival_time, service_time, priority)
#
# Expected behavior:
# 1. Sort jobs by arrival time.
# 2. When the server becomes free, consider every job that has already arrived.
# 3. Select the job with the largest priority.
# 4. If priorities tie, select the job with the earlier arrival time.
# 5. If priority and arrival time both tie, select the smaller job ID.
# 6. The selected job runs until completion; it cannot be preempted.
# 7. If no job is available, advance the server time to the next arrival.
#
# Original required output:
# {
#     "completion_times": {job_id: completion_time, ...},
#     "waiting_times": {job_id: waiting_time, ...},
#     "average_waiting_time": float
# }
#
# New optional feature:
# When include_operation_summary=True, return one additional field:
#
#     "operation_summary"
#
# The summary is deterministic and reports the major operations performed:
#   - sort_operations: 1 when sorting a non-empty input, otherwise 0
#   - heap_pushes: number of jobs inserted into the priority heap
#   - heap_pops: number of jobs selected from the priority heap
#   - idle_jumps: number of times the server had to jump forward to a
#                 future arrival because the heap was empty
#   - total_major_operations: sum of the above operations
#
# When include_operation_summary=False (the default), the original output
# remains unchanged.
#
# Algorithm:
# - Sort arrivals by (arrival_time, job_id).
# - Use a min-heap with the key:
#       (-priority, arrival_time, job_id, service_time)
#   so that the heap exactly enforces:
#       larger priority -> earlier arrival -> smaller job ID.
# - Add all currently arrived jobs to the heap.
# - Pop the highest-priority job and execute it completely.
#
# Complexity:
# - Sorting: O(n log n)
# - Heap insertion/removal: O(n log n)
# - Overall: O(n log n)
# - Extra space: O(n)
#
# Standard library only. No network, APIs, external services,
# randomness, or human interaction.


from heapq import heappush, heappop


def simulate_priority_queue(jobs, include_operation_summary=False):
    """
    Simulate a non-preemptive single-server priority queue.

    Parameters:
        jobs:
            Iterable of tuples:
            (job_id, arrival_time, service_time, priority)

        include_operation_summary:
            If False, return exactly the original required fields.
            If True, add the deterministic operation_summary field.

    Returns:
        Original format when include_operation_summary=False:
        {
            "completion_times": {...},
            "waiting_times": {...},
            "average_waiting_time": float
        }

        Extended format when include_operation_summary=True:
        {
            "completion_times": {...},
            "waiting_times": {...},
            "average_waiting_time": float,
            "operation_summary": {
                "sort_operations": int,
                "heap_pushes": int,
                "heap_pops": int,
                "idle_jumps": int,
                "total_major_operations": int
            }
        }
    """

    jobs = list(jobs)

    # Deterministic operation counters.
    sort_operations = 0
    heap_pushes = 0
    heap_pops = 0
    idle_jumps = 0

    if not jobs:
        result = {
            "completion_times": {},
            "waiting_times": {},
            "average_waiting_time": 0.0,
        }

        if include_operation_summary:
            result["operation_summary"] = {
                "sort_operations": 0,
                "heap_pushes": 0,
                "heap_pops": 0,
                "idle_jumps": 0,
                "total_major_operations": 0,
            }

        return result

    # Sort arrivals first.
    arrivals = sorted(jobs, key=lambda job: (job[1], job[0]))
    sort_operations = 1

    completion_times = {}
    waiting_times = {}

    heap = []

    current_time = 0
    next_index = 0
    total_waiting_time = 0

    while next_index < len(arrivals) or heap:

        # If there are no available jobs, jump to the next arrival.
        if not heap and next_index < len(arrivals):
            if current_time < arrivals[next_index][1]:
                current_time = arrivals[next_index][1]
                idle_jumps += 1
            else:
                current_time = max(
                    current_time,
                    arrivals[next_index][1]
                )

        # Add every job that has arrived.
        while (
            next_index < len(arrivals)
            and arrivals[next_index][1] <= current_time
        ):
            job_id, arrival_time, service_time, priority = arrivals[next_index]

            # Heap ordering:
            #   -priority       -> larger priority first
            #   arrival_time    -> earlier arrival first
            #   job_id          -> smaller ID first
            heappush(
                heap,
                (
                    -priority,
                    arrival_time,
                    job_id,
                    service_time,
                ),
            )

            heap_pushes += 1
            next_index += 1

        # Select the next job according to the required rules.
        _, arrival_time, job_id, service_time = heappop(heap)
        heap_pops += 1

        start_time = current_time
        waiting_time = start_time - arrival_time

        # Non-preemptive execution.
        current_time += service_time
        completion_time = current_time

        completion_times[job_id] = completion_time
        waiting_times[job_id] = waiting_time
        total_waiting_time += waiting_time

    average_waiting_time = total_waiting_time / len(jobs)

    result = {
        "completion_times": completion_times,
        "waiting_times": waiting_times,
        "average_waiting_time": average_waiting_time,
    }

    # Preserve the original result exactly unless the feature is requested.
    if include_operation_summary:
        total_major_operations = (
            sort_operations
            + heap_pushes
            + heap_pops
            + idle_jumps
        )

        result["operation_summary"] = {
            "sort_operations": sort_operations,
            "heap_pushes": heap_pushes,
            "heap_pops": heap_pops,
            "idle_jumps": idle_jumps,
            "total_major_operations": total_major_operations,
        }

    return result

# Tests
# ------------------------------------------------------------

def test_original_output_is_unchanged_when_feature_disabled():
    jobs = [
        (1, 0, 5, 1),
        (2, 1, 3, 5),
        (3, 2, 2, 3),
    ]

    result = simulate_priority_queue(jobs)

    assert result == {
        "completion_times": {
            1: 5,
            2: 8,
            3: 10,
        },
        "waiting_times": {
            1: 0,
            2: 4,
            3: 6,
        },
        "average_waiting_time": 10 / 3,
    }

    # No extra field is present when the feature is disabled.
    assert "operation_summary" not in result


def test_operation_summary():
    jobs = [
        (1, 0, 5, 1),
        (2, 1, 3, 5),
        (3, 2, 2, 3),
    ]

    result = simulate_priority_queue(
        jobs,
        include_operation_summary=True,
    )

    # Original fields remain unchanged.
    assert result["completion_times"] == {
        1: 5,
        2: 8,
        3: 10,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 4,
        3: 6,
    }

    assert result["average_waiting_time"] == 10 / 3

    # Deterministic operation counts:
    # one sort, three pushes, three pops, zero idle jumps.
    assert result["operation_summary"] == {
        "sort_operations": 1,
        "heap_pushes": 3,
        "heap_pops": 3,
        "idle_jumps": 0,
        "total_major_operations": 7,
    }


def test_priority_tie_uses_earlier_arrival():
    jobs = [
        (1, 0, 1, 1),
        (5, 1, 1, 5),
        (2, 2, 1, 5),
    ]

    result = simulate_priority_queue(
        jobs,
        include_operation_summary=True,
    )

    # Job 5 arrived before job 2 and has the same priority.
    assert result["completion_times"] == {
        1: 1,
        5: 2,
        2: 3,
    }

    assert result["waiting_times"] == {
        1: 0,
        5: 0,
        2: 0,
    }

    assert result["average_waiting_time"] == 0.0

    assert result["operation_summary"] == {
        "sort_operations": 1,
        "heap_pushes": 3,
        "heap_pops": 3,
        "idle_jumps": 0,
        "total_major_operations": 7,
    }


def test_same_arrival_and_priority_uses_smaller_job_id():
    jobs = [
        (3, 0, 1, 1),
        (5, 1, 2, 5),
        (2, 1, 1, 5),
    ]

    result = simulate_priority_queue(jobs)

    # Same priority and arrival:
    # smaller job ID (2) runs before job 5.
    assert result["completion_times"] == {
        3: 1,
        2: 2,
        5: 4,
    }

    assert result["waiting_times"] == {
        3: 0,
        2: 0,
        5: 1,
    }

    assert result["average_waiting_time"] == 1 / 3


def test_non_preemptive_behavior():
    jobs = [
        (1, 0, 10, 1),
        (2, 2, 1, 100),
    ]

    result = simulate_priority_queue(jobs)

    # Job 2 has higher priority but cannot interrupt job 1.
    assert result["completion_times"] == {
        1: 10,
        2: 11,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 8,
    }


def test_idle_jumps_are_counted():
    jobs = [
        (1, 5, 3, 2),
        (2, 10, 2, 5),
    ]

    result = simulate_priority_queue(
        jobs,
        include_operation_summary=True,
    )

    assert result["completion_times"] == {
        1: 8,
        2: 12,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 0,
    }

    assert result["average_waiting_time"] == 0.0

    # The server jumps from time 0 -> 5 and then 8 -> 10.
    assert result["operation_summary"] == {
        "sort_operations": 1,
        "heap_pushes": 2,
        "heap_pops": 2,
        "idle_jumps": 2,
        "total_major_operations": 7,
    }


def test_empty_input():
    result = simulate_priority_queue([])

    assert result == {
        "completion_times": {},
        "waiting_times": {},
        "average_waiting_time": 0.0,
    }


def test_empty_input_with_summary():
    result = simulate_priority_queue(
        [],
        include_operation_summary=True,
    )

    assert result == {
        "completion_times": {},
        "waiting_times": {},
        "average_waiting_time": 0.0,
        "operation_summary": {
            "sort_operations": 0,
            "heap_pushes": 0,
            "heap_pops": 0,
            "idle_jumps": 0,
            "total_major_operations": 0,
        },
    }


if __name__ == "__main__":
    test_original_output_is_unchanged_when_feature_disabled()
    test_operation_summary()
    test_priority_tie_uses_earlier_arrival()
    test_same_arrival_and_priority_uses_smaller_job_id()
    test_non_preemptive_behavior()
    test_idle_jumps_are_counted()
    test_empty_input()
    test_empty_input_with_summary()

    print("All tests passed.")
