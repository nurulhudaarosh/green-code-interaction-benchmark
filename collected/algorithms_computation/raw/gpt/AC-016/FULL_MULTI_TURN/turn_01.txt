# Priority Queue Simulator
#
# Problem restatement:
# Simulate a non-preemptive single-server queue. Each job has:
#   - arrival time
#   - service time
#   - priority
#   - job ID
#
# When the server becomes free:
#   1. Among all arrived jobs, choose the largest priority.
#   2. If priorities tie, choose the earlier arrival time.
#   3. If arrival times also tie, choose the smaller job ID.
#
# Once a job starts, it runs until completion (non-preemptive).
#
# Required output:
#   - completion time for every job
#   - waiting time for every job
#   - average waiting time
#
# Key constraints:
#   - Deterministic tie handling is required.
#   - Jobs must be processed according to arrival time.
#   - The server must never preempt a running job.
#   - If no job is currently available, the server jumps to the next arrival.
#   - Only the Python standard library is used.
#   - No network, API, external service, randomness, or human interaction.
#
# Algorithm:
#   1. Sort jobs by (arrival_time, job_id).
#   2. Keep unprocessed jobs in that sorted order.
#   3. Use a priority heap for all jobs that have already arrived.
#   4. Python's min-heap is converted into the required priority order by
#      storing:
#          (-priority, arrival_time, job_id, ...)
#      so that larger priority is selected first.
#   5. If the heap is empty, move the current time to the next arrival.
#   6. Otherwise, remove the highest-priority job, run it to completion,
#      and record completion and waiting time.
#   7. Average waiting time = total waiting time / number of jobs.
#
# Complexity:
#   Sorting: O(n log n)
#   Heap operations: O(n log n)
#   Overall: O(n log n)
#   Extra space: O(n)


from heapq import heappush, heappop


def simulate_priority_queue(jobs):
    """
    Simulate a non-preemptive single-server priority queue.

    Input:
        jobs: iterable of tuples
              (job_id, arrival_time, service_time, priority)

    Returns:
        {
            "completion_times": {job_id: completion_time, ...},
            "waiting_times": {job_id: waiting_time, ...},
            "average_waiting_time": float
        }

    Tie-breaking:
        1. Larger priority
        2. Earlier arrival time
        3. Smaller job ID
    """

    jobs = list(jobs)

    if not jobs:
        return {
            "completion_times": {},
            "waiting_times": {},
            "average_waiting_time": 0.0,
        }

    # Sort arrivals first.
    arrivals = sorted(jobs, key=lambda job: (job[1], job[0]))

    completion_times = {}
    waiting_times = {}

    heap = []
    current_time = 0
    next_index = 0
    total_waiting_time = 0

    while next_index < len(arrivals) or heap:

        # If no job is available, jump to the next arrival.
        if not heap and next_index < len(arrivals):
            current_time = max(current_time, arrivals[next_index][1])

        # Add every job that has arrived by current_time.
        while (
            next_index < len(arrivals)
            and arrivals[next_index][1] <= current_time
        ):
            job_id, arrival_time, service_time, priority = arrivals[next_index]

            # Negative priority gives larger priority first.
            heappush(
                heap,
                (
                    -priority,
                    arrival_time,
                    job_id,
                    service_time,
                ),
            )

            next_index += 1

        # Select the highest-priority available job.
        _, arrival_time, job_id, service_time = heappop(heap)

        # Non-preemptive job starts immediately.
        start_time = current_time
        waiting_time = start_time - arrival_time

        current_time += service_time
        completion_time = current_time

        completion_times[job_id] = completion_time
        waiting_times[job_id] = waiting_time
        total_waiting_time += waiting_time

    average_waiting_time = total_waiting_time / len(jobs)

    return {
        "completion_times": completion_times,
        "waiting_times": waiting_times,
        "average_waiting_time": average_waiting_time,
    }


# ------------------------------------------------------------
# Tests
# ------------------------------------------------------------

def test_basic_priority():
    jobs = [
        (1, 0, 5, 1),
        (2, 1, 3, 5),
        (3, 2, 2, 3),
    ]

    result = simulate_priority_queue(jobs)

    assert result["completion_times"] == {
        1: 5,
        2: 8,
        3: 10,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 4,
        3: 6,
    }

    assert result["average_waiting_time"] == 10 / 3


def test_priority_tie_uses_earlier_arrival():
    jobs = [
        (1, 0, 4, 5),
        (2, 1, 2, 5),
        (3, 2, 1, 5),
    ]

    result = simulate_priority_queue(jobs)

    # Job 1 runs first.
    # Then job 2 arrived before job 3.
    assert result["completion_times"] == {
        1: 4,
        2: 6,
        3: 7,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 3,
        3: 4,
    }


def test_same_arrival_and_priority_uses_smaller_job_id():
    jobs = [
        (3, 0, 2, 10),
        (1, 0, 3, 10),
        (2, 0, 1, 10),
    ]

    result = simulate_priority_queue(jobs)

    # Same arrival and priority:
    # job ID 1 -> job ID 2 -> job ID 3
    assert result["completion_times"] == {
        1: 3,
        2: 4,
        3: 6,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 3,
        3: 4,
    }


def test_server_waits_for_first_arrival():
    jobs = [
        (1, 5, 3, 2),
        (2, 10, 2, 5),
    ]

    result = simulate_priority_queue(jobs)

    assert result["completion_times"] == {
        1: 8,
        2: 12,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 0,
    }

    assert result["average_waiting_time"] == 0.0


def test_later_high_priority_job_does_not_preempt():
    jobs = [
        (1, 0, 10, 1),
        (2, 2, 1, 100),
    ]

    result = simulate_priority_queue(jobs)

    # Job 2 has higher priority but cannot interrupt job 1.
    assert result["completion_times"] == {
        1: 10,
        2: 11,
    }

    assert result["waiting_times"] == {
        1: 0,
        2: 8,
    }


def test_empty_input():
    result = simulate_priority_queue([])

    assert result == {
        "completion_times": {},
        "waiting_times": {},
        "average_waiting_time": 0.0,
    }


if __name__ == "__main__":
    test_basic_priority()
    test_priority_tie_uses_earlier_arrival()
    test_same_arrival_and_priority_uses_smaller_job_id()
    test_server_waits_for_first_arrival()
    test_later_high_priority_job_does_not_preempt()
    test_empty_input()

    print("All tests passed.")
