```python
"""
Minimum-Cost Grid Path

Problem:
Given a non-negative rectangular cost matrix, move only right or down
from the top-left cell (0, 0) to the bottom-right cell.

Expected behavior:
- Return the minimum total path cost.
- Return one deterministic optimal path as (row, column) coordinates.
- Use dynamic programming with predecessor coordinates.
- Deterministic tie handling: when top and left predecessors have equal
  cost, choose the TOP predecessor.

Bug demonstration:
For:
    [
        [1, 1],
        [1, 1]
    ]

Both paths have cost 3:
    Top path:  (0,0) -> (0,1) -> (1,1)
    Left path: (0,0) -> (1,0) -> (1,1)

The required deterministic rule chooses the top predecessor.

A buggy implementation using:
    if from_top < from_left:
would choose the left predecessor on a tie.

The corrected implementation uses:
    if from_top <= from_left:
so ties always choose the top predecessor.
"""

from typing import List, Tuple


def minimum_cost_grid_path(
    cost: List[List[int]]
) -> Tuple[int, List[Tuple[int, int]]]:
    """
    Return the minimum cost and one deterministic optimal path.

    Allowed moves:
        - Right
        - Down

    Tie-breaking:
        If the top and left predecessors have equal cost,
        choose the top predecessor.
    """

    # Validate input
    if not cost:
        raise ValueError("Cost matrix must not be empty.")

    if not cost[0]:
        raise ValueError("Cost matrix must not contain an empty row.")

    cols = len(cost[0])

    if any(len(row) != cols for row in cost):
        raise ValueError("Cost matrix must be rectangular.")

    if any(value < 0 for row in cost for value in row):
        raise ValueError("All costs must be non-negative.")

    rows = len(cost)

    # dp[r][c] = minimum cost to reach (r, c)
    dp = [[0] * cols for _ in range(rows)]

    # parent[r][c] = predecessor coordinate used to reach (r, c)
    parent = [[None] * cols for _ in range(rows)]

    # Starting cell
    dp[0][0] = cost[0][0]

    # First column: only possible move is down
    for r in range(1, rows):
        dp[r][0] = dp[r - 1][0] + cost[r][0]
        parent[r][0] = (r - 1, 0)

    # First row: only possible move is right
    for c in range(1, cols):
        dp[0][c] = dp[0][c - 1] + cost[0][c]
        parent[0][c] = (0, c - 1)

    # Fill DP table
    for r in range(1, rows):
        for c in range(1, cols):
            from_top = dp[r - 1][c]
            from_left = dp[r][c - 1]

            # Deterministic tie handling:
            # <= means ties choose the top predecessor.
            if from_top <= from_left:
                dp[r][c] = from_top + cost[r][c]
                parent[r][c] = (r - 1, c)
            else:
                dp[r][c] = from_left + cost[r][c]
                parent[r][c] = (r, c - 1)

    # Reconstruct path using predecessor coordinates
    path = []
    r, c = rows - 1, cols - 1

    while True:
        path.append((r, c))

        if (r, c) == (0, 0):
            break

        r, c = parent[r][c]

    path.reverse()

    return dp[rows - 1][cols - 1], path


# ---------------------------------------------------------
# Demonstration of the tie-breaking bug and its correction
# ---------------------------------------------------------

if __name__ == "__main__":
    cost = [
        [1, 1],
        [1, 1]
    ]

    minimum_cost, path = minimum_cost_grid_path(cost)

    print("Minimum cost:", minimum_cost)
    print("Path:", path)

    # Expected output:
    #
    # Minimum cost: 3
    # Path: [(0, 0), (0, 1), (1, 1)]
    #
    # Both possible paths have cost 3, but the deterministic
    # tie-breaking rule selects the path through the top predecessor.
```
