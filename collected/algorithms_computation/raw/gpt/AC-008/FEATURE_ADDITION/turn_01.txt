## Problem Understanding

We need to maintain an integer array while processing two types of operations:

1. **Point replacement:** Replace the value at a specific index.
2. **Inclusive range-sum query:** Find the sum of all elements from index `left` through index `right`, including both endpoints.

Every range-sum query result must be returned **in the same order as the queries appear**.

### Key Constraints

* The array contains integers.
* Updates change **one position at a time**.
* Range queries are **inclusive**: `[left, right]`.
* Results must be deterministic.
* A Fenwick tree gives:

  * Point update: `O(log n)`
  * Prefix sum: `O(log n)`
  * Range sum: `O(log n)`
* Overall complexity: **O((n + q) log n)**, where `q` is the number of operations.
* Only Python's **standard library** is used.

### Required Output

Return a list containing the result of every range-sum query, in their original order.

### Algorithm

A Fenwick tree stores cumulative information efficiently.

For a replacement:

```text
difference = new_value - old_value
Fenwick.update(index, difference)
```

For a query `[left, right]`:

```text
sum(left, right) = prefix_sum(right) - prefix_sum(left - 1)
```

Here is a deterministic implementation:

```python
from typing import List, Sequence, Tuple


Operation = Tuple[str, int, int]


class FenwickTree:
    """Fenwick tree supporting point additions and prefix sums."""

    def __init__(self, values: Sequence[int]) -> None:
        self.n = len(values)
        self.tree = [0] * (self.n + 1)

        # Build the tree in O(n).
        for i, value in enumerate(values, start=1):
            self.tree[i] += value
            parent = i + (i & -i)
            if parent <= self.n:
                self.tree[parent] += self.tree[i]

    def add(self, index: int, delta: int) -> None:
        """Add delta to the value at zero-based index."""
        if not 0 <= index < self.n:
            raise IndexError("array index out of range")

        i = index + 1
        while i <= self.n:
            self.tree[i] += delta
            i += i & -i

    def prefix_sum(self, end: int) -> int:
        """
        Return the sum of values in [0, end).

        end may be from 0 through n.
        """
        if not 0 <= end <= self.n:
            raise IndexError("prefix endpoint out of range")

        total = 0
        i = end
        while i > 0:
            total += self.tree[i]
            i -= i & -i

        return total

    def range_sum(self, left: int, right: int) -> int:
        """Return the inclusive sum of values in [left, right]."""
        if not (0 <= left <= right < self.n):
            raise IndexError("invalid range")

        return self.prefix_sum(right + 1) - self.prefix_sum(left)


def process_operations(
    initial: Sequence[int],
    operations: Sequence[Operation],
) -> List[int]:
    """
    Maintain an integer array under point replacements and
    inclusive range-sum queries.

    Operation format:
        ("update", index, new_value)
        ("query", left, right)

    Returns:
        The result of every query, in input order.
    """
    values = list(initial)
    fenwick = FenwickTree(values)
    results: List[int] = []

    for operation in operations:
        if len(operation) != 3:
            raise ValueError("each operation must contain exactly 3 values")

        op, a, b = operation

        if op == "update":
            index = a
            new_value = b

            if not 0 <= index < len(values):
                raise IndexError("array index out of range")

            # Convert replacement into a Fenwick point addition.
            delta = new_value - values[index]
            values[index] = new_value
            fenwick.add(index, delta)

        elif op == "query":
            left = a
            right = b

            results.append(fenwick.range_sum(left, right))

        else:
            raise ValueError(f"unknown operation: {op!r}")

    return results


# Example
if __name__ == "__main__":
    initial = [1, 2, 3, 4, 5]

    operations = [
        ("query", 1, 3),   # 2 + 3 + 4 = 9
        ("update", 2, 10), # [1, 2, 10, 4, 5]
        ("query", 1, 3),   # 2 + 10 + 4 = 16
        ("query", 0, 4),   # 1 + 2 + 10 + 4 + 5 = 22
    ]

    print(process_operations(initial, operations))
    # Output: [9, 16, 22]
```
