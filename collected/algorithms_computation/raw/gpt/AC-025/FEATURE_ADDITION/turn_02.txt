```python
"""
Multi-Stage Resource Allocation — Extended Specification

Original problem:
- There are multiple stages.
- Each stage requires exactly one option to be selected.
- Every option consumes some budget and memory and provides a score.
- The selected options must satisfy both global limits:
      total budget  <= budget_limit
      total memory  <= memory_limit
- The objective is to maximize total score.
- If multiple selections have the same maximum score, choose the
  lexicographically smallest choice vector.
- The solution must be deterministic.

Original required output:
{
    "max_score": maximum achievable total score,
    "total_budget": total budget used by the selected options,
    "total_memory": total memory used by the selected options,
    "choice_vector": zero-based selected option index for every stage
}

If no feasible selection exists:
{
    "max_score": None,
    "total_budget": None,
    "total_memory": None,
    "choice_vector": []
}

New feature:
- Add an optional `include_operation_summary` parameter.
- When it is True, return an additional:
      "operation_summary"
  field.
- `operation_summary` reports the number of major computational
  decisions/operations made by the layered two-resource DP.
- The count is deterministic.
- The counted operation is each feasible DP transition considered
  from a reachable state to an option in the next stage.
- Pruned transitions that exceed either resource limit are not counted
  as computational decisions because no DP state is produced from them.
- The summary is:
      {
          "major_operations": <integer>
      }

Backward compatibility:
- The default value of `include_operation_summary` is False.
- When the feature is disabled/not requested, the output contains
  exactly the original fields and no `operation_summary` field.
- All original optimization and tie-breaking requirements remain
  unchanged.

Algorithm:
- Use layered two-resource dynamic programming.
- A DP state is identified by:
      (used_budget, used_memory)
- For every reachable state in one layer, try every option in the
  next stage.
- For each resulting resource state, retain:
    1. the higher score;
    2. for equal scores, the lexicographically smaller choice vector.
- After processing all stages, choose the state with maximum score and
  apply the same deterministic lexicographic tie-breaking rule.
- Deterministic reconstruction is achieved by storing the choice vector
  with each DP state.
- `major_operations` is incremented once for every feasible transition
  considered.

Complexity:
- Time: O(S * B * M * O), where
    S = number of stages,
    B = budget_limit + 1,
    M = memory_limit + 1,
    O = maximum options in a stage.
- Space: O(B * M), excluding stored choice vectors.

Standard library only. No network access, APIs, external services,
randomness, or human interaction.
"""


def multi_stage_resource_allocation(
    stages,
    budget_limit,
    memory_limit,
    include_operation_summary=False,
):
    """
    Solve the Multi-Stage Resource Allocation problem.

    Parameters
    ----------
    stages : list[list[dict]]
        Each stage contains option dictionaries with:
            "budget"
            "memory"
            "score"

    budget_limit : int
        Maximum total budget.

    memory_limit : int
        Maximum total memory.

    include_operation_summary : bool, default=False
        If True, include the additional deterministic
        "operation_summary" field.

    Returns
    -------
    dict
        Original result fields, plus operation_summary when requested.
    """

    def make_result(
        max_score,
        total_budget,
        total_memory,
        choice_vector,
        operations,
    ):
        result = {
            "max_score": max_score,
            "total_budget": total_budget,
            "total_memory": total_memory,
            "choice_vector": choice_vector,
        }

        if include_operation_summary:
            result["operation_summary"] = {
                "major_operations": operations
            }

        return result

    # Invalid global limits make the problem infeasible.
    if budget_limit < 0 or memory_limit < 0:
        return make_result(None, None, None, [], 0)

    # No stages means there is nothing to select.
    if not stages:
        return make_result(0, 0, 0, [], 0)

    # Every stage requires exactly one option.
    # Therefore an empty stage makes the instance infeasible.
    if any(not stage for stage in stages):
        return make_result(None, None, None, [], 0)

    # dp[(budget, memory)] = (score, choice_vector)
    #
    # Only the current layer is retained.
    dp = {(0, 0): (0, ())}

    major_operations = 0

    for stage in stages:
        next_dp = {}

        for (used_budget, used_memory), (score, choices) in dp.items():
            for option_index, option in enumerate(stage):
                option_budget = option["budget"]
                option_memory = option["memory"]
                option_score = option["score"]

                new_budget = used_budget + option_budget
                new_memory = used_memory + option_memory

                # This transition cannot produce a feasible state.
                if new_budget > budget_limit:
                    continue

                if new_memory > memory_limit:
                    continue

                # One feasible DP transition is one major operation.
                major_operations += 1

                new_score = score + option_score
                new_choices = choices + (option_index,)

                state = (new_budget, new_memory)
                candidate = (new_score, new_choices)

                current = next_dp.get(state)

                if current is None:
                    next_dp[state] = candidate
                else:
                    current_score, current_choices = current

                    # Higher score wins.
                    # Equal score -> lexicographically smaller choice
                    # vector wins.
                    if (
                        new_score > current_score
                        or (
                            new_score == current_score
                            and new_choices < current_choices
                        )
                    ):
                        next_dp[state] = candidate

        # No reachable state means the remaining stages cannot be
        # completed within the resource limits.
        if not next_dp:
            return make_result(
                None,
                None,
                None,
                [],
                major_operations,
            )

        dp = next_dp

    # Select the globally best final state.
    best_budget = None
    best_memory = None
    best_score = None
    best_choices = None

    for (used_budget, used_memory), (score, choices) in dp.items():
        if best_score is None:
            best_score = score
            best_budget = used_budget
            best_memory = used_memory
            best_choices = choices
            continue

        if score > best_score:
            best_score = score
            best_budget = used_budget
            best_memory = used_memory
            best_choices = choices

        elif score == best_score and choices < best_choices:
            best_budget = used_budget
            best_memory = used_memory
            best_choices = choices

    return make_result(
        best_score,
        best_budget,
        best_memory,
        list(best_choices),
        major_operations,
    )

# Tests
# ----------------------------------------------------------------------

if __name__ == "__main__":
    # Original behavior: summary disabled by default.
    stages = [
        [
            {"budget": 2, "memory": 3, "score": 8},
            {"budget": 1, "memory": 2, "score": 5},
        ],
        [
            {"budget": 3, "memory": 2, "score": 7},
            {"budget": 2, "memory": 4, "score": 6},
        ],
    ]

    result = multi_stage_resource_allocation(
        stages,
        budget_limit=5,
        memory_limit=5,
    )

    assert result == {
        "max_score": 15,
        "total_budget": 5,
        "total_memory": 5,
        "choice_vector": [0, 0],
    }

    # New feature enabled.
    # Stage 1:
    #   (0,0) -> option 0
    #   (0,0) -> option 1
    # Therefore 2 feasible transitions.
    #
    # Stage 2:
    #   state (2,3): both options are feasible -> 2 operations
    #   state (1,2): both options are feasible -> 2 operations
    #
    # Total = 6 major operations.
    result = multi_stage_resource_allocation(
        stages,
        budget_limit=5,
        memory_limit=5,
        include_operation_summary=True,
    )

    assert result == {
        "max_score": 15,
        "total_budget": 5,
        "total_memory": 5,
        "choice_vector": [0, 0],
        "operation_summary": {
            "major_operations": 6
        },
    }

    # Equal-score tie: lexicographically smallest vector wins.
    stages = [
        [
            {"budget": 1, "memory": 1, "score": 10},
            {"budget": 2, "memory": 1, "score": 10},
        ],
        [
            {"budget": 1, "memory": 1, "score": 5},
            {"budget": 2, "memory": 1, "score": 5},
        ],
    ]

    result = multi_stage_resource_allocation(
        stages,
        budget_limit=4,
        memory_limit=2,
        include_operation_summary=True,
    )

    assert result["max_score"] == 15
    assert result["total_budget"] == 2
    assert result["total_memory"] == 2
    assert result["choice_vector"] == [0, 0]
    assert result["operation_summary"]["major_operations"] == 6

    # Feasible transition counting must exclude resource-limit
    # violations.
    stages = [
        [
            {"budget": 2, "memory": 2, "score": 10},
            {"budget": 5, "memory": 5, "score": 100},
        ],
        [
            {"budget": 1, "memory": 1, "score": 5},
            {"budget": 4, "memory": 4, "score": 20},
        ],
    ]

    result = multi_stage_resource_allocation(
        stages,
        budget_limit=3,
        memory_limit=3,
        include_operation_summary=True,
    )

    # First stage: only option 0 is feasible -> 1 operation.
    # Second stage from (2,2): only option 0 is feasible -> 1 operation.
    assert result == {
        "max_score": 15,
        "total_budget": 3,
        "total_memory": 3,
        "choice_vector": [0, 0],
        "operation_summary": {
            "major_operations": 2
        },
    }

    # No feasible solution still preserves all original fields and,
    # when requested, includes the deterministic summary.
    stages = [
        [
            {"budget": 5, "memory": 5, "score": 10},
        ],
        [
            {"budget": 1, "memory": 1, "score": 5},
        ],
    ]

    result = multi_stage_resource_allocation(
        stages,
        budget_limit=5,
        memory_limit=5,
        include_operation_summary=True,
    )

    # First option is feasible, but no second-stage transition is.
    assert result == {
        "max_score": None,
        "total_budget": None,
        "total_memory": None,
        "choice_vector": [],
        "operation_summary": {
            "major_operations": 1
        },
    }

    # Empty stage.
    stages = [
        [{"budget": 1, "memory": 1, "score": 10}],
        [],
    ]

    result = multi_stage_resource_allocation(
        stages,
        budget_limit=10,
        memory_limit=10,
        include_operation_summary=True,
    )

    assert result == {
        "max_score": None,
        "total_budget": None,
        "total_memory": None,
        "choice_vector": [],
        "operation_summary": {
            "major_operations": 1
        },
    }

    # Empty collection of stages.
    result = multi_stage_resource_allocation(
        [],
        budget_limit=10,
        memory_limit=10,
        include_operation_summary=True,
    )

    assert result == {
        "max_score": 0,
        "total_budget": 0,
        "total_memory": 0,
        "choice_vector": [],
        "operation_summary": {
            "major_operations": 0
        },
    }

    print("All tests passed.")
```
