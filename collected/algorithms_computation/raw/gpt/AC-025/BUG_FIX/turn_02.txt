"""
Multi-Stage Resource Allocation
================================

Problem Restatement
-------------------
There are multiple stages. Exactly one option must be selected from each
stage.

Each option has:
    - a budget cost,
    - a memory cost,
    - a score.

The selected options must satisfy both global limits:

    total budget <= budget_limit
    total memory <= memory_limit

The objective is to maximize total score.

If multiple feasible selections have the same maximum score, the required
tie-breaking rule is:

    choose the lexicographically smallest choice vector.

The returned choice vector contains the zero-based option index selected
from each stage.

Expected Behavior
-----------------
Return exactly:

    {
        "max_score": maximum achievable score,
        "choice_vector": lexicographically smallest optimal choice vector
    }

If no complete selection is feasible:

    {
        "max_score": None,
        "choice_vector": []
    }

The solution must use layered two-resource dynamic programming with
deterministic reconstruction.

Important Bug
-------------
A common incorrect implementation keeps only the first choice reaching a
DP state, or replaces a state only when a strictly larger score is found.

That is incorrect because two choices can reach the same
(budget, memory) state with the same score but different choice vectors.
The lexicographically smaller vector must be retained.

For example:

    stages = [
        [(1, 0, 5), (0, 1, 5)],
        [(0, 1, 5), (1, 0, 5)]
    ]

    budget_limit = 1
    memory_limit = 1

There are two optimal solutions:

    [0, 0] -> budget 1, memory 1, score 10
    [1, 1] -> budget 1, memory 1, score 10

Both have exactly the same resource usage and score.

Therefore the required answer is:

    {
        "max_score": 10,
        "choice_vector": [0, 0]
    }

An implementation that keeps whichever equal-score state is encountered
first can incorrectly return [1, 1]. The fix is to explicitly compare
choice vectors whenever scores are equal.

Correction
----------
The DP is layered by stage.

For every reachable state:

    (used_budget, used_memory)

store:

    (best_score, lexicographically_smallest_choice_vector)

When adding an option:
    1. Reject it if either resource limit would be exceeded.
    2. Replace the state if the new score is larger.
    3. If scores are equal, replace it only when the new choice vector is
       lexicographically smaller.

Finally, compare all feasible final states using the same deterministic
rules.

This preserves the original requirements and changes no unrelated behavior.
"""

from typing import Any, Dict, List, Tuple


Option = Tuple[int, int, int]


def multi_stage_resource_allocation(
    stages: List[List[Option]],
    budget_limit: int,
    memory_limit: int,
) -> Dict[str, Any]:
    """
    Return the maximum-score feasible allocation and its deterministic
    lexicographically smallest choice vector.

    Each option is:
        (budget_cost, memory_cost, score)

    Exactly one option is selected from every stage.
    """

    if budget_limit < 0 or memory_limit < 0:
        raise ValueError("Resource limits must be non-negative.")

    for stage_index, stage in enumerate(stages):
        if not stage:
            raise ValueError(
                f"Stage {stage_index} must contain at least one option."
            )

        for option_index, option in enumerate(stage):
            if len(option) != 3:
                raise ValueError(
                    f"Option {option_index} in stage {stage_index} "
                    "must be (budget, memory, score)."
                )

            budget, memory, _ = option

            if budget < 0 or memory < 0:
                raise ValueError(
                    "Budget and memory costs must be non-negative."
                )

    # State:
    # (used_budget, used_memory)
    #
    # Value:
    # (best_score, lexicographically smallest choice vector)
    dp: Dict[Tuple[int, int], Tuple[int, Tuple[int, ...]]] = {
        (0, 0): (0, ())
    }

    # Process one stage at a time.
    for stage in stages:
        next_dp: Dict[Tuple[int, int], Tuple[int, Tuple[int, ...]]] = {}

        for (used_budget, used_memory), (score, choices) in dp.items():
            for option_index, (
                option_budget,
                option_memory,
                option_score,
            ) in enumerate(stage):

                new_budget = used_budget + option_budget
                new_memory = used_memory + option_memory

                if new_budget > budget_limit:
                    continue

                if new_memory > memory_limit:
                    continue

                new_score = score + option_score
                new_choices = choices + (option_index,)

                state = (new_budget, new_memory)
                existing = next_dp.get(state)

                if existing is None:
                    next_dp[state] = (new_score, new_choices)
                    continue

                existing_score, existing_choices = existing

                # Primary rule: maximize score.
                if new_score > existing_score:
                    next_dp[state] = (new_score, new_choices)

                # Critical deterministic tie rule:
                # same score + same resource usage -> smallest vector wins.
                elif (
                    new_score == existing_score
                    and new_choices < existing_choices
                ):
                    next_dp[state] = (new_score, new_choices)

        dp = next_dp

        if not dp:
            return {
                "max_score": None,
                "choice_vector": [],
            }

    # Deterministically choose the best final state.
    best_score = None
    best_choices = None

    for score, choices in dp.values():
        if best_score is None:
            best_score = score
            best_choices = choices
        elif score > best_score:
            best_score = score
            best_choices = choices
        elif score == best_score and choices < best_choices:
            best_choices = choices

    if best_score is None:
        return {
            "max_score": None,
            "choice_vector": [],
        }

    return {
        "max_score": best_score,
        "choice_vector": list(best_choices),
    }


# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------

def _run_tests() -> None:

    # Test 1: Demonstrates the deterministic tie-breaking bug.
    #
    # Both [0, 0] and [1, 1] have:
    #   budget = 1
    #   memory = 1
    #   score  = 10
    #
    # [0, 0] must win because it is lexicographically smaller.
    stages = [
        [(1, 0, 5), (0, 1, 5)],
        [(0, 1, 5), (1, 0, 5)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=1,
        memory_limit=1,
    ) == {
        "max_score": 10,
        "choice_vector": [0, 0],
    }

    # Test 2: Maximum score takes priority over lexicographic order.
    stages = [
        [(1, 0, 5), (0, 1, 8)],
        [(0, 1, 5), (1, 0, 8)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=1,
        memory_limit=1,
    ) == {
        "max_score": 16,
        "choice_vector": [1, 1],
    }

    # Test 3: A locally lower-scoring choice can enable the global optimum.
    stages = [
        [(4, 4, 10), (2, 2, 7)],
        [(2, 2, 10), (3, 3, 20)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=5,
        memory_limit=5,
    ) == {
        "max_score": 27,
        "choice_vector": [1, 1],
    }

    # Test 4: No feasible complete allocation.
    stages = [
        [(5, 1, 10)],
        [(1, 5, 10)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=5,
        memory_limit=5,
    ) == {
        "max_score": None,
        "choice_vector": [],
    }

    # Test 5: Exactly one option is required from every stage.
    stages = [
        [(0, 0, -5), (1, 0, -2)],
        [(0, 0, -3), (1, 0, -1)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=0,
        memory_limit=0,
    ) == {
        "max_score": -8,
        "choice_vector": [0, 0],
    }

    # Test 6: Zero-resource limits with zero-cost options.
    stages = [
        [(0, 0, 3), (1, 0, 100)],
        [(0, 0, 4), (0, 1, 100)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=0,
        memory_limit=0,
    ) == {
        "max_score": 7,
        "choice_vector": [0, 0],
    }

    # Test 7: Multiple equal-score choices at different states.
    stages = [
        [(1, 1, 5), (0, 2, 5), (2, 0, 5)],
        [(1, 1, 5), (0, 2, 5), (2, 0, 5)],
    ]

    assert multi_stage_resource_allocation(
        stages,
        budget_limit=2,
        memory_limit=2,
    ) == {
        "max_score": 10,
        "choice_vector": [0, 0],
    }


if __name__ == "__main__":
    _run_tests()
    print("All tests passed.")
