from typing import List, Dict


def minimum_coin_construction(
    denominations: List[int],
    target: int
) -> Dict[str, object]:
    """
    Given coin denominations and a target amount, find a
    minimum-number coin combination. Coins may be reused.

    Returns:
        {"count": minimum_number_of_coins, "coins": [...]}

    If impossible:
        {"count": -1, "coins": []}

    Deterministic tie-breaking:
        For equal minimum counts, prefer the smaller denomination.
    """

    if target < 0:
        raise ValueError("target must be non-negative")

    if any(coin <= 0 for coin in denominations):
        raise ValueError("all coin denominations must be positive")

    # Remove duplicates and sort for deterministic processing.
    coins = sorted(set(denominations))

    # Smallest permitted target.
    if target == 0:
        return {
            "count": 0,
            "coins": []
        }

    # Empty coin set: no positive target can be constructed.
    if not coins:
        return {
            "count": -1,
            "coins": []
        }

    # dp[amount] = minimum number of coins needed.
    INF = target + 1
    dp = [INF] * (target + 1)

    # choice[amount] = coin selected for reconstruction.
    choice = [-1] * (target + 1)

    dp[0] = 0

    # One-dimensional unbounded coin-change DP.
    for amount in range(1, target + 1):
        for coin in coins:
            if coin > amount:
                break

            if dp[amount - coin] == INF:
                continue

            candidate = dp[amount - coin] + 1

            if candidate < dp[amount]:
                dp[amount] = candidate
                choice[amount] = coin

            # Deterministic tie-breaking:
            # prefer the smaller denomination.
            elif candidate == dp[amount]:
                if choice[amount] == -1 or coin < choice[amount]:
                    choice[amount] = coin

    # Target is unreachable.
    if dp[target] == INF:
        return {
            "count": -1,
            "coins": []
        }

    # Deterministic reconstruction.
    result = []
    amount = target

    while amount > 0:
        coin = choice[amount]

        if coin == -1:
            return {
                "count": -1,
                "coins": []
            }

        result.append(coin)
        amount -= coin

    return {
        "count": dp[target],
        "coins": result
    }


# ---------------------------------------------------------
# Tests for normal and difficult valid cases
# ---------------------------------------------------------

def run_tests() -> None:

    # Smallest permitted input:
    assert minimum_coin_construction([], 0) == {
        "count": 0,
        "coins": []
    }

    # Empty coin structure with positive target.
    assert minimum_coin_construction([], 1) == {
        "count": -1,
        "coins": []
    }

    # Single coin denomination, reachable.
    assert minimum_coin_construction([1], 1) == {
        "count": 1,
        "coins": [1]
    }

    # Single denomination with unlimited reuse.
    assert minimum_coin_construction([2], 6) == {
        "count": 3,
        "coins": [2, 2, 2]
    }

    # Single denomination, unreachable target.
    assert minimum_coin_construction([2], 3) == {
        "count": -1,
        "coins": []
    }

    # Standard minimum-coin case.
    assert minimum_coin_construction([1, 3, 4], 6) == {
        "count": 2,
        "coins": [3, 3]
    }

    # Deterministic tie:
    # 6 = 3 + 3 and 6 = 2 + 4.
    # Both use two coins, so smaller denomination 2 is preferred.
    assert minimum_coin_construction([2, 3, 4], 6) == {
        "count": 2,
        "coins": [2, 4]
    }

    # Impossible target.
    assert minimum_coin_construction([4, 6], 5) == {
        "count": -1,
        "coins": []
    }

    # Duplicate denominations.
    assert minimum_coin_construction([1, 1, 3, 3], 6) == {
        "count": 2,
        "coins": [3, 3]
    }

    # Zero target with available coins.
    assert minimum_coin_construction([1, 5, 10], 0) == {
        "count": 0,
        "coins": []
    }

    # Reusable denominations.
    assert minimum_coin_construction([1, 2, 5], 11) == {
        "count": 3,
        "coins": [5, 5, 1]
    }

    # Disconnected/unreachable structure.
    assert minimum_coin_construction([2, 4, 8], 7) == {
        "count": -1,
        "coins": []
    }

    print("All tests passed.")


if __name__ == "__main__":
    run_tests()
