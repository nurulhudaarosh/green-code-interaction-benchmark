The original **Minimum Coin Construction** problem is to find a minimum-number combination of reusable positive coin denominations that sums to a non-negative target. If impossible, return `(-1, [])`. The existing deterministic tie-breaking rule is preserved: **when equally optimal choices exist, prefer the smaller denomination**.

The difficult valid cases are explicitly handled below:

* Smallest permitted input: `coins=[1]`, `target=1`.
* Empty coin list: valid input; returns `(-1, [])` for a positive target.
* Empty/zero target: `target=0` returns `(0, [])`.
* Optional `operation_summary` behavior from the previous version is preserved.

```python
"""
Minimum Coin Construction

Original problem:
Given positive coin denominations and a non-negative target amount,
find a combination of coins whose sum equals the target while using
the minimum possible number of coins.

Coins may be reused unlimited times.

Original required output:
    (minimum_coin_count, coin_combination)

If impossible:
    (-1, [])

Deterministic tie-breaking:
    If multiple minimum-coin solutions exist, prefer the smaller
    denomination at each DP decision.

Algorithm:
    One-dimensional unbounded coin-change DP with deterministic
    reconstruction.

Optional feature:
    If include_operation_summary=True, return:

    {
        "count": count,
        "combination": combination,
        "operation_summary": {
            "dp_decisions": number_of_DP_decisions
        }
    }

If include_operation_summary=False, the original output is preserved.

Valid difficult cases:
    1. Smallest permitted input: [1], target=1
    2. Empty coin list: []
    3. Empty target / zero target: target=0
    4. Target that cannot be constructed
"""


from typing import List, Tuple, Union, Dict, Any


def minimum_coin_construction(
    coins: List[int],
    target: int,
    include_operation_summary: bool = False
) -> Union[Tuple[int, List[int]], Dict[str, Any]]:
    """
    Find a minimum-coin construction for target.

    Parameters:
        coins:
            List of positive integer denominations. An empty list is
            allowed.

        target:
            Non-negative target amount.

        include_operation_summary:
            False -> return the original:
                (count, combination)

            True -> return:
                {
                    "count": count,
                    "combination": combination,
                    "operation_summary": {
                        "dp_decisions": ...
                    }
                }

    Returns:
        Minimum coin count and deterministic combination.

        If impossible:
            (-1, [])

    Tie-breaking:
        Smaller denomination wins when multiple choices give the same
        minimum number of coins.
    """

    # ---------------------------------------------------------------
    # Validate target.
    # ---------------------------------------------------------------

    if target < 0:
        raise ValueError("target must be non-negative")

    # ---------------------------------------------------------------
    # Difficult case 1:
    # Empty/zero target.
    #
    # Zero coins are needed to construct amount 0, regardless of whether
    # the denomination structure is empty or non-empty.
    # ---------------------------------------------------------------

    if target == 0:
        if include_operation_summary:
            return {
                "count": 0,
                "combination": [],
                "operation_summary": {
                    "dp_decisions": 0
                }
            }

        return 0, []

    # ---------------------------------------------------------------
    # Normalize denominations.
    #
    # An empty coin list is valid. For a positive target it simply
    # means that construction is impossible.
    # ---------------------------------------------------------------

    unique_coins = sorted(set(coins))

    if any(coin <= 0 for coin in unique_coins):
        raise ValueError("coin denominations must be positive")

    usable_coins = [
        coin for coin in unique_coins
        if coin <= target
    ]

    # ---------------------------------------------------------------
    # Difficult case 2:
    # Empty coin list / no usable denominations.
    # ---------------------------------------------------------------

    if not usable_coins:
        if include_operation_summary:
            return {
                "count": -1,
                "combination": [],
                "operation_summary": {
                    "dp_decisions": 0
                }
            }

        return -1, []

    # ---------------------------------------------------------------
    # One-dimensional unbounded coin-change DP.
    # ---------------------------------------------------------------

    INF = target + 1

    # dp[amount] = minimum number of coins needed to make amount.
    dp = [INF] * (target + 1)

    # choice[amount] = denomination chosen for reconstruction.
    choice = [None] * (target + 1)

    dp[0] = 0

    # Count every valid DP transition considered.
    dp_decisions = 0

    for amount in range(1, target + 1):

        for coin in usable_coins:

            if coin > amount:
                break

            dp_decisions += 1

            previous = dp[amount - coin]

            if previous == INF:
                continue

            candidate = previous + 1

            # Strictly better solution.
            if candidate < dp[amount]:
                dp[amount] = candidate
                choice[amount] = coin

            # Equal-quality solution.
            #
            # usable_coins is sorted, and the explicit comparison makes
            # the tie-breaking rule clear and independent of input order.
            elif candidate == dp[amount]:
                if choice[amount] is None or coin < choice[amount]:
                    choice[amount] = coin

    # ---------------------------------------------------------------
    # Impossible target.
    # ---------------------------------------------------------------

    if dp[target] == INF:
        if include_operation_summary:
            return {
                "count": -1,
                "combination": [],
                "operation_summary": {
                    "dp_decisions": dp_decisions
                }
            }

        return -1, []

    # ---------------------------------------------------------------
    # Deterministic reconstruction.
    # ---------------------------------------------------------------

    combination = []
    current = target

    while current > 0:
        coin = choice[current]

        # Defensive safeguard.
        if coin is None:
            if include_operation_summary:
                return {
                    "count": -1,
                    "combination": [],
                    "operation_summary": {
                        "dp_decisions": dp_decisions
                    }
                }

            return -1, []

        combination.append(coin)
        current -= coin

    # ---------------------------------------------------------------
    # Preserve original output unless the optional feature is requested.
    # ---------------------------------------------------------------

    if not include_operation_summary:
        return dp[target], combination

    return {
        "count": dp[target],
        "combination": combination,
        "operation_summary": {
            "dp_decisions": dp_decisions
        }
    }


# ===================================================================
# Tests
# ===================================================================

def run_tests() -> None:

    # ---------------------------------------------------------------
    # 1. Smallest permitted input.
    #
    # One positive denomination and the smallest positive target.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [1], 1
    ) == (1, [1])

    # Same smallest case with the optional summary enabled.
    result = minimum_coin_construction(
        [1], 1,
        include_operation_summary=True
    )

    assert result == {
        "count": 1,
        "combination": [1],
        "operation_summary": {
            "dp_decisions": 1
        }
    }

    # ---------------------------------------------------------------
    # 2. Empty coin structure.
    #
    # Empty denominations are allowed. A positive target cannot be
    # constructed without any coins.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [], 1
    ) == (-1, [])

    # Empty coins with operation summary.
    result = minimum_coin_construction(
        [], 1,
        include_operation_summary=True
    )

    assert result == {
        "count": -1,
        "combination": [],
        "operation_summary": {
            "dp_decisions": 0
        }
    }

    # ---------------------------------------------------------------
    # 3. Empty target / zero target.
    #
    # No coins are needed to construct zero.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [], 0
    ) == (0, [])

    assert minimum_coin_construction(
        [1, 2, 5], 0
    ) == (0, [])

    result = minimum_coin_construction(
        [], 0,
        include_operation_summary=True
    )

    assert result == {
        "count": 0,
        "combination": [],
        "operation_summary": {
            "dp_decisions": 0
        }
    }

    # ---------------------------------------------------------------
    # 4. Normal minimum-coin case.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [1, 3, 4], 6
    ) == (2, [3, 3])

    # ---------------------------------------------------------------
    # 5. Impossible positive target.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [2, 4], 7
    ) == (-1, [])

    # ---------------------------------------------------------------
    # 6. Coins can be reused.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [2, 5], 10
    ) == (2, [5, 5])

    # ---------------------------------------------------------------
    # 7. Deterministic tie-breaking.
    #
    # For target 6:
    #     [3, 3] -> 2 coins
    #     [2, 4] -> 2 coins
    #
    # Both are optimal. The smaller denomination is preferred,
    # resulting in [2, 4].
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [4, 3, 2], 6
    ) == (2, [2, 4])

    # Input order must not change the result.
    assert minimum_coin_construction(
        [2, 4, 3], 6
    ) == (2, [2, 4])

    assert minimum_coin_construction(
        [3, 2, 4], 6
    ) == (2, [2, 4])

    # ---------------------------------------------------------------
    # 8. Duplicate denominations.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [1, 1, 3, 3], 6
    ) == (2, [3, 3])

    # ---------------------------------------------------------------
    # 9. Denominations larger than target.
    #
    # No usable coin exists.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [10, 20, 50], 7
    ) == (-1, [])

    # ---------------------------------------------------------------
    # 10. Mixed usable and unusable denominations.
    # ---------------------------------------------------------------

    assert minimum_coin_construction(
        [1, 5, 20, 100], 11
    ) == (3, [5, 5, 1])

    # ---------------------------------------------------------------
    # 11. Verify reconstruction.
    # ---------------------------------------------------------------

    count, combination = minimum_coin_construction(
        [1, 3, 4], 11
    )

    assert count == 3
    assert len(combination) == count
    assert sum(combination) == 11

    # ---------------------------------------------------------------
    # 12. Optional operation summary does not change the solution.
    # ---------------------------------------------------------------

    original = minimum_coin_construction(
        [1, 5, 10], 12
    )

    extended = minimum_coin_construction(
        [1, 5, 10],
        12,
        include_operation_summary=True
    )

    assert original == (
        extended["count"],
        extended["combination"]
    )

    assert extended["operation_summary"]["dp_decisions"] > 0

    # ---------------------------------------------------------------
    # 13. Negative target remains invalid.
    # ---------------------------------------------------------------

    try:
        minimum_coin_construction([1, 2, 5], -1)
        assert False, "Expected ValueError for negative target"
    except ValueError:
        pass

    # ---------------------------------------------------------------
    # 14. Non-positive denomination remains invalid.
    # ---------------------------------------------------------------

    try:
        minimum_coin_construction([1, 0, 2], 5)
        assert False, "Expected ValueError for zero denomination"
    except ValueError:
        pass

    try:
        minimum_coin_construction([1, -2, 3], 5)
        assert False, "Expected ValueError for negative denomination"
    except ValueError:
        pass

    print("All tests passed.")


if __name__ == "__main__":
    run_tests()
```
