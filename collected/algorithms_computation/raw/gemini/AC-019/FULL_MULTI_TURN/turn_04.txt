"""
Minimum-Cost Grid Path Solver with Boundary Value Handling and Verification Tests

Original Problem Restatement:
Given an m x n grid filled with non-negative costs, move only Right or Down 
from top-left (0, 0) to bottom-right (m-1, n-1). Find the minimum cumulative 
path cost and construct one deterministic optimal path using dynamic programming 
with predecessor tracking and deterministic tie handling.

Boundary & Edge Cases Explicitly Handled:
1. Empty grid or empty sub-rows (`[]` or `[[]]`).
2. Single-cell grid (`1 x 1`).
3. Single-row grid (`1 x n`).
4. Single-column grid (`m x 1`).
5. Grids with 0-cost cells (boundary value for non-negative costs).
6. Grids with large dimensions or large cost values.
7. Optional operation summary flag maintaining full backward compatibility.
"""


def min_path_cost(
    grid: list[list[int]], include_summary: bool = False
) -> tuple[int, list[tuple[int, int]]] | tuple[
    int, list[tuple[int, int]], dict[str, int]
]:
    """
    Computes minimum path cost and deterministic path with boundary handling.

    Deterministic Tie-Breaking Rule:
    When costs from LEFT and UP are equal, prioritize coming from LEFT
    (horizontal movement before vertical movement).
    """
    # Case 1: Empty grid check
    if not grid or not grid[0]:
        summary = {
            "total_cells": 0,
            "boundary_initializations": 0,
            "dp_transitions": 0,
            "ties_encountered": 0,
            "backtrack_steps": 0,
            "total_major_operations": 0,
        }
        return (0, [], summary) if include_summary else (0, [])

    m, n = len(grid), len(grid[0])

    dp = [[0] * n for _ in range(m)]
    parent = [[None] * n for _ in range(m)]

    boundary_inits = 1
    dp_transitions = 0
    ties_encountered = 0

    dp[0][0] = grid[0][0]

    # Case 2 & 3: Single row initialization / First row boundary
    for j in range(1, n):
        dp[0][j] = dp[0][j - 1] + grid[0][j]
        parent[0][j] = (0, j - 1)
        boundary_inits += 1

    # Case 2 & 4: Single column initialization / First column boundary
    for i in range(1, m):
        dp[i][0] = dp[i - 1][0] + grid[i][0]
        parent[i][0] = (i - 1, 0)
        boundary_inits += 1

    # Fill DP table for general cells (skipped for 1xN, Mx1, and 1x1 grids)
    for i in range(1, m):
        for j in range(1, n):
            dp_transitions += 1
            from_up = dp[i - 1][j]
            from_left = dp[i][j - 1]

            if from_left == from_up:
                ties_encountered += 1

            # Prioritize LEFT over UP on equality (deterministic tie-breaking)
            if from_left <= from_up:
                dp[i][j] = grid[i][j] + from_left
                parent[i][j] = (i, j - 1)
            else:
                dp[i][j] = grid[i][j] + from_up
                parent[i][j] = (i - 1, j)

    # Reconstruct optimal path from parent matrix
    path = []
    curr = (m - 1, n - 1)
    backtrack_steps = 0

    while curr is not None:
        path.append(curr)
        curr = parent[curr[0]][curr[1]]
        backtrack_steps += 1

    path.reverse()

    min_cost = dp[m - 1][n - 1]

    if include_summary:
        operation_summary = {
            "total_cells": m * n,
            "boundary_initializations": boundary_inits,
            "dp_transitions": dp_transitions,
            "ties_encountered": ties_encountered,
            "backtrack_steps": backtrack_steps,
            "total_major_operations": boundary_inits
            + dp_transitions
            + backtrack_steps,
        }
        return min_cost, path, operation_summary

    return min_cost, path


# =====================================================================
# Verification Unit Tests
# =====================================================================


def test_empty_grid():
    cost, path = min_path_cost([])
    assert cost == 0 and path == []

    cost, path = min_path_cost([[]])
    assert cost == 0 and path == []


def test_single_cell():
    grid = [[5]]
    cost, path = min_path_cost(grid)
    assert cost == 5
    assert path == [(0, 0)]


def test_single_row():
    grid = [[1, 2, 3, 4]]
    cost, path = min_path_cost(grid)
    assert cost == 10
    assert path == [(0, 0), (0, 1), (0, 2), (0, 3)]


def test_single_column():
    grid = [[1], [2], [3], [4]]
    cost, path = min_path_cost(grid)
    assert cost == 10
    assert path == [(0, 0), (1, 0), (2, 0), (3, 0)]


def test_zero_costs():
    grid = [[0, 0], [0, 0]]
    cost, path = min_path_cost(grid)
    assert cost == 0
    # Left preferred over Up -> (0,0) -> (0,1) -> (1,1)
    assert path == [(0, 0), (0, 1), (1, 1)]


def test_tie_handling_preference():
    grid = [[1, 2], [2, 1]]
    cost, path = min_path_cost(grid)
    assert cost == 4
    # Equal paths: (0,0)->(0,1)->(1,1) vs (0,0)->(1,0)->(1,1)
    # Prefers coming from LEFT: (0,0)->(0,1)->(1,1)
    assert path == [(0, 0), (0, 1), (1, 1)]


def test_large_values():
    grid = [[10**9, 10**9], [10**9, 10**9]]
    cost, path = min_path_cost(grid)
    assert cost == 3 * (10**9)
    assert path == [(0, 0), (0, 1), (1, 1)]


def test_operation_summary_flag():
    grid = [[1, 2], [2, 1]]
    cost, path, summary = min_path_cost(grid, include_summary=True)
    assert cost == 4
    assert summary["total_cells"] == 4
    assert summary["ties_encountered"] == 1


def run_all_tests():
    test_empty_grid()
    test_single_cell()
    test_single_row()
    test_single_column()
    test_zero_costs()
    test_tie_handling_preference()
    test_large_values()
    test_operation_summary_flag()
    print("All test cases passed successfully!")


if __name__ == "__main__":
    run_all_tests()
